# T3. 甜筒的排列树

## 子任务 $1$

$n=8$，直接枚举所有 $p$ 即可，复杂度 $O(n^2n!)$

## 子任务 $2$

$n=100$，考虑恰好为 $k$ 是不好处理的，改为用存在 $0,1,\dots,k-1$ 的方案数减去存在 $0,1,\dots,k$ 的方案数。

接着考虑当 $k$ 固定时该怎么做。$k=0$ 是比较特殊的，因为其要求所有叶子要选 $0$，特判是链的话答案为 $(n-1)!$，否则为 $0$。

而当 $k\neq0$ 时，考虑 DP，设 $f_{u,j}$ 表示考虑以 $u$ 为根的子树，其中 $w$ 存在 $1,\dots j$ 的方案数。对每个点 $v$ 向其父亲转移即可，当 $j$ 不变时，要求 $j\sim k$ 的数均不出现在 $u$ 的子树中，方程为：

$$f_{u,j}\gets f_{u,j}+A_{n-siz_v-k+j}^{siz_u-siz_v}f_{v,j}$$

而当 $j$ 增加时，要求 $j+1\sim k$ 的数均不出现在 $u$ 的子树中，$j$ 出现在 $u$ 子树内，$v$ 子树外，方程为：
$$f_{u,j+1}\gets f_{u,j+1}+(siz_u-siz_v)A_{n-siz_v-k+j}^{siz_u-siz_v-1}f_{v,j}$$

暴力转移即可，复杂度为 $O(n\sum maxdep_i)$，其中 $maxdep_i$ 表示每个点到其子树中所有点的最远距离。也允许 $O(n^4)$ 的未知算法通过。

## 子任务 $3$

对于链的性质，$k=0$ 时答案为 $(n-1)!$，否则要求 $p_n\neq0$，存在 $0,1,\dots,k-1$ 的方案数为 $\binom{n-1}{k}(n-k)!$，复杂度 $O(n)$，但为了放一些神秘的 $O(n^2)$ DP 过去只开了 $3\times10^3​$。

## 子任务 $4\sim5$

考虑优化 DP，发现转移方程中与 $k$ 有关的只是那个排列数，其限定了 $k+1\sim n$ 中数的位置，但这些数的位置只需要在最后贡献答案的时候计算，因此每个点向答案贡献 $ans_k\gets ans_k+(n-k)!(f_{u,k}-\sum f_{v,k})$，减去 $\sum f_{v,k}$ 是为了去重，复杂度 $O(\sum maxdep_i)$。

## 子任务 $6$

由于数据随机，$maxdep_i$ 只有 $O(\log_2n)$ 级别，做法和上一部分相同。

## 小巧思

为了致敬 [P15652 [省选联考 2026] 排列游戏 / perm](https://www.luogu.com.cn/problem/P15652) 不特判 $p_0=0$ 会挂掉 $90$ 分，本题开启子任务捆绑，除随机数据外的每个子任务都有链的数据，如果链的部分写挂可能会获得 $25$ 分好成绩。

#include<iostream>
#include<vector>
#include<algorithm>
#define rep(x,l,r) for(int x=(l);x<=(r);x++)
#define per(x,l,r) for(int x=(l);x>=(r);x--)
using namespace std;
typedef long long ll;
const int maxn=1e5+5,mod=1e9+7;
vector<int> T[maxn];
int ap[maxn],siz[maxn],mxdep[maxn],dep[maxn],in,cnt;
bool vis[maxn];
vector<ll> f[maxn];
ll fac[maxn],inv[maxn],ans[maxn];
inline ll qpow(ll a,ll b){
    ll res=1;
    while(b){
        if(b&1)res=res*a%mod;
        a=a*a%mod;
        b>>=1;
    }
    return res;
}
inline ll C(int m,int n){
    if(m<n||m<0||n<0)return 0;
    return fac[m]*inv[n]%mod*inv[m-n]%mod;
}
inline ll A(int m,int n){
    if(m<n||m<0||n<0)return 0;
    return fac[m]*inv[m-n]%mod;
}
inline void getsiz(int x,int fa){
    dep[x]=dep[fa]+1;
    mxdep[x]=siz[x]=1;
    for(int v:T[x])if(v!=fa){
        getsiz(v,x);
        siz[x]+=siz[v];
        mxdep[x]=max(mxdep[x],mxdep[v]+1);
    }
    if(siz[x]==1)cnt++;
    f[x].resize(mxdep[x]+10);
}
inline void dfs(int x,int fa){
    f[x][1]=1;
    for(int v:T[x])if(v!=fa){
        dfs(v,x);
        rep(v1,1,mxdep[v])if(f[v][v1]){
            f[x][v1]=(f[x][v1]+f[v][v1])%mod;
            f[x][v1+1]=(f[x][v1+1]+f[v][v1]*(siz[x]-siz[v]))%mod;
            ans[v1]=(ans[v1]-f[v][v1]*A(in-v1-1,siz[x]-v1)%mod*fac[in-siz[x]]%mod+mod)%mod;
        }
    }
    rep(v1,1,mxdep[x]){
        ans[v1]=(ans[v1]+f[x][v1]*A(in-v1-1,siz[x]-v1)%mod*fac[in-siz[x]])%mod;
    }
}
inline ll cal(int k){
    return (C(in-1,k)*fac[in-k])%mod;
}
int main(){
    freopen("perm.in","r",stdin);
    freopen("perm.out","w",stdout);
    cin>>in;
    fac[0]=1;
    rep(v1,1,in)fac[v1]=fac[v1-1]*v1%mod;
    inv[in]=qpow(fac[in],mod-2);
    per(v1,in,1)inv[v1-1]=inv[v1]*v1%mod;
    rep(v1,1,in-1){
        int iu,iv,cone;
        scanf("%d %d",&iu,&iv);
        T[iu].push_back(iv);
        T[iv].push_back(iu);
    }
    getsiz(1,0);
    if(cnt==1){
        printf("%lld\n",fac[in-1]);
        rep(v1,1,in){
            printf("%lld\n",(cal(v1)-cal(v1+1)+mod)%mod);
        }
    }
    else{
        dfs(1,0);
        ans[0]=(cnt>1?fac[in]:fac[in-1]);
        printf("%lld\n",0ll);
        rep(v1,1,in){
            printf("%lld\n",(ans[v1-1]-ans[v1]+mod)%mod);
        }
    }
    return 0;
}
