# 题目描述

[原题链接](https://www.luogu.com.cn/problem/P3384)。

如题，已知一棵包含 $n$ 个结点的树（连通且无环），每个节点上包含一个数值，需要支持以下操作：

- `1 x y z`，表示将树从 $x$ 到 $y$ 结点最短路径上所有节点的值都加上 $z$。
- `2 x y`，表示求树从 $x$ 到 $y$ 结点最短路径上所有节点的值之和。
- `3 x z`，表示将以 $x$ 为根节点的子树内所有节点值都加上 $z$。
- `4 x` 表示求以 $x$ 为根节点的子树内所有节点值之和。

# 题目分析

题目的查找与修改操作很像线段树的区间操作，如果能把一棵树分成几条链，之后就可以像线段树一样修改与查询了。

## 树链剖分的定义

如图，我们可以把一棵树分成几条链，其中绿色边连接的就是一条链。

![](https://cdn.luogu.com.cn/upload/image_hosting/d3gvj74y.png)

- 重儿子：父节点的所有儿子中子树节点数目最多的儿子。
- 轻儿子：父结点中除重儿子外的所有儿子。
- 重边：父结点和重儿子连成的边。
- 轻边：父结点和轻儿子连成的边。
- 重链：由多条重边连接而成的路径。
1. 整棵树会被剖分成若干条重链。
2. 轻儿子一定是每条重链的顶点。
3. 任意一条路径被切分成不超过 $\log n$ 条链。

- `fa[i]`，表示 $i$ 的父节点。
- `dep[i]`，表示 $i$ 节点的深度。
- `son[i]`，表示 $i$ 节点的重儿子。
- `sz[i]`，表示以 $i$ 为根的子树的大小。
- `top[i]`，表示 $i$ 的链头。
- `id[i]`，表示 $i$ 节点在 dfs 序中的编号。
- `nw[i]`，表示 $i$ 节点在 dfs 序中的点权值。

其中的 dfs 序可以理解为下图（就是先遍历重儿子，再遍历轻儿子。特别的，我们默认根节点为轻儿子）。

![](https://cdn.luogu.com.cn/upload/image_hosting/97x0lhgu.png)

## 如何求出树链剖分需要的数组

第一遍 `dfs1()`，求出 `fa`，`dep`，`son`，`sz`。

第二遍 `dfs2()`，求出 `top`，`id`，`nw`。

具体操作如下：

```cpp
void dfs1(int x, int father) { //第一遍dfs1
	fa[x] = father, dep[x] = dep[father] + 1, sz[x] = 1; //得到fa[x]，dep[x]，初始化sz[x]
	for (auto v:linker[x]) {
		if(v == father) continue; 
		dfs1(v, x); //先遍历子节点
		sz[x] += sz[v]; //树的大小
		if(sz[son[x]] < sz[v]) son[x] = v; //求出重儿子
	}
}

void dfs2(int x, int t) { //t是指top（链头）
	top[x] = t, id[x] = ++cnt, nw[cnt] = w[x]; //得到top，id，nw
	if(!son[x]) return; //没有重儿子，return
	dfs2(son[x], t); //否则先遍历重儿子
	for (auto v:linker[x]) {
		if(v == fa[x] || v == son[x]) continue;
		dfs2(v, v); //轻儿子自己为链头
	}
}
```

## 如何使用树链剖分

- 搞定 1 操作：

  > 由于任意一条路径被切分成不超过 $\log n$ 条链。因此我们可以把每一条链对应的 dfs 序（即 `id` 数组）进行区间修改。
  >
  > 具体操作：
  >
  > - 给出 $x$ 和 $y$，让深度大的节点修改自己到链头的位置。
  > - 深度较大的那个节点跳到链头的父节点。
  > - 重复上述操作，直到 $x$ 和 $y$ 在同一条链上。此时修改 $x\sim y$ 的区间即可。
  ```cpp
  void update_path(int x, int y, int k) {
  	while(top[x] != top[y]) { //链头不相等，那么接着循环
  		if(dep[top[x]] < dep[top[y]]) swap(x, y); //修改深度较大的节点
  		update(1, 1, n, id[top[x]], id[x], k); //修改自己到链头的区间
  		x = fa[top[x]]; //跳到链头的父节点
  	}
  	if(dep[x] < dep[y]) swap(x, y);
  	update(1, 1, n, id[y], id[x], k); //修改x~y的区间
  }
  ```

- 搞定 2 操作：

  > 与操作 1 类似，把修改代码改为查询代码即可。

  ```cpp
  int query_path(int x, int y) {
  	int res = 0;
  	while(top[x] != top[y]) {
  		if(dep[top[x]] < dep[top[y]]) swap(x, y);
  		res += query(1, 1, n, id[top[x]], id[x]);
  		x = fa[top[x]];
  	}
  	if(dep[x] < dep[y]) swap(x, y);
  	res += query(1, 1, n, id[y], id[x]);
  	return res;
  }
  ```

- 搞定 3 操作：

  > 由于我们在定义 dfs 序时，都是先遍历完整颗子树在遍历别的子树。也就是说，子树内的下标都是连续的，可以直接修改。

  ```cpp
  void update_tree(int x, int k) {
  	update(1, 1, n, id[x], id[x] + sz[x] - 1, k); //节点编号~节点编号+子树大小-1
  }
  ```
  
- 搞定 4 操作：

  > 与操作 3 类似，把修改代码改为查询代码即可。

  ```cpp
  int query_tree(int x) {
  	return query(1, 1, n, id[x], id[x] + sz[x] - 1);
  }
  ```

注意其中的查询、修改操作都是使用线段树完成的。之所以没有提到线段树，是因为重点不在这里。

//以上都是快读快写

int n, m, root, P, cnt;
int w[maxn], sumv[4 * maxn], lazy[4 * maxn];
int fa[maxn], dep[maxn], son[maxn], sz[maxn]; //父亲节点，节点深度，重儿子、子树大小 
int top[maxn], id[maxn], nw[maxn]; //树链链头，dfs序节点编号、新点权 
vector<int>linker[maxn];
void pushup(int id) {
	sumv[id] = sumv[id * 2] + sumv[id * 2 + 1];
}

void pushdown(int id, int l, int r) {
	if(lazy[id] != 0) {
		int mid = (l + r) >> 1;
		sumv[id * 2] = sumv[id * 2] + lazy[id] * (mid - l + 1);
		sumv[id * 2 + 1] = sumv[id * 2 + 1] + lazy[id] * (r - mid);
		lazy[id * 2] += lazy[id];
		lazy[id * 2 + 1] += lazy[id];
		lazy[id] = 0;
	}
}

void build(int id, int l, int r) {
	if(l == r) {
		sumv[id] = nw[l];
		return;
	}
	int mid = (l + r) >> 1;
	build(id * 2, l, mid);
	build(id * 2 + 1, mid + 1, r);
	pushup(id);
}

void update(int id, int l, int r, int x, int y, int v) {
	if(x <= l && r <= y) {
		sumv[id] = sumv[id] + v * (r - l + 1);
		lazy[id] = lazy[id] + v;
		return;
	}
	int mid = (l + r) >> 1;
	pushdown(id, l, r);
	if(x <= mid) update(id * 2, l, mid, x, y, v);
	if(y > mid) update(id * 2 + 1, mid + 1, r, x, y, v);
	pushup(id);
}

int query(int id, int l, int r, int x, int y) {
	if(x <= l && r <= y) return sumv[id];
	int mid = (l + r) >> 1, ans = 0;
	pushdown(id, l, r);
	if(x <= mid) ans += query(id * 2, l, mid, x, y);
	if(y > mid) ans += query(id * 2 + 1, mid + 1, r, x, y);
	return ans;
}

//以上都是线段树

void dfs1(int x, int father) {
	fa[x] = father, dep[x] = dep[father] + 1, sz[x] = 1;
	for (auto v:linker[x]) {
		if(v == father) continue; 
		dfs1(v, x);
		sz[x] += sz[v];
		if(sz[son[x]] < sz[v]) son[x] = v;
	}
}

void dfs2(int x, int t) { //t是指top 
	top[x] = t, id[x] = ++cnt, nw[cnt] = w[x];
	if(!son[x]) return;
	dfs2(son[x], t);
	for (auto v:linker[x]) {
		if(v == fa[x] || v == son[x]) continue;
		dfs2(v, v);
	}
}

int query_path(int x, int y) {
	int res = 0;
	while(top[x] != top[y]) {
		if(dep[top[x]] < dep[top[y]]) swap(x, y);
		res += query(1, 1, n, id[top[x]], id[x]);
		x = fa[top[x]];
	}
	if(dep[x] < dep[y]) swap(x, y);
	res += query(1, 1, n, id[y], id[x]);
	return res;
}

void update_path(int x, int y, int k) {
	while(top[x] != top[y]) {
		if(dep[top[x]] < dep[top[y]]) swap(x, y);
		update(1, 1, n, id[top[x]], id[x], k);
		x = fa[top[x]];
	}
	if(dep[x] < dep[y]) swap(x, y);
	update(1, 1, n, id[y], id[x], k);
}

int query_tree(int x) {
	return query(1, 1, n, id[x], id[x] + sz[x] - 1);
}

void update_tree(int x, int k) {
	update(1, 1, n, id[x], id[x] + sz[x] - 1, k);
}

//以上都是树链剖分

void add(int x, int y) {
	linker[x].push_back(y);
}

signed main() {
	cin >> n >> m >> root >> P;
	for (int i = 1; i <= n; i++) cin >> w[i]; 
	for (int i = 1, x, y; i < n; i++) {
		cin >> x >> y;
		add(x, y), add(y, x);
	}
	dfs1(root, 0), dfs2(root, root);
	build(1, 1, n);
	while(m--) {
		int opt, x, y, z;
		cin >> opt;
		if(opt == 1) {
			cin >> x >> y >> z;
			update_path(x, y, z);
		}
		if(opt == 2) {
			cin >> x >> y;
			cout << query_path(x, y) % P << endl;
		}
		if(opt == 3) {
			cin >> x >> z;
			update_tree(x, z);
		}
		if(opt == 4) {
			cin >> x;
			cout << query_tree(x) % P << endl;
		}
	}
	return 0;
