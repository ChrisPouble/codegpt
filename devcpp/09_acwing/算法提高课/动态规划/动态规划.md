# 数字三角形模型

## [摘花生](https://www.acwing.com/problem/content/1017/)

1. **状态：**$f_{i\ j}$ 表示走到 $(i,j)$ 时能获得到的最大价值。

2. **状态转移方程：**考虑走到 $(i,j)$ 的方式有两种，分别是从 $(i-1,j)$ 走过来与从 $(i,j-1)$ 走过来。因此，$f_{i\ j}=\max \{f_{i-1\ j},f_{i\ j-1} \}$

3. **初始化：**$f_{i\ 1} = a_{i\ 1},f_{1\ i}=a_{1\ i}$
4. **答案：**$f_{n\ m}$

```cpp
#include <bits/stdc++.h>
#define maxn 110
using namespace std;

int T, n, m, a[maxn][maxn], f[maxn][maxn];
void solve() {
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            scanf("%d", &a[i][j]);
    for (int i = 1; i <= n; i++) f[i][1] = f[i - 1][1] + a[i][1]; //初始化
    for (int i = 1; i <= m; i++) f[1][i] = f[1][i - 1] + a[1][i];
    for (int i = 2; i <= n; i++) //由于前面有了初始化，因此这里从2开始枚举
        for (int j = 2; j <= m; j++) //同上
            f[i][j] = a[i][j] + max(f[i - 1][j], f[i][j - 1]); //状态转移方程
    printf("%d\n", f[n][m]);
}

signed main() {
    scanf("%d", &T);
    while(T--) solve();
    return 0;
}
```

## [方格取数](https://www.acwing.com/problem/content/1029/)

一道有点难度的 `DP` 题......

**题目大意**

有一个 $ n \times n $ 的方格图，每个格子都有一个对应的权值 $ a_{i,j} $，你从 $(1, 1)$ 开始走，每次可以从 $(i, j)$ 走到 $(i + 1, j)$ 或 $(i, j + 1)$，最终走到 $(n, n)$，你需要这样走两次，求你所经过的格子上的权值之和的最大值，每个格子上的权值只计算一次。

**题目分析**

1. 状态：定义一个四维数组 $f$，$f_{i\ j\ k\ l}$ 表示第一次走到第 $i$ 行，第 $j$ 列，第二次到达第 $k$ 行，第 $l$ 列能获得的最大值。

2. 状态转移方程：我们要考虑以下四种情况：

- 第一次从左边过来，第二次从左边过来

- 第一次从左边过来，第二次从上边过来

- 第一次从上边过来，第二次从左边过来

- 第一次从上边过来，第二次从上边过来

那么我们就要取这四种情况的最大值了，即：$f_{i\ j\ k\ l} = \max\{f_{i-1\ j\ k-1\ l}, f_{i-1\ j\ k\ l-1}, f_{i\ j-1\ k-1\ l}, f_{i\ j-1\ k\ l-1}\}+a_{i\ j}+a_{k\ l}$。

但要注意的是，如果 $(i, j)=(k,l)$，那么就只能加其中一个（不能重复），所以此时要在原来的基础上减去一个 $a_{i\ j}$。

3. 初始化：刚开始也就是到点 $(1, 1)$ 能获得的最大值，即 $f_{1\ 1\ 1\ 1} = 0$。

4. 答案：由于我们要求第一次和第二次走到右下角的最大值，所以我们的答案为 $f_{n\ n\ n\ n}$。

上代码：

```cpp
#include <bits/stdc++.h> 
using namespace std;

int n, x, y, w, a[35][35], f[35][35][35][35]; //表示第一次走到第i行，第j列，第二次到达第k行，第l列能获得的最大值。
int main() {
	cin >> n;
	while(cin >> x >> y >> w) { //输入 
		if(x == 0 && y == 0 && w == 0) break; //退出 
		a[x][y] = w;
	}
	for (int i = 1; i <= n; i++)
		for (int j = 1; j <= n; j++)
			for (int k = 1; k <= n; k++)
				for (int l = 1; l <= n; l++) {
					int t1 = max(f[i - 1][j][k - 1][l], f[i - 1][j][k][l - 1]); //记录 
					int t2 = max(f[i][j - 1][k - 1][l], f[i][j - 1][k][l - 1]); //记录 
					f[i][j][k][l] = max(t1, t2) + a[i][j] + a[k][l]; //求最大值 
					if(i == k && j == l) f[i][j][k][l] -= a[i][j]; //如果位置相同，则减去其中一个 
				}
	cout << f[n][n][n][n] << endl; //答案 
	return 0;
}
```

## [传纸条](https://www.acwing.com/problem/content/277/)

这道题看似是先从 $(1,1)$ 到 $(n,m)$ 跑一遍，之后再反着跑一遍，实际它们是类似的。  
所以题目要求是从 $(1,1)$ 到 $(n,m)$ 跑两遍，不能采到重复的格子，要求最后的最大获益。

![](https://cdn.luogu.com.cn/upload/image_hosting/7b5eqgyt.png)

1. 状态：定义一个四维数组 $f$，$f_{i\ j\ k\ l}$ 表示蓝色路径走到第 $i$ 行，第 $j$ 列，红色路径到达第 $k$ 行，第 $l$ 列能获得的最大值。

2. 状态转移方程：我们要考虑以下四种情况：

- 第一次从左边过来，第二次从左边过来

- 第一次从左边过来，第二次从上边过来

- 第一次从上边过来，第二次从左边过来

- 第一次从上边过来，第二次从上边过来

那么我们就要取这四种情况的最大值了，即：$f_{i\ j\ k\ l} = \max\{f_{i-1\ j\ k-1\ l}, f_{i-1\ j\ k\ l-1}, f_{i\ j-1\ k-1\ l}, f_{i\ j-1\ k\ l-1}\}+a_{i\ j}+a_{k\ l}$。

但要注意的是，由于我们不能让两条路径重合。因此，这里的 $k=[i,n],l=[1,j]$，如果 $i=k,j=l$，那么直接跳过不考虑。

3. 初始化：刚开始也就是到点 $(1, 1)$ 能获得的最大值，即 $f_{1\ 1\ 1\ 1} = 0$。

4. 答案：由于我们要求蓝色路径和红色路径走到右下角的最大值，但是我们的状态转移方程中是不会考虑这一点的，因此考虑右下角上方的蓝色路径与右下角左边的红色路径的价值和，所以我们的答案为 $f_{n-1\ m\ n\ m-1}$。

```cpp
#include <bits/stdc++.h>
#define maxn 55
using namespace std;

int n, m, a[maxn][maxn], f[maxn][maxn][maxn][maxn];
int my_max(int x, int s, int w, int l) {
    return max(x, max(s, max(w, l)));
}

signed main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++)
		for (int j = 1; j <= m; j++)
			cin >> a[i][j]; 
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            for (int k = i; k <= n; k++)
                for (int o = 1; o <= j; o++) {
                    if(i == k && j == o) continue; //跳过
                    int walk = my_max(f[i - 1][j][k - 1][o], f[i - 1][j][k][o - 1], f[i][j - 1][k - 1][o], f[i][j - 1][k][o - 1]);
                    f[i][j][k][o] += a[i][j] + a[k][o] + walk;
//                    if (i == k && j == o) f[i][j][k][o] -= a[i][j];
                }
    cout << f[n - 1][m][n][m - 1] << endl; //这里是到不了终点的 
    return 0;
}
```

## [怪盗基德的滑翔翼](https://www.acwing.com/problem/content/1019/)

由于这道题中，怪盗基德可以向左右两个方向滑翔。因此，既要求一遍最长下降子序列，又得求一遍最长上升子序列。当然也可以从后往前再求一遍最长下降子序列。

```cpp
#include <bits/stdc++.h>
#define maxn 110
using namespace std;

int K, n, ans, h[maxn], f[maxn], g[maxn];
void solve() {
    cin >> n;
    ans = 0;
	for (int i = 1; i <= n; i++) cin >> h[i]; //输入每栋楼的高度
	for (int i = 1; i <= n; i++) {
		f[i] = 1;
		for (int j = 1; j < i; j++)
			if(h[i] > h[j]) f[i] = max(f[i], f[j] + 1); //最长上升子序列
	}
	for (int i = 1; i <= n; i++) {
		g[i] = 1;
		for (int j = 1; j < i; j++)
			if(h[i] < h[j]) g[i] = max(g[i], g[j] + 1); //最长下降子序列
	}
	for (int i = 1; i <= n; i++) ans = max(ans, f[i]), ans = max(ans, g[i]); //求出最多能滑翔的距离
	cout << ans << endl;
}

signed main() {
    cin >> K;
    while(K--) solve();
	return 0;
}
```

## [登山](https://www.acwing.com/problem/content/1016/)

1. **状态：**$f_{i\ j}$ 表示走到第 $i$ 个景点时，状态为 $j$ 时最多能看到的景点数量，其中 $j=0$ 表示上山状态，$j=1$ 表示下山状态。
2. **状态转移方程：**分两种情况讨论：
   - 如果现在是上山状态，那只能找到 $1\sim i-1$ 中正在上升的状态，即 $f_{i\ 0}=\max_{j=1}^{i-1}(h_j<h_i) \{f_{j\ 0}+1\}$
   - 如果现在是下山状态，那么可以找 $1\sim i-1$ 中正在上升与下降的状态，即 $f_{i\ 1}=\max_{j=1}^{i-1}(h_j>h_i)\{\max(f_{j\ 0},f_{j\ 1})+1\}$
3. **初始化：**$f_{i\ 0} = 1,f_{i\ 1}=1$
4. **答案：**$\max_{i=1}^n\{\max(f_{i\ 0},f_{i\ 1})\}$

```cpp
#include <bits/stdc++.h>
#define maxn 1010
using namespace std;

int n, ans, h[maxn], f[maxn][2]; //f[i][0]表示当前状态为上山状态，最多能看到的景点数量；f[i][1]表示当前状态为下山状态，最多能看到的景点数量；
signed main() {
    cin >> n;
	for (int i = 1; i <= n; i++) cin >> h[i];
	for (int i = 1; i <= n; i++) {
		f[i][0] = f[i][1] = 1; //初始化
		for (int j = 1; j < i; j++)
			if(h[i] > h[j]) f[i][0] = max(f[i][0], f[j][0] + 1); //上山状态
			else if(h[i] < h[j]) f[i][1] = max(f[i][1], max(f[j][0], f[j][1]) + 1); //下山状态
	}
	for (int i = 1; i <= n; i++) ans = max(ans, f[i][0]), ans = max(ans, f[i][1]); //求答案
	cout << ans << endl;
	return 0;
}
```

## [合唱队形](https://www.acwing.com/problem/content/484/)

看完题目，很容易想到使用最长上升子序列的做法来求，但是这似乎并不能用在这里。

这道题需要求两次 `DP`，第一次求 $1\sim i$ 中以 $h_i$ 结尾的最长上升子序列，第二次求 $i\sim n$ 中以 $h_i$ 开头的最长下降子序列，两边加起来再 $-1$ 就行了。

```cpp
#include <bits/stdc++.h>
#define maxn 110
using namespace std;

int n, ans, a[maxn], f[maxn], g[maxn];
void up() { //求最长上升子序列
	for (int i = 1; i <= n; i++) {
		f[i] = 1;
		for (int j = 1; j < i; j++)
			if(a[i] > a[j]) f[i] = max(f[i], f[j] + 1); 
	}
}

void down() { //求最长下降子序列（反着求）
	for (int i = n; i >= 1; i--) { //从后往前枚举
		g[i] = 1;
		for (int j = i + 1; j <= n; j++)
			if(a[i] > a[j]) g[i] = max(g[i], g[j] + 1); //这里要求i为开头
	}
}

int main() {
	cin >> n;
	for (int i = 1; i <= n; i++) cin >> a[i];
	up(), down();
	for (int i = 1; i <= n; i++) ans = max(ans, f[i] + g[i] - 1); //中间的i算了两次，要-1
	cout << n - ans << endl;
	return 0;
}
```

## [友好城市](https://www.acwing.com/problem/content/1014/)

此题难点并非在于 `DP`，而是在于如何转成 `DP`。

我们可以考虑一个枚举方案，按照上岸的坐标从小到大来枚举。

然后我们只需关心下岸的坐标之间有何关系即可。

此时，由于上岸的坐标已经排好序，只需要堆下岸的下标做最长上升子序列就行。

```cpp
#include <bits/stdc++.h>
#define maxn 5010
using namespace std;

int n, ans, k, f[maxn];
struct node {
    int x, y;
} a[maxn];

bool cmp(node x, node y) { return x.x < y.x; } //排序
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i].x >> a[i].y;
    sort(a + 1, a + n + 1, cmp);
    for (int i = 1; i <= n; i++) {
        f[i] = 1;
        for (int j = 1; j < i; j++)
            if (a[i].y > a[j].y && f[j] + 1 > f[i]) //更新
                f[i] = f[j] + 1;
    }
	for (int i = 1; i <= n; i++) ans = max(ans, f[i]);
	cout << ans << endl;
    return 0;
}
```

## [最大上升子序列和](https://www.acwing.com/problem/content/1018/)

1. **状态：**$f_i$ 表示以 $i$ 为结尾的最大上升子序列的和。

2. **状态转移方程：**$f_i=\max_{j=1}^{i-1}(a_i>a_j)\{f_j+a_j\}$

3. **初始化：**$f_i=a_i$
4. **答案：**$\max \{f_i\}$

```cpp
#include <bits/stdc++.h>
#define maxn 1010
#define int long long
using namespace std;

int n, ans, a[maxn], f[maxn];
signed main() {
	cin >> n;
	for (int i = 1; i <= n; i++) cin >> a[i];
	for (int i = 1; i <= n; i++) {
		f[i] = a[i];
		for (int j = 1; j < i; j++)
			if(a[i] > a[j]) f[i] = max(f[i], f[j] + a[i]);
	}
	for (int i = 1; i <= n; i++) ans = max(ans, f[i]);
	cout << ans << endl;
	return 0;
}
```

## [拦截导弹](https://www.acwing.com/problem/content/1012/)

第一问是求最长不升子序列的长度。使用 `DP` 解决即可。

第二问使用贪心，定义一个容器 `vec`，里面存放的是一些系统当前的高度，如果能被拦截，则更新系统高度，否则另开一个系统来拦截导弹。

```cpp
#include <bits/stdc++.h>
using namespace std;

int n = 1, ans, a[5010], f[5010];
vector<int> vec;
bool is(int x) {
    for (int j = 0; j < vec.size(); j++)
        if (vec[j] >= x) { //如 果能拦截下来
            vec[j] = x;
            return true;
        }
    return false;
}

int main() {
    while (cin >> a[n]) n++;
    n--;
    f[1] = 1;
    for (int i = 2; i <= n; i++) {
        f[i] = 1;
        for (int j = 1; j < i; j++)
            if (a[i] <= a[j]) //最长不升子序列
                f[i] = max(f[i], f[j] + 1);
    }
    for (int i = 1; i <= n; i++) ans = max(ans, f[i]);
    cout << ans << endl;
    vec.push_back(a[1]);
    for (int i = 2; i <= n; i++)
        if (!is(a[i]))
            vec.push_back(a[i]);
    cout << vec.size() << endl; //里面的系统数量
    return 0;
}
```

## [导弹防御系统](https://www.acwing.com/problem/content/189/)

此题虽然不是 `DP`，但是带有 `DP` 的思想。

定义 `dfs(int u, int d, int s)` 表示当前的上升系统的数量为 $u$，下降系统的数量为 $d$，当前枚举到的导弹为 $s$。

如果 $s=n+1$，说明已经枚举完了，则更新 $ans$ 的值。

紧接着，从 `up` 数组里找到第一个高度小于 $a_s$ 的系统，更新这个系统的高度，并继续枚举；如果没有，则创建一个新的系统。`down` 数组也类似。

需要注意的是，因为时间复杂度为 $O(n2^n)$，因此需要优化。再搜索的过程中，如果 $u+d\ge ans$，那么接着枚举就没有意义了，在此时，就直接返回，可以减少很多计算量。

```cpp
#include <bits/stdc++.h>
#define maxn 55
using namespace std;

int n, ans, up[maxn], down[maxn], a[maxn];
void dfs(int u, int d, int s) { //选择的上升子序列个数，下降子序列个数和当前的位置
    if (u + d >= ans) return; //剪枝
    if (s == n + 1) { //搜索完毕
        ans = min(ans, u + d);
        return;
    }
    int k = 1;
    while (k <= u && up[k] >= a[s]) k++; //由于up数组内一定是单调递减的，因此这样找是没有错的
    if (k <= u) { //能找到
        int t = up[k];
        up[k] = a[s]; //记录
        dfs(u, d, s + 1); //搜索
        up[k] = t; //回溯
    } else { //自己创建一个
        up[k] = a[s]; //记录
        dfs(u + 1, d, s + 1); //搜索
    }
    
    k = 1;
    while (k <= d && down[k] <= a[s]) k++;
    if (k <= d) {
        int t = down[k];
        down[k] = a[s];
        dfs(u, d, s + 1);
        down[k] = t;
    } else {
        down[k] = a[s];
        dfs(u, d + 1, s + 1);
    }
}

void solve() {
    memset(up, 0, sizeof(up)), memset(down, 0, sizeof(down)), memset(a, 0, sizeof(a));
    for (int i = 1; i <= n; i++) cin >> a[i];
    ans = n, dfs(0, 0, 1), cout << ans << endl; //搜索
}

signed main() {
    while(cin >> n && n) solve();
    return 0;
}
```

## [最长公共上升子序列](https://www.acwing.com/problem/content/description/274/)

1. **状态：**$f_{i\ j}$ 表示 $a$ 中前 $i$ 个元素，$b$ 中前 $j$ 个元素 ，且当前以 $b_j$ 结尾的子序列的方案。
2. **状态转移方程：**考虑两种情况：
   - 以 $a_i$ 结尾（$a_i=b_j$）：结合最长上升子序列的转移方程，则 $f_{i\ j}=\max\{\max_{k=1}^{j-1}(b_j>b_k)\{f_{i-1\ k}+1\},f_{i-1\ j}\}$。
   - 不以 $a_i$ 结尾：$f_{i\ j}=f_{i-1\ j}$。

3. **初始化：**$f_{i\ j}=0$

4. **答案：**$\max_{i=1}^n\{f_{n\ i}\}$

时间复杂度为 $Θ(n^3)$。

```cpp
#include <bits/stdc++.h>
#define maxn 3010
using namespace std;

int n, ans = -INT_MAX, a[maxn], b[maxn], f[maxn][maxn];
signed main() {
    for (int i = 1; i < maxn; i++)
        for (int j = 1; j < maxn; j++)
            f[i][j] = 0;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int j = 1; j <= n; j++) cin >> b[j];
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            f[i][j] = f[i - 1][j]; //先考虑不包含a[i]
            if(a[i] == b[j]) { //再考虑包含a[i]
                for (int k = 0; k < j; k++)
                    if(b[j] > b[k]) f[i][j] = max(f[i][j], f[i - 1][k] + 1); //类似最长上升子序列的状态转移方程
            }
        }
    }
    for (int i = 1; i <= n; i++) ans = max(ans, f[n][i]); //答案（还是仿照最长上升子序列）
    cout << ans << endl;
    return 0;
}
```

在转移方程的第一种情况中，我们要求的是 $(a_i>b_k)\{f_{i-1\ k}+1\}$ 的最大值。其实我们可以在每次循环的过程中用一个变量 $maxv$ 来记录这个最大值。之后再更新这个最大值就可以了。

```cpp
#include <bits/stdc++.h>
// #define int int
#define maxn 3010
using namespace std;

int n, ans = -INT_MAX, a[maxn], b[maxn], f[maxn][maxn];
signed main() {
    for (int i = 1; i < maxn; i++)
        for (int j = 1; j < maxn; j++)
            f[i][j] = 0;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int j = 1; j <= n; j++) cin >> b[j];
    for (int i = 1; i <= n; i++) {
        int maxv = 1;
        for (int j = 1; j <= n; j++) {
            f[i][j] = f[i - 1][j]; //第二种情况
            if(a[i] == b[j]) f[i][j] = max(f[i][j], maxv); //在前面的循环中，已经使用maxv算出了f[i-1][k]+1的最大值 
            if(a[i] > b[j]) maxv = max(maxv, f[i - 1][j] + 1); //在这里，a[i] > b[j]，表示着可以用来更新后面的f值 
        }
    }
    for (int i = 1; i <= n; i++) ans = max(ans, f[n][i]);
    cout << ans << endl;
    return 0;
}
```

## [采药](https://www.acwing.com/problem/content/description/425/)

1. **状态：**$f_{i\ j}$ 表示前 $i$ 种草药时间为 $j$ 时能采摘到达最大价值

2. **状态转移方程：**

```cpp
if(j >= a[i].t) f[i][j] = max(f[i - 1][j], f[i - 1][j - a[i].t] + a[i].w); //找到最大值 
else f[i][j] = f[i - 1][j];
```

3. **初始化：**$f_{0\ 0} = 0$
4. **答案：**$\max\{f_{i\ T}\}$

上代码（二维数组解法）：

```cpp
#include <bits/stdc++.h>
using namespace std;

int T, m, ans, f[110][1010]; //f[i][j]表示前i种草药时间为j时能采摘到达最大价值 
struct node {
	int t, w; //一个草药需要花费时间和得到价值 
} a[110];

int main() {
	cin >> T >> m;
	for (int i = 1; i <= m; i++)
		cin >> a[i].t >> a[i].w;
	for (int i = 1; i <= m; i++)
		f[i][0] = 0; //初始化 
	for (int i = 1; i <= m; i++)
		for (int j = 1; j <= T; j++)
			if(j >= a[i].t) f[i][j] = max(f[i - 1][j], f[i - 1][j - a[i].t] + a[i].w); //找到最大值 
			else f[i][j] = f[i - 1][j];
	for (int i = 1; i <= m; i++)
		ans = max(ans, f[i][T]); //找答案 
	cout << ans << endl;
    return 0;
}
```

很明显，空间占用的太多了，那怎么办呢？

我们发现其实我们在更新 $f_{i\ j}$ 的时候只会用到两个值：$i和i-1$，所以我们的第一维只需要两个维度就可以了。我们称这种思路为滚动数组。

```cpp
#include <bits/stdc++.h>
using namespace std;

int T, m, ans, f[2][1010]; //f[i % 2][j]表示前i种草药时间为j时能采摘到达最大价值 
struct node {
	int t, w; //一个草药需要花费时间和得到价值 
}a[110];

int main() {
	cin >> T >> m;
	for (int i = 1; i <= m; i++)
		cin >> a[i].t >> a[i].w;
	for (int i = 1; i <= m; i++)
		for (int j = 1; j <= T; j++)
			if(j >= a[i].t) f[i % 2][j] = max(f[(i - 1) % 2][j], f[(i - 1) % 2][j - a[i].t] + a[i].w);
			else f[i % 2][j] = f[(i - 1) % 2][j];
	cout << f[m % 2][T] << endl;
    return 0;
}
```

那么我们能不能再进一步优化空间呢？答案肯定是可以的

上面讲解滚动数组的时候我们已经知道每次只会用到两个值了，所以我们可以直接把第一维省去，直接由 $i-1$ 复制到 $i$，这样就可以了

不过需要注意的一点：如果我们枚举 $j$ 的时候从前往后遍历，是会错误的，我们可以尝试一下：

## [装箱问题](https://www.acwing.com/problem/content/1026/)

看似不能使用 01 背包来做，但我们可以把物品的体积当成重量和价值。这样就转化成了 01 背包问题啦！

- **状态**：定义 $f_{i\ j}$ 为考虑前 $i$ 个物品，容量为 $j$，能装下的最多体积。
- **状态转移方程**：仿照 01 背包的转移方程，在这里我们只需要把体积当作价值就可以了。
```cpp
if(j >= a[i]) f[i][j] = max(f[i - 1][j], f[i - 1][j - a[i]] + a[i]); //找到最大体积
else f[i][j] = f[i - 1][j]; //如果装不下就复制前面的
```
- **初始化**：背包容量为 $0$ 时，什么物品都放不下。因此，初始化为 $f_{i\ 0}=0$。
- **答案**：仿照 01 背包，最终的答案为 $V-f_{n\ V}$。
```cpp
#include <bits/stdc++.h>
#define V 20010
#define maxn 35
using namespace std;

int n, m;
int f[maxn][V], a[maxn];
int main() {
	cin >> m >> n;
	for (int i = 1; i <= n; i++)
		cin >> a[i];
	for (int i = 1; i <= n; i++)
		f[i][0] = 0; //初始化 
	for (int i = 1; i <= n; i++)
		for (int j = 1; j <= m; j++)
			if(j >= a[i]) f[i][j] = max(f[i - 1][j], f[i - 1][j - a[i]] + a[i]); //找到最大体积
			else f[i][j] = f[i - 1][j]; //如果背包容量转不下就复制前面的
	cout << m - f[n][m] << endl; //最后剩余的空间：原有的背包熔炼-最终可以装下的最多体积
    return 0;
}
```
## [宠物小精灵之收服](https://www.acwing.com/problem/content/description/1024/)

与一维 01 背包类似，只不过多了一维罢了。

注意不能把背包填满，因此在开始时要让背包容量 $-1$。在最后算剩余体积的时候又要 $+1$。

```cpp
#include <bits/stdc++.h>
#define maxn 1010
#define maxm 510
#define maxk 110
using namespace std;

int n, A, B, see[maxn], smile[maxn], dp[maxn][maxm]; //这里在迭代的过程中把第一维给省了（空间会炸）
int main() {
	cin >> A >> B >> n;
	B--; //开局-1，控制背包容量
	for (int i = 1; i <= n; i++) cin >> see[i] >> smile[i];
	for (int i = 1; i <= n; i++)
		for (int j = A; j >= see[i]; j--)
			for (int k = B; k >= smile[i]; k--)
    			dp[j][k] = max(dp[j][k], dp[j - see[i]][k - smile[i]] + 1); //接下来就是普通的二维背包了
	int k = B;
	while(k > 0 && dp[A][k - 1] == dp[A][B]) k--; //找到最优解
	cout << dp[A][B] << " " << B + 1 - k << endl; //最后的背包容量+1
	return 0;
}
```

## [数字组合](https://www.acwing.com/problem/content/280/)

这题属于是 01 背包求方案数，但只要把状态把握好就迎刃而解了。

- **状态**：定义 $f_{i\ j}$ 为考虑前 $i$ 个数字，得到的总和**恰好**为 $j$ 的方案数。
- **状态转移方程**：仿照 01 背包的转移方程，但是这里求的是 $\sum$，不是 $\max$。

```cpp
f[i][j] = f[i - 1][j] + f[i - 1][j - a[i]]; //复制前面的+自己制造的
```

- **初始化**：总和体积为 $0$ 时，只有什么都不选这一种方案。因此，初始化为 $f_{i\ 0}=1(0\le i\le n)$。
- **答案**：仿照 01 背包，最终的答案为 $f_{n\ m}$。

```cpp
#include <bits/stdc++.h>
#define maxn 10010
#define N 110
using namespace std;

int n, m, a[maxn], f[N][maxn];
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 0; i <= n; i++) f[i][0] = 1; //初始化
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++) {
            f[i][j] = f[i - 1][j];
            if(j >= a[i]) f[i][j] = f[i - 1][j] + f[i - 1][j - a[i]]; //只有能装得下才可以加上f[i - 1][j - a[i]]
        }
    cout << f[n][m] << endl;
	return 0;
}
```

## [货币系统](https://www.acwing.com/problem/content/534/)

从这道题里面我们可以得到一个性质：

> $(m,b)$ 一定是 $(n,a)$ 的一个子集。
>
> 证明：
>
> 若存在 $b_i\ne (n,a)\text{ 中的任何数}$，则分两种情况：
>
> - 如果 $b_i$ 不能被 $(n,a)\text{ 中的任何数}$ 表示，则 $(m,b)$ 可以表示 $b_i$ 这个面额，而 $(n,a)$ 不可以。不符合题意。
> - 如果 $b_i$ 能被 $a_j$ 表示，且 $b_i\ne a_j$，那 $b_i$ 就是个废物，还不如换成 $a_j$。
>
> 综上所述，$(m,b)$ 一定是 $(n,a)$ 的一个子集。

那么，我们只要把 $(n,a)$ 中那些**废物**给筛掉就行了。

由于 $a_i$ 只会被比 $a_i$ 小的数，所以先给 $(n,a)$ 排序。

之后就是我们的 `DP`，而且这一段很像我们的埃式筛。

- **状态**：定义 $f_i$ 为 $i$ 这个数是否能被别的 $a_j$ 表示出来。
- **状态转移方程**：由于货币可以无限取，因此仿照完全背包的转移方程，但是这里求的是 $\operatorname{bool}$，不是 $\max$。

```cpp
if(f[j - a[i]]) f[j] = 1; //如果 f[j - a[i]] 可以被表示，那么 f[j] 就可以被表示
```

- **初始化**：一个数为 $0$ 时，只要什么都不选，就可以表示这个数。因此，初始化为 $f_{0}=1$。
- **答案**：仿照完全背包，最终的答案为 $\sum_{i=1}^n (f_i=0)$。

```cpp
#include <bits/stdc++.h>
#define maxn 110
#define inf 25010
#define endl "\n"
using namespace std;

int T, n, a[maxn], f[inf];
void solve() {
    cin >> n;
    memset(f, 0, sizeof(f));
    for (int i = 1; i <= n; i++) cin >> a[i];
    sort(a + 1, a + n + 1); //排序
    f[0] = 1; //初始化
    int maxm = a[n], ans = 0; //maxm是最大的数
    for (int i = 1; i <= n; i++) {
        if(f[a[i]]) continue; //由于已经排好序了，就不用担心后面的数更新自己了。一个数被标记成1了，表示别的数就没有意义了。
        ans++; //边算DP，边记答案
        for (int j = a[i]; j <= maxm; j++)
            if(f[j - a[i]]) f[j] = 1; //f[j - a[i]] 能被表示，f[j] 就能被表示
    }
    cout << ans << endl;
}

int main() {
    cin >> T;
    while(T--) solve();
    return 0;
}
```

## [多重背包问题 III](https://www.acwing.com/problem/content/6/)

多重背包的原始状态转移方程：

$ f(i, j) = \max(f(i-1, j), f(i-1, j-v) + w, \cdots, f(i-1, j-sv) + sw) $。

考虑用完全背包的优化方式来优化这个方程：

$ f(i, j-v) = \max(f(i-1, j-v), f(i-1, j-2v) + w, \cdots, f(i-1, j-(s+1)v) + (s)w) $。

写出这个公式好像并不是那么管用。

因为完全背包是一口气把所有体积全部用掉，即：

$ \max(a, b, c, d) = \max(a, \max(b, c, d)) $。

然而多重背包对于每个物品的个数是有限制的，导致我们最终的等式是如下样子：

$ \max(a, b, c, d) \neq \max(a, \max(b, c, d, e)) $。

但是，我们可以把这个式子继续推导下去，直到背包体积被用到不能再用为止：

$$
\left\{
\begin{aligned}
f(i, j) &= \max(f(i-1, j), f(i-1, j-v) + w, \cdots, f(i-1, j-sv) + sw) \\
f(i, j-v) &= \max(f(i-1, j-v), f(i-1, j-2v) + w, \cdots, f(i-1, j-(s+1)v) + sw) \\
f(i, j-2v) &= \max(f(i-1, j-2v), f(i-1, j-3v) + w, \cdots, f(i-1, j-(s+2)v) + sw) \\
&\cdots \\
f(i, r+sv) &= \max(f(i-1, r+sv), f(i-1, r+(s-1)v) + w, \cdots, f(i-1, r) + sw) \\
f(i, r+(s-1)v) &= \max(f(i-1, r+(s-1)v), \cdots, f(i-1, r) + (s-1)w) \\
&\cdots \\
f(i, r+2v) &= \max(f(i-1, r+2v), f(i-1, r+v) + w, f(i-1, r) + 2w) \\
f(i, r+v) &= \max(f(i-1, r+v), f(i-1, r) + w) \\
f(i, r) &= f(i-1, r)
\end{aligned}
\right.
$$
可能看上去还是有点复杂，为了再方便大家观察，我们去掉 $w$，然后把数组展开成一条链

具体如下图：

![](https://cdn.acwing.com/media/article/image/2021/06/15/55909_fa6e36f6cd-%E5%A4%87%E6%B3%A8-2020%E5%B9%B47%E6%9C%8825%E6%97%A5.png)

于是通过该滑动窗口，我们就能在线性的时间里求出 $i$ 阶段里，所有满足 $j \equiv r \mod (v)$ 的 $f(i, j)$。

滑动窗口求最大值的实现，只需利用队列在队头维护一个最大值的单调递减的单调队列即可。

为了更新所有 $i$ 阶段里的状态 $f(i, j)$，我们只需再额外枚举所有的余数 $r$ 即可。

不要忘记，滑动窗口内部比较最大值的时候，有一个在之前为了方便观察，被删掉的偏移量 $w$ 要记得加上再比较。

具体就是当前下标和该最大值的下标之间差了 $x$ 个 $v$，那么就要加上 $x$ 个 $w$。

```cpp
#include <bits/stdc++.h>
#define maxn 1010
#define maxm 20010
using namespace std;

int n, V, v[maxm], w[maxm], s[maxm], f[maxn][maxm];
int q[maxm], hh, tt;
signed main() {
	cin >> n >> V;
	for (int i = 1; i <= n; i++) cin >> v[i] >> w[i] >> s[i];
	for (int i = 1; i <= n; i++)
    	for (int r = 0; r < v[i]; r++) { //枚举%v[i]的余数
    	    hh = 1, tt = 0; //单调队列的队头与队尾
    	    for (int j = r; j <= V; j += v[i]) {
	    	    while(hh <= tt && j - q[hh] > s[i] * v[i]) hh++; //队头已经不在窗口内了
	    	    while(hh <= tt && f[i - 1][j] >= f[i - 1][q[tt]] + (j - q[tt]) / v[i] * w[i]) tt--; //队尾太小了
	    	    q[++tt] = j; //把当前体积放入队尾
	    	    f[i][j] = f[i - 1][q[hh]] + (j - q[hh]) / v[i] * w[i]; //状态转移
		    }
    	}
	cout << f[n][V] << endl; //答案
	return 0;
}
```

- **状态转移方程**：仿照二维 01 背包的状态转移方程。

```cpp
f[i][j][k] = min(f[i - 1][j][k], f[i - 1][max(j - a[i], 0)][max(k - b[i], 0)] + c[i]); //要不然
```

- **初始化**：如果要求氧气和氮气不超过 $0$ 时，那么最小重量为 $0$，而其他的赋值为 $\inf$。因此，初始化为 $f_{i\ j\ k}=\inf,f_{0\ 0\ 0}=0$。

```cpp
#include <bits/stdc++.h>

#define maxm 110
#define inf 0x3f3f3f3f

	memset(dp, 0x3f, sizeof(dp));
	cin >> A >> B >> n;
	dp[0][0][0] = 0;

	for (int i = 1; i <= n; i++)
	    for (int j = 0; j <= A; j++)
            /*

            */
	        for (int k = 0; k <= B; k++) {
	            dp[i][j][k] = min(dp[i - 1][j][k], dp[i - 1][max(j - a[i], 0)][max(k - b[i], 0)] + c[i]); //负数情况看成0
```

## [机器分配](https://www.acwing.com/problem/content/description/1015/)

考虑分组背包。

- **状态**：定义 $f_{i\ j}$ 为考虑前 $i$ 个公司，且花费总时间不超过 $j$ 的集合下能获得的最大价值。

- **状态转移方程**：分为两种情况：
  - 不选：`f[i][j] = f[i - 1][j]`

  - 选：`f[i][j] = max(f[i][j], f[i - 1][j - k] + w[i][k]) //0<=k<=j`

- **初始化**：$f_{0\ 0}=0$。
- **答案**：$f_{n\ m}$。

但如何求具体方案呢？

只要顺着状态转移方程往前推就行了。

```cpp
#include <bits/stdc++.h>
#pragma GCC optimize(2)
#pragma GCC optimize(3)
#define maxn 20
using namespace std;

int n, m, way[maxn], w[maxn][maxn], f[maxn][maxn];
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            cin >> w[i][j];
    for (int i = 1; i <= n; i++) //DP过程
        for (int j = 0; j <= m; j++) {
            f[i][j] = f[i - 1][j];
            for (int k = 0; k <= j; k++)
                f[i][j] = max(f[i][j], f[i - 1][j - k] + w[i][k]);
        }
    cout << f[n][m] << endl;
    int res = m; //当前拥有的量
    for (int i = n; i >= 1; i--) { //从后往前推
        for (int k = 0; k <= res; k++)
            if(f[i][res] == f[i - 1][res - k] + w[i][k]) { //正中状态转移方程
                way[i] = k; //记录答案
                res -= k; //减去这个公司花费的量
                break;
            }
    }
    for (int i = 1; i <= n; i++) cout << i << " " << way[i] << endl;
    return 0;
}
```

## [有依赖的背包问题](https://www.acwing.com/problem/content/description/10/)

考虑使用树形 `DP`。

- **状态**：定义 $f_{x\ j}$ 为以 $x$ 为根的子树，且限重为 $j$ 的最大权值（选 $i$）。

- **状态转移方程**：

  `f[x][j] = max(f[x][j], f[x][j - k] + f[son][k]);`

- **初始化**：$f_{x\ j}=w_x$。
- **答案**：$f_{root\ m}$。

```cpp
#include <bits/stdc++.h>
#define maxn 110
using namespace std;

int n, m, root, ans, v[maxn], w[maxn], f[maxn][maxn]; //f[i][j]表示以i为根的子树限重为j的最大权值（选i）
vector<int> linker[maxn];
void dfs(int x) {
    for (int j = m; j >= v[x]; j--) f[x][j] = w[x];
    for (auto son:linker[x]) {
        dfs(son); //先递归儿子
        for (int j = m; j >= v[x]; j--) //倒着枚举的原因：因为降维了，因此要倒着枚举。不降维的是f[x][j][k]
            for (int k = 0; k <= j - v[x]; k++) //要给v[x]留个位置
                f[x][j] = max(f[x][j], f[x][j - k] + f[son][k]); //状态转移方程
    }
}

int main() {
    cin >> n >> m;
    for (int i = 1, p; i <= n; i++) {
        cin >> v[i] >> w[i] >> p;
        if(p != -1) linker[p].push_back(i);
        else root = i; //找到根节点
    }
    dfs(root);
    cout << f[root][m] << endl;
	return 0;
}
```

## [背包问题求方案数](https://www.acwing.com/problem/content/11/)

首先，先跑正常的 01 背包。

其次，开始我们的路径跟踪 `DP`。

- **状态**：定义 $g_{i\ j}$ 考虑前 $i$ 个物品，当前已使用体积**恰好**是 $j$ 的，且价值为最大的方案数。
- **状态转移方程**：

  - 如果 $ f_{i,j} = f_{i-1,j} $ 且 $ f_{i,j} = f_{i-1,j-v} + w $ 则 $ g_{i,j} = g_{i-1,j} + g_{i-1,j-v} $

  - 如果 $ f_{i,j} = f_{i-1,j} $ 且 $ f_{i,j} \neq f_{i-1,j-v} + w $ 则 $ g_{i,j} = g_{i-1,j} $

  - 如果 $ f_{i,j} \neq f_{i-1,j} $ 且 $ f_{i,j} = f_{i-1,j-v} + w $ 则 $ g_{i,j} = g_{i-1,j-v} $

- **初始化**：$g_{0\ 0}=1$。
- **答案**：$\sum_{i=0}^m(f_{n\ i}=f_{n\ m})g_{n\ i}$。

```cpp
#include <bits/stdc++.h>
#define maxn 1010
#define mod 1000000007
using namespace std;

int n, m, ans, v[maxn], w[maxn], f[maxn][maxn], g[maxn][maxn];
signed main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> v[i] >> w[i];
    for (int i = 1; i <= n; i++)
        for (int j = 0; j <= m; j++) {
            f[i][j] = f[i - 1][j];
            if(j >= v[i]) f[i][j] = max(f[i][j], f[i - 1][j - v[i]] + w[i]);
        }
    g[0][0] = 1;
    for (int i = 1; i <= n; i++)
        for (int j = 0; j <= m; j++) {
            if(f[i][j] == f[i - 1][j]) g[i][j] = (g[i][j] + g[i - 1][j]) % mod;
            if(j >= v[i] && f[i][j] == f[i - 1][j - v[i]] + w[i]) g[i][j] = (g[i][j] + g[i - 1][j - v[i]]) % mod;
        }
    for (int i = 0; i <= m; i++)
        if (f[n][i] == f[n][m]) ans = (ans + g[n][i]) % mod;
    cout << ans << endl;
    return 0;
}
```

## [背包问题求具体方案](https://www.acwing.com/problem/content/12/)

## [能量石](https://www.acwing.com/problem/content/736/)

发现有可能存在最优解的某些宝石的贡献为 $0$，我们剔除了这些宝石。

假设最优解的能量石排列长度为 $ k $ ($ 1 \leq k \leq n $) 因为去掉了那些没有贡献的宝石，位置为：$ a_1, a_2, a_3, \ldots, a_k $。

那么对于任意两个位置 $ i = a_l, j = a_{l+1} $ ($ 1 \leq l < k $)

交换后两个宝石的贡献总和不会变得更大，即（假设之前的总时间为 $ t $ ）：
$ E_i - t * L_i + E_j - (t + S_i) * L_j \geq E_j - t * L_j + E_i - (t + S_j) * L_i $

整理后：
$ S_i * L_j \leq S_j * L_i $

我们可以把跟 $ i $ 有关的放到一边，调整一下：
$ \frac{S_i}{L_i} \leq \frac{S_j}{L_j} $

这样，我们只要以如上条件作为宝石间排序的条件，进行一次 sort。

因为对于其他形式的放置规律，必然可以通过交换满足 $\frac{S_i}{L_i} > \frac{S_j}{L_j}$ 的相邻的两项来得到更小值。

那么最优解的坐标（新的坐标）一定满足：
$ a_1 < a_2 < a_3 \ldots < a_k $

---

那么，我们只要搞个 01 背包，$ S_i $ 作为费用，$ \max(0, E_i - (t - S_i) * L_i) $ 作为价值 ($ t $ 为当前花费时长)。

$ f_t$ 表示当前正好花 $ t $ 时间得到的最大能量。

状态转移方程：
$ f_t = \max(f_t, f[t - S_i] + \max(0, E_i - (t - S_i) * L_i)) $

由于我们背包放物品（宝石）的顺序是坐标从 $1$ 到 $ n $ 的，所以一定能枚举到最优解。

初始状态：$ f_0 = 0 $，其余为负无穷。

答案：$ \max(f_i) (1 \leq i \leq \sum_{i=1}^{n} s_i) $。

```cpp
#include <bits/stdc++.h>
#define int long long
#define N 100010
using namespace std;

int T, n, f[N];
struct node {
    int s, e, l;
} a[N];

bool cmp(node x, node y) {
    return x.s * y.l <= x.l * y.s; //排序
}

void solve(int x) {
    cin >> n;
    memset(f, 0, sizeof(f));
    int t = 0, ans = 0;
    for (int i = 1; i <= n; i++) {
        cin >> a[i].s >> a[i].e >> a[i].l;
        t += a[i].s; //最大时间
    }
    sort(a + 1, a + n + 1, cmp);
    for (int i = 1; i <= n; i++)
        for (int j = t; j >= 0; j--)
            if(j >= a[i].s) f[j] = max(f[j], f[j - a[i].s] + a[i].e - (j - a[i].s) * a[i].l); //状态转移方程
    for (int i = 0; i <= t; i++) ans = max(ans, f[i]); //最后的答案
    cout << "Case #" << x << ": " << ans << endl;
}

signed main() {
    cin >> T;
    for (int i = 1; i <= T; i++) solve(i);
    return 0;
}
```

## [金明的预算方案](https://www.acwing.com/problem/content/489/)

这道题其实是分组背包，每一个主件与它的附件的组合就是一个分组。最后跑一遍分组背包就行了。

```cpp
#include <bits/stdc++.h>
#define maxn 70
#define maxm 32010
using namespace std;

int n, m, f[maxm];
struct node {
    int v, w;
} master[maxn];
vector<node>linker[maxn];
int main() {
    cin >> m >> n;
    for (int i = 1; i <= n; i++) {
        int v, p, q;
        cin >> v >> p >> q;
        p *= v; //真·价值
        if(!q) master[i] = {v, p}; //表示一个物品的贡献 
        else linker[q].push_back({v, p}); //表示一个物品的附件的贡献
    }
    for (int i = 1; i <= n; i++) {
        int siz = linker[i].size();
        for (int u = m; u >= 0; u--)
            for (int j = 0; j < (1 << siz); j++) { //最多四种状态：选自己、选自己和a儿子、选自己和B儿子、选自己和两个儿子
                int v = master[i].v, w = master[i].w; //加上自己
                for (int k = 0; k < siz; k++)
                    if(j >> k & 1) { //判断这个儿子是否被枚举到了
                        v += linker[i][k].v;
                        w += linker[i][k].w;
                    }
                if(u >= v) f[u] = max(f[u], f[u - v] + w); //分组背包状态转移方程
            }
    }
    cout << f[m] << endl;
```

# 状态机模型

## [大盗阿福](https://www.acwing.com/problem/content/1051/)

因为每家店铺在第 $i$ 次被偷和在第 $j$ 次被偷，获得的价值是一样的。因此考虑我们能否按照从 $1$ 到 $n$ 的线性顺序找出**最优解**。

- **状态**：定义 $f_i$ 为偷前 $i$ 个店铺能偷得的最大价值。

- **状态转移方程**：考虑第 $i$ 个店铺偷或不偷的两种情况：

  - 如果偷第 $i$ 个店铺，则 $f_i=f_{i-2}+w_i$。
  - 如果不偷第 $i$ 个店铺，则 $f_i=f_{i-1}$。

- **初始化**：由于直接套状态转移方程会在第 $1$ 个店铺和第 $2$ 店铺访问负数下标，因此在 $i=1\& 2$ 的时候初始化就行了：

  $f_1=w_i,f_2=\max\{w_i,f_1\}$。

- **答案**：$f_n$。

```cpp
#include <bits/stdc++.h>
#define maxn 100010
using namespace std;

int T, n, a[maxn], f[maxn];
void solve() {
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) scanf("%d", &a[i]);
    f[1] = a[1], f[2] = max(a[1], a[2]); //初始化
    for (int i = 3; i <= n; i++) f[i] = max(f[i - 2] + a[i], f[i - 1]); //状态转移方程
    printf("%d\n", f[n]);
}

signed main() {
    cin >> T;
    while(T--) solve();
    return 0;
}
```

## [股票买卖 IV](https://www.acwing.com/problem/content/1059/)

对于第 $i$ 天购入的股票，当且仅当第 $j (i \leq j)$ 天才能卖掉，获得的利润为 $w_j - w_i$。

由此可知求解股票交易问题是一个线性的过程，于是我们思考一下能否用线性DP进行状态记录和转移。

线性 DP 如何记录当前的状态？

我们以当前递推到的天数，作为线性 DP 的阶段。

对于递推到的第 $i$ 天来说，我们需要记录的信息有：
1. 在完成第 $i$ 天的决策后，状态是持仓（$k=1$）还是空仓（$k=0$）。
2. 在第 $i$ 天交易完成时，总共完成的完整的交易数 $j$ （题目规定一次买入一次卖出是一笔完整的交易）。

那么如何利用该状态表示出题设中的状态转移行为呢？
1. 买入行为：$k=0 \rightarrow k=1$。
2. 卖出行为：$k=1 \rightarrow k=0$。
3. 持仓行为：$k=1 \rightarrow k=1$。
4. 空仓行为：$k=0 \rightarrow k=0$。

当然，因为第 $i$ 天的 $k$ 可以表示在一起，因此我们以 $j\bmod 2=1$ 表示第 $\frac{j}{2}$ 次买入，$j\bmod 2=0$ 表示第 $\frac{j}{2}$ 次卖出。

于是，就有了我们的**自制 DP 分析法**。 

- **状态**：定义 $f(i,j)$ 表示前 $i$ 天，第 $i$ 天状态为 $j$ 时的最大利润。其中 $j\bmod 2=1$ 表示第 $\frac{j+1}{2}$ 次买入，$j\bmod 2=0$ 表示第 $\frac{j+1}{2}$ 次卖出。
- **状态转移方程**：按 $j$ 的奇偶性分两种情况讨论：
  - 若 $j\bmod 2=1$，则 $f(i,j)=\max\{f(i-1,j),f(i-1,j-1)-a_i\}$。
  - 若 $j\bmod 2=0$，则 $f(i,j)=\max\{f(i-1,j),f(i-1,j-1)+a_i\}$。

- **初始化**：$f(i,j)=-\inf,f(i,0)=0$。
- **答案**：要求最大利润，那就必须得是卖出了，因此答案为 $\max\{f(n,2i)\}$。

```cpp
#include <bits/stdc++.h>
#define maxn 100010
#define maxk 110
using namespace std;

int n, k, ans, a[maxn], f[maxn][2 * maxk];
signed main() {
    cin >> n >> k;
    memset(f, -0x3f, sizeof(f)); //全部赋值最小值
    f[0][0] = 0; //初始化
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) {
        f[i][0] = f[i - 1][0]; //初始化
        for (int j = 1; j <= 2 * k; j++) { //枚举状态
            if(j & 1) f[i][j] = max(f[i - 1][j], f[i - 1][j - 1] - a[i]); //状态转移方程
            else f[i][j] = max(f[i - 1][j], f[i - 1][j - 1] + a[i]);
        }
    }
    for (int i = 1; i <= k; i++) ans = max(ans, f[n][2 * i]); //答案
    cout << ans << endl;
    return 0;
}
```

实在看不懂可以仿照[这一篇题解](https://www.acwing.com/solution/content/5049/)，通俗易懂。

## [股票买卖 V](https://www.acwing.com/problem/content/1060/)

由于题目中给出了三种状态：空仓、持仓、冷冻期，因此考虑用线性 DP 对应到这三种状态上。

- **状态**：定义 $f(i,j)$ 为考虑前 $i$ 天，第 $i$ 天状态为 $j$ 时的最大利润。$j$ 为 $0/1/2$，分别表示空仓、持仓、冷冻期。这里的状态是指第 $i$ 天刚开始什么都没干的状态。
- **状态转移方程**：按三种状态分三种情况考虑：
  - $j=0$，此时第 $i$ 天是空仓，那么第 $i-1$ 天可能是空仓或冷冻期，因此 $f(i,0)=\max\{f(i-1,0),f(i-1,2)\}$。
  - $j=1$，此时第 $i$ 天是持仓，那么第 $i-1$ 天既可以是持仓不动，也可以是空仓时买入股票，因此 $f(i,1)=\max\{f(i-1,1),f(i-1,0)-a_i\}$。
  - $j=2$，此时第 $i$ 天是冷冻期，那么第 $i-1$ 天只能是持仓后卖出股票，因此 $f(i,2)=f(i,1)+a_i$。
- **初始化**：刚开始的时候，第 $0$ 天肯定是空仓没有利润，因此 $f(0,0)=0$。
- **答案**：考虑到最后一天如果还拿着股票肯定是亏的，因此答案为 $\max\{f(n,0),f(n,2)\}$。

```cpp
#include <bits/stdc++.h>
#define maxn 100010
using namespace std;

int n, a[maxn], f[maxn][3];
signed main() {
    cin >> n;
    memset(f, -0x3f, sizeof(f));
    for (int i = 1; i <= n; i++) cin >> a[i];
    f[0][0] = 0;
    for (int i = 1; i <= n; i++) {
        f[i][0] = max(f[i - 1][0], f[i - 1][2]);
        f[i][1] = max(f[i - 1][1], f[i - 1][0] - a[i]);
        f[i][2] = f[i - 1][1] + a[i];
    }
    cout << max(f[n][0], f[n][2]) << endl;
    return 0;
}
```

## [设计密码](https://www.acwing.com/problem/content/1054/)

首先，使用 KMP 预处理出 $T$ 的 $ne$ 数组（别问为什么，待会就知道了）。

- **状态**：
  - 定义 $f(i,j)$ 表示长度为 $ i $ 的密码，且与 $ T $ 的前 $ j $ 个字符匹配的方案数。
  - 这里 $ j $ 表示当前密码的后缀与 $ T $ 的前缀匹配的长度。


- **状态转移方程**：

  > 对于每个位置 $ i $，我们尝试添加一个新的字符 $ c $（从 'a' 到 'z'）。
  >
  > 定义一个变量 $u$ 来记录 $c$ 的位置，初始时 $u=j$。
  >
  > 根据新字符 $ c $ 和 $ T $ 的第 $ u + 1 $ 个字符的关系。如果它们相等，则记录它们的匹配长度 $u$，否则 $u=ne_u$，继续匹配。知道匹配成功或 $u=0$。
  >
  > 若匹配完后，$c=T_{u+1}$，则 $u=u+1$。
  >
  > 最后，可以得到 $f(i+1,u)=f(i+1,u)+f(i,j)$。  
  > 这是因为，对于每一个长度为 $ i $ 的密码（匹配长度为 $ j $），添加字符 $ c $ 后，会生成一个新的长度为 $ i + 1 $ 的密码（匹配长度为 $u$）。

- **初始化**：$f(0,0)=1$：表示空密码的匹配长度为 $0$。

- **答案**：我们需要计算所有长度为 $n$ 的密码，且匹配长度 $ j < |T| $ 的方案数之和。即 $\sum_{i=0}^{|T|-1} f(n,i)$。

```cpp
#include <bits/stdc++.h>
#define maxn 55
#define mod 1000000007
using namespace std;

int n, len, ans, ne[maxn], f[maxn][maxn];
char s[maxn];
void init() { //初始化 ne 数组（KMP 模板）
    for (int i = 2, j = 0; i <= len; i++) {
        while(j > 0 && s[j + 1] != s[i]) j = ne[j];
        if(s[j + 1] == s[i]) j++;
        ne[i] = j;
    }
}

signed main() {
    cin >> n >> (s + 1);
    len = strlen(s + 1);
    init();
    f[0][0] = 1; //空串
    for (int i = 0; i < n; i++) //枚举密码长度，由于f[i][j]是用来更新别的f[i+1][...]的动态规划值，因此不需要枚举到n
        for (int j = 0; j < len; j++) { //枚举匹配长度（不可以是 |T|）
            for (char c = 'a'; c <= 'z'; c++) { //枚举密码的j + 1的位置的字符
                int u = j;
                while(u > 0 && s[u + 1] != c) u = ne[u]; //匹配过程
                if(s[u + 1] == c) u++; //若匹配，则u++
                f[i + 1][u] = (f[i + 1][u] + f[i][j]) % mod; //转移方程
            }
        }
    for (int i = 0; i < len; i++) ans = (ans + f[n][i]) % mod; //答案
    cout << ans << endl;
    return 0;
}
```

# 状态压缩DP

## [小国王](https://www.acwing.com/problem/content/1066/)

这种**棋盘放置类**问题，在没有事先知道一些特定**性质**的情况下来做，都会想到**爆搜**。

本题的数据规模，也是向着**爆搜**去设置的。

如果我们直接**爆搜**，则时间复杂度为 $O(2^{n^2})$，是会超时的。因此会想到用**记忆化搜索**来进行优化。

---

考虑一下如何进行**动态规划**。

由于在第 $i$ 层放置国王的行为，受到 $i−1$ 层和 $i+1$ 层以及 $i$ 层的状态影响，那么我们就可以规定从上往下枚举的顺序，这样考虑第 $i$ 层状态时，只需考虑 $i−1$ 层的状态即可。

于是乎我们可以考虑把层数 $i$ 作为动态规划的阶段进行线性DP

而第 $i$ 阶段需要记录的就是前 $i$ 层放置了的国王数量 $j$，以及在第 $i$ 层的**棋盘状态** $k$。

这里先分析一下，哪些**棋盘状态**是合法的，哪些**棋盘状态的转移**是合法的。

**合法的棋盘状态**：

![](https://cdn.luogu.com.cn/upload/image_hosting/iatdsv52.png)

如上图所示，绿色方块为摆放国王的位置，红色方块为国王的**攻击范围**。

只要任意国王之间只要不相邻，就是**合法的状态**。

**合法的棋盘状态的转移**：

![](https://cdn.acwing.com/media/article/image/2021/07/12/55909_e5346b6ae3-IMG_0477B08F1210-1.jpeg)

如上图所示，绿色方块为摆放国王的位置，红色方块为王的**攻击范围**。

只要任意王的 纵坐标不相邻，就是**合法的转移状态**。

我们可以用**二进制**来表示当前的状态。

怎么用二进制表示状态？

若 `(state >> i) == 1` 则表示在当前状态中，第 i(0\le i<n)$ 个位置放置了国王（下标从 $0$ 开始）。

再举一个简单的例子，见下图：

![](https://cdn.acwing.com/media/article/image/2021/07/12/55909_24df4784e3-IMG_F0A4293FC94D-1.jpeg)

用二进制表示该状态，就是 $(00100010)_2$（我们存储的时候是倒过来的）。

- **状态**：定义 $f(i,j,k)$ 为考虑前 $i$ 层的棋盘，前 $i$ 层放置了 $j$ 个国王，且第 $i$ 层的状态是 $k$ 的总方案数。

- **状态转移方程**：假设当前第 $i$ 行的状态是 $a$，第 $i-1$ 行的状态是 $b$，第 $i$ 行中有 $c$ 个国王（即 $a$ 中 $1$ 的数量），则状态转移方程为：
  $$
  f(i,j,a)=f(i,j,a)+f(i-1,j-c,b)
  $$

- **初始化**：刚开始的时候，前 $0$ 行放置 $0$ 个国王的方案数为 $1$。因此，$f_{i\ j\ k}=0,f_{0\ 0\ 0}=1$。

- **答案**：把最后一行的每个状态都枚举一遍，即 $\sum_{i=0}^{2^n-1} f_{n\ k\ i}$，其中 $k$ 表示最多可以放多少个国王。

如果直接这么做，那么时间复杂度为 $O(n^3k2^n2^n)$，是会超时的。

但是可以预处理出所有**合法状态**和**合法转移状态**，由于**合法状态**和**合法转移状态**就那么多，所以真实的运算次数会大大降低。

```cpp
#include <bits/stdc++.h>
#define int long long
using namespace std;

const int N = 13, K = 110, M = 1 << N;
int n, k, ans, cnt[M], f[N][K][M];
//定义 cnt[i] 为状态 i 中国王的数量
//定义 f[i][j][k] 为考虑前 i 层的棋盘，前 i 层放置了 j 个国王，且第 i 层的状态是 k 的总方案数。
vector<int> state; //合法状态
vector<int> head[M]; //合法转移状态
bool check(int x) { //判断是否为合法状态
    for (int i = 0; i < n; i++)
        if((x >> i & 1) && (x >> i + 1 & 1))
            return false;
    return true;
}

int count(int x) { //计算一个状态有多少国王
    int res = 0;
    for (int i = 0; i < n; i++) res += x >> i & 1;
    return res;
}

signed main() {
    cin >> n >> k;
    for (int i = 0; i < 1 << n; i++) //枚举每个状态
        if(check(i)) { //判断条件
            state.push_back(i); //合法状态+1
            cnt[i] = count(i); //记录这个状态共有多少个国王
        }
    for (int i = 0; i < state.size(); i++)
        for (int j = 0; j < state.size(); j++) {
            int a = state[i], b = state[j]; //每一个状态多记下来
            if((a & b) == 0 && check(a | b)) //(a & b) == 0这个条件要求不能上下纵坐标相同，check(a | b)这个条件要求不能成对角线
                head[a].push_back(b); //合法转移状态+1
        }
    f[0][0][0] = 1; //初始化
    for (int i = 1; i <= n; i++) //从前往后枚举每一行
        for (int j = 0; j <= k; j++) //枚举所有国王
            for (int a = 0; a < state.size(); a++) //枚举所有合法状态
                for (auto b:head[state[a]]) { //枚举a所有的合法转移状态
                    int c = cnt[state[a]]; //计录a这个状态中的国王数
                    if(j >= c) f[i][j][state[a]] += f[i - 1][j - c][b]; //状态转移方程
                }
    for (int i = 0; i < 1 << n; i++) ans += f[n][k][i]; //答案
    cout << ans << endl;
    return 0;
}
```

## [玉米田](https://www.acwing.com/problem/content/329/)

其实这道题与上一道题很像，而且由于攻击范围并没有那么的大了，导致变简单了一点。但是仔细读题可以发现这道题有了“坏格子”这个限制，因此要在这里特判。

- **状态**：定义 $f(i,j)$ 为考虑前 $i$ 行的玉米地，且第 $i$ 行的状态是 $j$ 的总方案数。

- **状态转移方程**：假设当前第 $i$ 行的状态是 $j$，第 $i-1$ 行的状态是 $k$，则状态转移方程为：
  $$
  f(i,j)=\{f(i,j)+f(i-1,k)\}\bmod 10^8
  $$

- **初始化**：刚开始的时候，前 $0$ 行不种玉米的方案数为 $1$。因此，$f_{i\ j}=0,f_{0\ 0}=1$。

- **答案**：把最后一行的每个状态都枚举一遍，即 $\{\sum_{i=0}^{2^m-1}f_{n\ i}\}\bmod 10^8$。

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 15, M = 1 << N, mod = 1e8;
int n, m, ans, w[N], f[N][M];
//w[i]表示题目要求的好格子和坏格子的状态，但是与题目给出的相反，即0表示好格子，1表示坏格子
//定义 f[i][j] 为考虑前 i 行的玉米地，且第 i 行的状态是 j 的总方案数。
vector<int> state; //合法状态
vector<int> head[M]; //合法转移状态
bool check(int x) { //判断是否为合法状态
    for (int i = 0; i < m; i++)
        if((x >> i & 1) && (x >> i + 1 & 1))
            return false;
    return true;
}

signed main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++)
        for (int j = 0, t; j < m; j++) {
            cin >> t;
            w[i] += !t * (1 << j); //存下题目要求状态
        }
    for (int i = 0; i < 1 << m; i++)
        if(check(i)) state.push_back(i); //合法状态+1
    for (int i = 0; i < state.size(); i++)
        for (int j = 0; j < state.size(); j++) {
            int a = state[i], b = state[j];
            if((a & b) == 0) head[a].push_back(b); //合法转移状态+1
        }
    f[0][0] = 1;
    for (int i = 1; i <= n; i++)
        for (int j = 0; j < state.size(); j++)
            if (!(state[j] & w[i])) //不能与坏格子碰到一起
                for (auto k:head[state[j]]) //枚举第 i - 1 行的状态
                    f[i][state[j]] = (f[i][state[j]] + f[i - 1][k]) % mod; //状态转移方程
    for (int i = 0; i < 1 << m; i++)
        ans = (ans + f[n][i]) % mod; //答案
    cout << ans << endl;
    return 0;
}
```

## [炮兵阵地](https://www.acwing.com/problem/content/294/)

这题与上两题的不同之处在于：前两题的攻击范围是 $1$，而此题为 $2$，如果只考虑第 $i$ 行，那么第 $i$ 行与 $i-2$ 行的关系可能出错，因此这道题的 DP 要开到三维。且这道题求的是能放的最多炮兵的数量。

- **状态**：定义 $f(i,j,k)$ 为考虑前 $i$ 行地图，且第 $i$ 行的状态是 $j$，第 $i-1$ 行是 $k$ 的总方案数。

- **状态转移方程**：假设当前第 $i$ 行的状态是 $j$，第 $i-1$ 行的状态是 $k$，第 $i-2$ 行的状态是 $p$，第 $i$ 行的炮兵数量是 $c$，则状态转移方程为：
  $$
  f(i,j,k)=\max\{f(i-1,k,p)+c\}\bmod 10^8
  $$

- **初始化**：刚开始的时候，前 $0$ 行最多能放的炮兵数量为 $0$。因此，$f_{0\ 0\ 0}=0$。

- **答案**：把第 $n$ 行和第 $n-1$ 行的每个状态都枚举一遍，即 $\max_{i\ j}\{f_{n\ i\ j}\}$。

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 110, M = 11;
int n, m, ans, w[N], cnt[1 << M], f[2][1 << M][1 << M];
vector<int>state;
vector<int>head[1 << M];
bool check(int st) { //不可以两两相邻，也不可以隔着一个空
    for (int i = 0; i < m; i++) {
        int x = (st >> i & 1), y = (st >> (i + 1) & 1), z = (st >> (i + 2) & 1);
        if((x && y) || (x && z) || (y && z)) return false;
    }
    return true;
}

int Count(int st) { //计算这个状态中有多少个1
    int res = 0;
    for (int i = 0; i < m; i++) res += st >> i & 1;
    return res;
}

signed main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++)
        for (int j = 0; j < m; j++) {
            char c;
            cin >> c;
            w[i] += (c == 'H') * (1 << j); //存下每种状态
        }
    for (int i = 0; i < 1 << m; i++)
        if(check(i)) state.push_back(i), cnt[i] = Count(i); //合法状态
    for (auto cur:state)
        for (auto pre:state)
            if(!(cur & pre))
                head[cur].push_back(pre); //合法转移状态
    for (int i = 1; i <= n; i++)
        for (auto st:state) //第i行的状态
            if(!(st & w[i]))
                for (auto p1:head[st]) //第i-1行的状态
                    for (auto p2:head[p1]) //第i-2行的状态
                        if(!(st & p2))
                            f[i & 1][st][p1] = max(f[i & 1][st][p1], f[(i - 1) & 1][p1][p2] + cnt[st]); //状态转移方程
    for (auto i:state)
        for (auto j:head[i])
            ans = max(ans, f[n & 1][i][j]); //答案
    cout << ans << endl;
    return 0;
}
```

## [愤怒的小鸟](https://www.acwing.com/problem/content/description/526/)

一般抛物线方程：$ y = ax^2 + bx + c $。

题目中的抛物线有两个特点：
1. 过原点，即 $ c = 0 $。
2. 开口向下，即 $ a < 0 $。

因此抛物线方程为：$ y = ax^2 + bx $，有两个未知数，因此两点即可确定一条抛物线。

方程中有两个参数 $ a $ 和 $ b $，因此我们可以用具体两个点的坐标来唯一的确定一条抛物线。

参数的计算公式如下：

$$
\begin{cases}
y_1 = ax_1^2 + bx_1 \\
y_2 = ax_2^2 + bx_2
\end{cases}
\quad \Rightarrow \quad
\begin{cases}
a = \frac{\frac{y_1}{x_1} - \frac{y_2}{x_2}}{x_1 - x_2} \\
b = \frac{y_1}{x_1} - ax_1
\end{cases}
$$
因此最多有 $ n^2 $ 个不同的抛物线。接下来求出所有不同的抛物线，及其能覆盖的所有点的点集。

此时问题变成了经典的“重复覆盖问题”，即给定 01 矩阵，要求选择尽量少的行，将所有列覆盖住。这里标准做法是使用 Dancing Links。

但由于 $ n\le 18 $，因此可以直接使用状态压缩 DP 求解，代码更简单。

$f_i$ 表示当前已经覆盖的列的状态是 $ i $ 需要最少的抛物线的数量。

转移时随便找到当前未被覆盖的某一列 $ x $，然后枚举所有包含 $ x $ 的行 $ j $ 来选择即可。

即：$f_{i | j} = \min\{f_i + 1\}$。

预处理时需要枚举所有点对来确定抛物线，然后枚举其余点是否在抛物线上，初始状态是 $f_0=0$，总的计算量是 $ O(n^3) $。

最后的答案需要计算最后的状态，即所有小猪都被打到了的最少方案 $f_{2^n-1}$。

状态压缩 DP 的过程中，一共有 $ 2^n $ 个状态，每个状态需要 $ O(n) $ 的计算量，因此每个关卡需花费的时间复杂度是 $ O(n^3 + n2^n) $，总时间复杂度是 $ O(T(n^3 + n2^n)) $。

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 20, M = 1 << N;
const double eps = 1e-8;
int n, m, T;
struct node {
    double x, y; //存下每只小猪的坐标
} ver[N];
int path[N][N], f[M]; //path[i][j]表示小猪 i 与小猪 j 的抛物线能覆盖的小猪数量，f[i]如上文分析
int cmp(double a, double b) { //比较大小
    if (fabs(a - b) < eps) return 0;
    if (a > b) return 1;
    return -1;
}

void init() { //初始化
    memset(path, 0, sizeof(path));
    for (int i = 0; i < n; i++) {
        path[i][i] = 1 << i; //自己到自己的抛物线只能覆盖自己
        for (int j = 0; j < n; j++) {
            double x1 = ver[i].x, y1 = ver[i].y;
            double x2 = ver[j].x, y2 = ver[j].y;
            if (!cmp(x1, x2)) continue;  //单独一个点无法生成参数a,b的抛物线
            double a = (y1 / x1 - y2 / x2) / (x1 - x2); //套用公式，算出 a,b
            double b = (y1 / x1) - a * x1;
            if (cmp(a, 0.0) >= 0) continue; //a必须小于0
            //参数a,b的抛物线已生成，枚举他经过的所有点
            for (int k = 0; k < n; k++) {
                double x = ver[k].x, y = ver[k].y;
                if (!cmp(y, a * x * x + b * x)) path[i][j] += 1 << k; //如果包含进了当前枚举到的抛物线，则放入点集
            }
        }
    }
}

void solve() {
    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> ver[i].x >> ver[i].y;
    init(); //初始化
    memset(f, 0x3f, sizeof(f));
    f[0] = 0; //DP 初始化
    for (int cur_st = 0; cur_st + 1 < (1 << n); cur_st++) { //枚举每一种状态
        int t = -1;
        for (int i = 0; i < n; i++)
            if (!(cur_st >> i & 1)) //如果不包含这个小猪，则选择这个小猪待会用来更新 DP
                t = i;
        for (int i = 0; i < n; i++) {
            int ne_st = path[t][i] | cur_st; //抛物线&当前状态后的点集
            f[ne_st] = min(f[ne_st], f[cur_st] + 1); //DP 更新
        }
    }
    cout << f[(1 << n) - 1] << endl; //答案
}

int main() {
    cin >> T;
    while (T--) solve();
    return 0;
}
```

## [环形石子合并](https://www.acwing.com/problem/content/description/1070/)

这道题与石子合并那道题本质上是一样的，但是多了个环形限制，只要断环为链就行了。即把原数组复制一遍放在后面。

```cpp
#include <bits/stdc++.h>
#define maxn 210
using namespace std;

int n, a[maxn * 2], w[maxn * 2][maxn * 2], f[maxn * 2][maxn * 2], g[maxn * 2][maxn * 2], maxans, minans = 1e9;
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = n + 1; i <= 2 * n - 1; i++)
        if (i % n != 0) a[i] = a[i % n]; //断环为链
        else a[i] = a[n];
    int len = n;
    n = 2 * n + 1;
    for (int i = 1; i <= n; i++)
        for (int j = i; j <= n; j++)
			w[i][j] = w[i][j - 1] + a[j];
    memset(f, 0x3f, sizeof(f));
    for (int i = n; i >= 1; i--) {
        f[i][i] = 0;
        for (int j = i + 1; j <= n; j++) {
            for (int k = i; k < j; k++)
				f[i][j] = min(f[i][j], f[i][k] + f[k + 1][j]);
            f[i][j] += w[i][j];
        }
    }
    for (int i = n; i >= 1; i--)
        for (int j = i + 1; j <= n; j++) {
            for (int k = i; k < j; k++)
				g[i][j] = max(g[i][j], g[i][k] + g[k + 1][j]);

        }

    	minans = min(minans, f[i][i + len - 1]);

    }
```

## [能量项链](https://www.acwing.com/problem/content/322/)

给定 $n$ 个数，相邻两数可以合并，并同时释放出能量，但不同合并方式所获得的能量不同。

要求设计一个聚合顺序，使这串数释放出的总能量最大。

---

众所周知（都众所周知了我还说干嘛），环形不好处理，因此我们断环为链，复制一份在数组的后面。

接着，这道题是一道 `DP` 题目（很明显使用合并果子的做法不是正确的，贪心不行就换 `DP`）。由于每一个区间的答案都是可以由子区间得来的，所以考虑区间 `DP`。

1. 定义：定义 $f_{l\ r}$ 为合并区间 $l\sim r$ 的最优答案。
2. 状态转移方程：定义 $i,j$ 为区间左端点与右端点，而 $k$ 为它们的分割点，可得 $f_{i\ j}=\max\{f_{i\ k}+f_{k+1\ j}+\text{价值}\}$。
此处，我们设 $a_{i\ 0/1}$ 为第 $i$ 个点的头标记与尾标记。而价值为两颗珠子的合并后的值：$a_{i\ 0}\times a_{k+1\ 0}\times a_{j\ 1}$。因此，状态转移方程为 $f_{i\ j}=\max\{f_{i\ k}+f_{k+1\ j}+a_{i\ 0}\times a_{k+1\ 0}\times a_{j\ 1}\}$。
3. 初始化：$f_{i\ i}=0$。
4. 答案：$\max\{f_{i\ \ i+n-1}\}$。
5. 推到顺序与范围：我们可以注意到，如果 $i$ 从 $1$ 搜到 $2n-1$，则 $f_{k\ j}$ 还没求出来。也就是说，有些值我们还没有更新就要用！因此，改变推到顺序，$i$ 从 $2n-1$ 枚举到 $1$ 就可以了。
至于 $i,j,k$ 的范围，$i$ 的范围已经说过了，这里不再阐述。$j$ 要求必须 $>i$，因此 $j$ 的范围是 $i+1\sim 2n-1$。而 $k$ 要求是 $i$ 和 $j$ 的分割点，因此 $i\le k \le j$，但是 `DP` 中涉及到了 $k+1$，必须保证 $k+1\le j$，否则无意义。综上所述 $1\le i \le 2n-1,i+1\le j \le 2n-1,i\le k<j$。

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, m, ans = -1, a[210][2], f[210][210]; //由于复制了数组，因此空间要开2倍
/*
a[i][0]表示i的头标记，a[i][1]表示i的尾标记
f[l][r]表示合并l~r的最优答案
*/
int main() {
	cin >> n;
	m = n * 2 - 1; //由于后面基本用不到n，直接让m=2n-1节省打字时间，且可以减少常数（循环不变量）
	for (int i = 1; i <= n; i++) {
		cin >> a[i][0];
		a[i + n][0] = a[i][0]; //copy
	}
	for (int i = 1; i <= m - 1; i++)
		a[i][1] = a[i + 1][0]; //copy
	a[m][1] = a[n][0]; //还是在copy
	for (int i = m; i >= 1; i--) { //从后往前推
		f[i][i] = 0; //初始化
		for (int j = i + 1; j <= m; j++)
			for (int k = i; k < j; k++)
				f[i][j] = max(f[i][j], f[i][k] + f[k + 1][j] + a[i][0] * a[k + 1][0] * a[j][1]); //状态转移
	}
	for (int i = 1; i <= n; i++)
		ans = max(ans, f[i][i + n - 1]); //找答案
	cout << ans << endl;
    return 0;
}
```

## [加分二叉树](https://www.acwing.com/problem/content/481/)

给定一个含有 $n$ 个结点的二叉树的**中序遍历**序列中每个节点的**权值**。

定义一棵子树的分数为 $\text{左子树的分数×右子树的分数+根节点的权值}$。

额外规定**空树**的分数为 $1$，**叶子**的分数为该点的权值。

求一种满足该中序遍历的建树方案，使得整棵树的分数最大。

---

- **定义**：$f(i,j)$ 表示中序遍历是 $w_{i\sim j}$ 的所有二叉树的得分的最大值。
- **状态转移方程**：$f(i,j) = \max\{f(i,k-1) \times f(k+1,j) + w_k\}$，即将 $f_{i\ j}$ 表示的二叉树集合按根节点分类，则根节点在 $k$ 时的最大得分即为 $f(i,k-1) \times f(k+1,j) + w_k$，则 $f(i,j)$ 即为遍历 $k$ 所取到的最大值。

- **初始化**：$f_{i\ i}=0$。

- **答案**：$f_{1\ n}$。

在 DP 的过程中，一边 DP 一边记录最大分数的根节点。之后再递归输出先序遍历即可。

```cpp
#include <bits/stdc++.h>
#define maxn 35
using namespace std;

int n, w[maxn], f[maxn][maxn], root[maxn][maxn];
void dfs(int l, int r) {
    if(l > r) return; //越界了，返回
    int k = root[l][r]; //根节点
    cout << k << " ";
    dfs(l, k - 1); //递归左子树
    dfs(k + 1, r); //递归右子树
}

signed main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> w[i];
    for (int l = n; l >= 1; l--) //枚举左端点
        for (int r = l; r <= n; r++) { //枚举右端点
            for (int k = l; k <= r; k++) { //枚举根节点
                int left = l == k ? 1 : f[l][k - 1]; //左子树
                int right = r == k ? 1 : f[k + 1][r]; //右子树
                int score = left * right + w[k]; //分数
                if(l == r) score = w[k]; //叶子节点
                if(f[l][r] < score) {
                    f[l][r] = score; //更新 DP
                    root[l][r] = k; //记录根节点
                }
            }
        }
    cout << f[1][n] << endl; //最后的答案
    dfs(1, n); //递归输出先序遍历
    return 0;
}
```

## [凸多边形的划分](https://www.acwing.com/problem/content/description/1071/)

回归本题，本题是一个给定的凸多边形求**三角剖分**的最小费用方案。

很显然一个凸多边形的剖分方案并不唯一：

在选定多边形中两个点后，找出三角形的第三个点的方案有 $n−2$ 个。

然后还要分别**划分**他的左右两块区域。

因此我们就会想到用**记忆化搜索**或者**区间 DP** 来进行处理。

- **定义**：当前划分到的多边形的左端点是 $l$，右端点是 $r$ 中所有三角形的顶点权值乘积之和的最小值。
- **状态转移方程**：枚举它们的分界点，也就是第三点 $k$，即 $f_{i\ j}=\max\{f_{i\ k-1}+f_{k+1\ j}+w_i\times w_k\times w_j\}$。

- **初始化**：$f_{i\ i+1}=0$。

- **答案**：$f_{1\ n}$。

```cpp
#include <bits/stdc++.h>
#define int __int128 //全新登场，__int128
#define maxn 55
using namespace std;

#define typeOfReadAndWrite __int128
typeOfReadAndWrite read() {
	typeOfReadAndWrite x = 0, f = 1;
	char ch = getchar();
	while(ch < '0' || ch > '9') f *= (ch == '-' ? -1 : 1), ch = getchar();
	while(ch >= 48 && ch <= 57) x = (x << 1) + (x << 3) + (ch ^ 48), ch = getchar();
	return x * f;
}

void write(typeOfReadAndWrite x) {
	if(x < 0) putchar('-'), x = -x;
	if(x > 9) write(x / 10);
	putchar(x % 10 ^ 48);
}


int n, w[maxn], f[maxn][maxn];
signed main() {
    n = read();
    for (int i = 1; i <= n; i++) w[i] = read();
    for (int l = n; l >= 1; l--) //倒着枚举
        for (int r = l + 2; r <= n; r++) { //必须得选三个点及以上
            f[l][r] = 1e30; //准备找最小值
            for (int k = l + 1; k < r; k++)
                f[l][r] = min(f[l][r], f[l][k] + f[k][r] + w[l] * w[k] * w[r]); //状态转移方程
        }
    write(f[1][n]); //答案
    return 0;
}
```

## [棋盘分割](https://www.acwing.com/problem/content/323/)

这个题目很重要的条件就是：
- 每次切割都只能沿着棋盘格子的边进行。

关于这个条件的解释，题面中已经写得很清楚了，我就不赘述了。

考虑使均方差 $\sigma = \sqrt{\frac{\sum_{i=1}^{n}(x_i - \bar{x})^2}{n}}$ 最小

则考虑使标准差 $\alpha = \frac{\sum_{i=1}^{n}(x_i - \bar{x})^2}{n}$ 最小

分析得到 $\bar{x} = \frac{\sum_{i=1}^{n} x_i}{n}$ 无论 $x_i$ 的取值，$\bar{x}$ 恒等于 $\frac{\sum_{i=1}^{8} \sum_{j=1}^{8} w_{i,j}}{n}$，其中 $w_{i,j}$ 代表每一个格子上的数。

所以可以预处理 $\bar{x}$。

此时可以发现将每个 $x_i$ 分开考虑，使得 $\frac{(x_i - \bar{x})^2}{n}$ 最小，即考虑使用动态规划。

由于此题是一个比较明显的区间分割问题，考虑使用区间 DP。

令 $f_{x_1, y_1, x_2, y_2, k}$ 表示将子矩阵 $(x_1, y_1)(x_2, y_2)$ 划分为 $k$ 个子矩阵的最小标准差 $\alpha$，则最终答案为 $\sqrt{f_{1, 1, 8, 8, n}}$。

边界 $ k = 1 $ 时，$ f_{x_1, y_1, x_2, y_2, k} = \frac{((\sum_{i=x_1}^{x_2} \sum_{j=y_1}^{y_2} w_{i,j}) - \bar{x})^2}{n} $

在状态转移时，分类讨论水平切还是竖直切，枚举在哪里切，再分类讨论是要哪一块。

则可得出状态转移方程（详见代码）。

注意可用二维前缀和来维护子矩阵，从而 $ O(1) $ 得到子矩阵的和。

总时间复杂度 $ O(8^5 n) $。

由于循环太多，采用记忆化搜索写。

代码中 `x` 即为 $\bar{x}$，`get` 函数是求 $\frac{(x_i - \bar{x})^2}{n}$，`s` 即为二维前缀和数组。

注意 `x` 计算时别忘了强制转换 `s` 数组的

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 20, M = 10;
int n, s[M][M];
double X, f[M][M][M][M][N];
double get(int x1, int y1, int x2, int y2) { //计算这个矩阵的值
    double sum = s[x2][y2] - s[x2][y1 - 1] - s[x1 - 1][y2] + s[x1 - 1][y1 - 1] - X;
    return (double)sum * sum / (double)n; //按照分析所说的返回
}

double dp(int x1, int y1, int x2, int y2, int w) {
    if(f[x1][y1][x2][y2][w] >= 0) return f[x1][y1][x2][y2][w]; //记忆化搜索
    if(w == 1) return f[x1][y1][x2][y2][w] = get(x1, y1, x2, y2); //w = 1的时候的特判
    f[x1][y1][x2][y2][w] = 1e9;
    for (int i = x1; i < x2; i++) { //横着切一刀
        f[x1][y1][x2][y2][w] = min(f[x1][y1][x2][y2][w], dp(i + 1, y1, x2, y2, w - 1) + get(x1, y1, i, y2)); //下面的
        f[x1][y1][x2][y2][w] = min(f[x1][y1][x2][y2][w], dp(x1, y1, i, y2, w - 1) + get(i + 1, y1, x2, y2)); //上面的
    }
    for (int i = y1; i < y2; i++) { //竖着切一刀
        f[x1][y1][x2][y2][w] = min(f[x1][y1][x2][y2][w], dp(x1, i + 1, x2, y2, w - 1) + get(x1, y1, x2, i)); //右边的
        f[x1][y1][x2][y2][w] = min(f[x1][y1][x2][y2][w], dp(x1, y1, x2, i, w - 1) + get(x1, i + 1, x2, y2)); //左边的
    }
    return f[x1][y1][x2][y2][w]; //答案
}

signed main() {
    cin >> n;
    for (int i = 1; i <= 8; i++)
        for (int j = 1; j <= 8; j++) {
            cin >> s[i][j];
            s[i][j] += s[i - 1][j] + s[i][j - 1] - s[i - 1][j - 1]; //二位前缀和
        }
    X = (double)s[8][8] / (double)n; //计算X
    memset(f, -1, sizeof(f));
    printf("%.3lf\n", sqrt(dp(1, 1, 8, 8, n))); //最终的答案
    return 0;
}
```

# 树形DP

## [树的最长路径](https://www.acwing.com/problem/content/description/1074/)

我们知道：树上**任意两点**的路径是**唯** 确定的，因此我们可以暴力枚举**起点**和**终点**找出最长路径

如果这样做的话，我们来思考一下时间复杂度：

1. 枚举**起点**和**终点** — $O(n^2)$。
2. 找出两点之间的路径长度 — $O(\log n)$。

这里找出两点之间的路径有很多种做法，预处理 st 表然后倍增求 LCA、树链剖分、Link-Cut Tree。

但是光是枚举**起点**和**终点**，时间复杂度 就直接拉满了，显然这种做法不可取。

既然这 $O(n^2)$ 条路径不能一一枚举，那么有什么方式可以把他们**分类枚举**呢？

我们考虑换一种**枚举方式**：枚举路径的**起点和终点** → 枚举路径的**中间节点**。

我们先讨论一下，对于给定拓扑结构的树里的任意节点，经过他的路径有哪些：

![](https://cdn.acwing.com/media/article/image/2021/08/25/55909_2a282a8605-IMG_61398D0A7BFC-1.jpeg)

观察我标出的**红色节点**，那么经过他的路径有：

1. 以其**子树中的某个节点**作为**起点**，以他作为**终点**的**粉色路径**。
2. 以其**子树中的某个节点**作为**起点**，以**子树中的某个节点**作为**终点**的**蓝色路径**。
3. 以其**子树中的某个节点**作为**起点**，以**非其子树的节点**作为**终点**的**橙色路径**。

对于第 $1$ 种情况，我们可以直接递归处理其子树，找出到当前子树根节点最长的路径长度即可。

对于第 $2$ 种情况，我们在处理第 $1$ 种情况时，顺便找出 $1$ 类路径的**次长路径**。

把**最长**和**次长**拼在一起，就是我们要的第 $2$ 种情况。

而对于第 $3$ 种情况，我们可以把它归类为其**祖先节点**的第 $1,2$ 种情况，让其**祖先节点**去处理即可。

把上述两类路径的分析，用**自制 DP 分析法**写出，就是如下形式：

- **状态**：$f_{1/2,i}$ 表示以节点 $i$ 为根的子树中，从子树某个节点到 $i$ 的最长路为 $f_{1,i}$，次长路为 $f_{2,i}$，两个加在一起表示路径长度的最大值 $\max$。

- **状态转移方程**：

$$
\begin{cases}
f_{1,i} = max(f_{1,i_c1} + w_{i,i_c1}) & c1 \in i_{child} \\
f_{2,i} = max(f_{1,i_c2} + w_{i,i_c2}) & c2 \in i_{child} \& c2 \neq c1 \& f_{2,i} \leq f_{1,i}
\end{cases}
$$

- **初始化 & 答案**：抱歉，本题没有这个东西。

最后的**真·答案**是 $\max_{i=1}^n f_{1\ i}+f_{2\ i}$。

```cpp
#include <bits/stdc++.h>
#define maxn 10010
using namespace std;

struct edge {
    int to, w;
};

int n, ans = -1e9, dis[maxn], f[maxn][2];
vector<edge> linker[maxn];
void add(int x, int y, int w) {
    linker[x].push_back({y, w});
}

void dfs(int x, int fa) {
    for (auto v:linker[x])
        if(v.to != fa) dfs(v.to, x); //先去枚举子树
    for (auto v:linker[x]) {
        int son = v.to, w = v.w;
        if(son == fa) continue;
        if(f[son][0] + w >= f[x][0]) f[x][1] = f[x][0], f[x][0] = f[son][0] + w; //更新次大值和最大值
        else if(f[son][0] + w > f[x][1]) f[x][1] = f[son][0] + w; //更新次大值
    }
    ans = max(ans, f[x][0] + f[x][1]); //找出最大的答案
}

int main() {
    cin >> n;
    for (int i = 1; i < n; i++) {
        int x, y, w;
        cin >> x >> y >> w;
        add(x, y, w), add(y, x, w);
    }
    dfs(1, 0);
	cout << ans << endl;
	return 0;
}
```

## [树的中心](https://www.acwing.com/problem/content/1075/)

此处就要引入**换根DP**的思想了。

换根DP 一般分为三个步骤：

1. 指定任意一个根节点。
2. 一次 `DFS` 遍历，统计出当前子树内的节点对当前节点的贡献。
3. 一次 `DFS` 遍历，统计出当前节点的父节点对当前节点的贡献，然后合并统计答案。

那么我们就要先 `DFS` 一遍，预处理出当前子树对于根的最大贡献（距离）和次大贡献（距离）。

处理次大贡献（距离）的原因是：

如果当前节点是其父节点子树的最大路径上的点，则父节点子树的最大贡献不能算作对该节点的贡献。

因为我们的路径是简单路径，不能走回头路。

然后我们再 `DFS` 一遍，求解出每个节点的父节点对他的贡献（即每个节点往上能到的最远路径。

两者比较，取一个 $\max$ 即可。

- **状态**：定义 $ f_{1/2,i} $ 为对于树中 $ i $ 号节点，其子节点的最大贡献为 $ f_{1,i} $，父节点的最大贡献为 $ f_{2,i} $。

- **状态转移方程**：

$$
\begin{cases}
f_{1,i} = \max\{f_{1,i_c}\} \\
f_{2,i} = \max\{f_{1,i_p}, f_{2,i_p}\} \quad (son_{i_p} \neq i)
\end{cases}
$$

```cpp
#include <bits/stdc++.h>
#define maxn 10010
using namespace std;

struct edge {
    int to, w;
};

int n, ans = 1e9, d1[maxn], d2[maxn], p1[maxn], p2[maxn], up[maxn];
// up[i] 表示从自己的祖宗到自己的最大距离，可以理解为往上探索
vector<edge> linker[maxn];
int dfs(int x, int fa) {
	d1[x] = d2[x] = -1e9;
    for (auto i:linker[x]) {
        int son = i.to, w = i.w;
        if(son == fa) continue;
        int dist = dfs(son, x) + w;
        if(dist > d1[x]) { //更新一下最长和第二长的路径，并记录下从该路径是从哪一个点下去的
            d2[x] = d1[x], d1[x] = dist;
            p2[x] = p1[x], p1[x] = son;
        } else if(dist > d2[x]) {
            d2[x] = dist;
            p2[x] = son;
        }
    }
    if(d1[x] == -1e9) d1[x] = d2[x] = 0; //如果没有改变过该点的距离，就证明这个点是叶节点
    return d1[x];
}

void dep(int x, int fa) {
	for (auto i:linker[x]) {
        int son = i.to, w = i.w;
        if(son == fa) continue;
        if(p1[x] == son) up[son] = max(up[x], d2[x]) + w; //这条路径上已经走过了 
        else up[son] = max(up[x], d1[x]) + w; //这个没有 
        dep(son, x);
    }
}

int main() {
    cin >> n;
    for (int i = 1, x, y, w; i < n; i++) cin >> x >> y >> w, linker[x].push_back({y, w}), linker[y].push_back({x, w});
	dfs(1, 0), dep(1, 0);
	for (int i = 1; i <= n; i++) ans = min(ans, max(d1[i], up[i]));
    cout << ans << endl;
    return 0;
}
```

## [数字转换](https://www.acwing.com/problem/content/description/1077/)

根据题意，单看一个数字 $ x $，那么他能转换到的数有：

1. $ x $ 的约数之和 $ x' $ ($ x' < x $)
2. 所有大于 $ x $，且约数之和等于 $ x $ 的数 $ x_i $

那么我们就能绘出如下抽象图示了：

![](https://cdn.acwing.com/media/article/image/2021/09/01/55909_fdd02f770b-IMG_0A4612307E1F-1.jpeg)

由于任意正整数 $ x $，的 **约数之和** 是 **唯一的**，且本题要求只有约数之和 **小于** 自身才能转换。

故对于所有的 $ x $ 来说，他向 **小于** 自己的数转换的边 **至多** 只有一条，那就是 $ x $ 的 **约数之和** $ x' (x' < x) $。

这样一个图论模型我们就很熟悉了，我们把每一个 **数** 看作一个 **点**，把上述这个 **转换** 看作该点的 **入边**。

每一个 **点**，至多只有一条 **入边**，这就是 **森林** 呀。

建好图后，本题就 **等价于** 找出一个森林中，**直径最长** 的树，并求出该树的 **直径**。

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 5e4 + 10;
int n, sum[N];
int d1[N], d2[N], ans; //d1表示最大，d2表示次大
vector<int>linker[N];
void add(int a, int b) {
    linker[a].push_back(b);
}

void dfs(int x) {
    if (d1[x]) return;  //记忆化搜索
    for (auto son:linker[x]) {
        dfs(son);
        if (d1[son] + 1 >= d1[x]) d2[x] = d1[x], d1[x] = d1[son] + 1;
        else if (d1[son] + 1 > d2[x]) d2[x] = d1[son] + 1;
    }
    ans = max(ans, d1[x] + d2[x]);
}

int main() {
    scanf("%d", &n);
    for (int i = 1; i <= n; i++)
        for (int j = 2; j <= n / i; j++)
            sum[i * j] += i;
    for (int i = 2; i <= n; i++)
        if (sum[i] < i) add(sum[i], i);
    for (int i = 1; i <= n; i++) dfs(i); //这里因为不知道是否再1里面，因此每个都要跑一遍（反正有记搜）
    printf("%d\n", ans);
    return 0;
}
```

## [二叉苹果树](https://www.acwing.com/problem/content/1076/)

这道题似乎是一个**有依赖的背包问题**与**树形DP**的结合，直接给出**自制DP分析法**。

- **状态**：$f_{u\ i}$ 表示以 $u$ 为根节点，保留树枝数量为 $i$ 时的最大获益。
- **状态转移方程**：对两颗子树分别进行枚举子树应保留的树枝数量，可得:

```cpp
f[x][i] = max(f[x][i], f[lc][j - 1] + f[rc][i - j - 1] + tmp);
```

- **初始化**：$f_{u\ i}=0$。
- **答案**：$f_{1\ Q}$。

```cpp
#include <bits/stdc++.h>
#define maxn 110
using namespace std;

struct edge {
    int v, len;
};

vector<edge> linker[maxn];
int n, Q, f[maxn][maxn];
void add(int x, int y, int w) {
    linker[x].push_back({y, w});
}

void dfs(int x, int fa) {
    int lc = 0, rc = 0, lv = 0, rv = 0; //左右儿子的编号与值
    for (auto k:linker[x]) {
        int to = k.v, w = k.len;
        if(to == fa) continue;
        if(lc) rc = to, rv = w;
        else lc = to, lv = w;
        dfs(to, x); //继续递归
    }
    if(!lc || !rc) return; //没有儿子的叶节点
    for (int i = 1; i <= Q; i++) {
        for (int j = i; j >= 0; j--) { //01背包压维了
            int tmp = 0;
            if(j >= 1) tmp += lv;
            if(i - j >= 1) tmp += rv;
            f[x][i] = max(f[x][i], f[lc][j - 1] + f[rc][i - j - 1] + tmp); //更新DP
        }
    }
}

int main() {
    cin >> n >> Q;
    for (int i = 1, x, y, w; i < n; i++) {
        cin >> x >> y >> w;
        add(x, y, w), add(y, x, w);
    }
    dfs(1, 0);
    cout << f[1][Q] << endl;
    return 0;
}
```

## [战略游戏](https://www.acwing.com/problem/content/325/)

一个节点，分选或不选来讨论就可以了。

- **状态**：$f_{i\ 0/1}$ 表示不选 / 选第 $i$ 个节点的最大收益。
- **状态转移方程**：
  - 不选第 $i$ 个节点：那么儿子必须选，$f_{i\ 0}=\sum_{\substack{son\\}} f_{son\ 1}$。
  - 选第 $i$ 个节点：那么儿子可选可不选，$f_{i\ 1}=\sum_{\substack{son\\}} \min\{f_{son\ 0},f_{son\ 1}\}$。
- **初始化**：$f_{i\ 1}=1$。
- **答案**：$\min\{f_{i\ 0}, f_{i\ 1}\}$。

```cpp
#include <bits/stdc++.h>
#define maxn 1510
using namespace std;

int n, fa[maxn], f[maxn][2]; //f[i][0]表不选，f[i][1]表选
vector<int>linker[maxn];
void dfs(int x, int father) {
    f[x][1] = 1;
    for (auto v:linker[x]) {
        if(v == father) continue;
        dfs(v, x);
        f[x][0] += f[v][1];
        f[x][1] += min(f[v][0], f[v][1]);
    }
}

void add(int u, int v) {
    linker[u].push_back(v);
    linker[v].push_back(u);
}

void solve() {
    for (int i = 1; i <= n; i++) while(!linker[i].empty()) linker[i].pop_back();
    memset(f, 0, sizeof(f)), memset(fa, 0, sizeof(fa));
    for (int i = 1, u, len; i <= n; i++) {
        char c;
        cin >> u >> c >> c >> len >> c; u++;
        for (int j = 1, k; j <= len; j++) cin >> k, add(u, k + 1), fa[k + 1] = 1;
    }
    for (int i = 1; i <= n; i++)
        if (!fa[i]) {
            dfs(i, 0);
            cout << min(f[i][0], f[i][1]) << endl;
            return;
        }
}

int main() {
    while(cin >> n) solve();
    return 0;
}
```

## [皇宫看守](https://www.acwing.com/problem/content/1079/)

这样每个结点就有 三种情况 要转移，简略状态机模型如下：

被父节点观察 （0）。
被子节点观察 （1）。
被自己来观察 （2）。

![](https://cdn.luogu.com.cn/upload/image_hosting/o0fbfe2v.png)

状态表示 — $ f_{i,0/1/2} $ 集合： 以结点 $ i $ 为 **根节点** 的子树，在 $ i $ 状态为 $ j $ 的方案

状态表示 — $ f_{i,0/1/2} $ 属性： 方案中，放置哨兵的代价 **最少**

状态计算：

在 $ i $ 上被父节点观察 (0)：
$$
f_{i,0} = \sum_{i_{ch}} \min(f_{i_{ch},1}, f_{i_{ch},2})
$$
$$
i_{ch} \in \{ver | ver \text{是 } i \text{ 的子节点}\}
$$

在 $ i $ 上被子节点观察 (1)：
$$
f_{i,1} = \min(f_{i_{ch},2} + \sum_{other \, i_{ch}} \min(f_{other \, i_{ch},1}, f_{other \, i_{ch},2}))
$$
$$
i_{ch} \in \{ver | ver \text{是 } i \text{ 的子节点}\} \quad other \, i_{ch} \in \{ver | ver \text{是 } i \text{ 的子节点，且 } ver \neq i_{ch}\}
$$

在 $ i $ 上被自己来观察 (2)：
$$
f_{i,2} = \sum_{i_{ch}} \min\{f_{i_{ch},0}, f_{i_{ch},1}, f_{i_{ch},2}\} + w_i
$$
$$
i_{ch} \in \{ver | ver \text{是 } i \text{ 的子节点}\}
$$

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 1510;
int n, f[N][3], w[N];
vector<int>linker[N];
bool st[N];
void add(int a, int b) {
    linker[a].push_back(b);
}

void dfs(int u) {
    f[u][0] = 0;
    f[u][1] = 1e9; //f[u][1]求最小值，初始化为最大值
    f[u][2] = w[u]; //初始化放置自己的代价
    for (auto i:linker[u]) {
        dfs(i);
        f[u][0] += min(f[i][1], f[i][2]);
        f[u][2] += min(min(f[i][0], f[i][1]), f[i][2]);
    }
    for (auto i:linker[u])
        f[u][1] = min(f[u][1], f[u][0] + f[i][2] - min(f[i][1], f[i][2]));
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        int id, cnt, cost;
        cin >> id >> cost >> cnt;
        w[id] = cost;
        while (cnt--) {
            int ver;
            cin >> ver;
            add(id, ver);
            st[ver] = true;
        }
    }
    int root = 1;
    while (st[root]) root++;
    dfs(root);
    printf("%d\n", min(f[root][1], f[root][2]));
    return 0;
}
```

# 数位DP

## [度的数量](https://www.acwing.com/problem/content/1083/)

这是一道数位 dp 的题目，数位 dp 的题目一般会问，**某个区间内，满足某种性质的数的个数**。

对应的数位 dp 问题有相应的解题技巧：
1. 利用前缀和，比如求区间 $[x,y]$ 中的个数，转化成求 $[0,y]\texttt{ 的个数}$ $-$ $[0,x-1]\texttt{ 的个数}$。
2. 利用树的结构来考虑（按位分类讨论）。

该类题目的核心就是**根据数位进行分类讨论**。

思路如下：

第一步，先将数 $x$ 转化成为 $B$ 进制，记为 $N$。

第二步，分类讨论。既然每一位上的数都已给定，我们称之为**每一位的“限定值”都已给出**。比如一个数是 $456$，最高位是 $4$，这一位的“限定值”就是 $4$；第三位上是 $6$，这一位的“限定值”就是 $6​$。如此说来，每位上可取的数的大小就确定了。每一位都分两种情况：
1. 取限定值；
2. 取 $0\sim\texttt{限定值}-1$ 中的任一个数。

其实，对应本题，**这里每个数的取值可能用不到限定值！！！只有限定值是 $1$ 的时候才会用到！！！** 为什么呢？因为本题求的是 $K$ 个位上填 $1$，其他位上填 $0$，根本不会出现其他数字，所以当出现 $3，4，9$ 这样的数字的时候直接 `break`。

对于 $B$ 进制数 $N$（共 $n$ 位），从最高位（第 $n$ 位）开始遍历每 $1$ 位，对于第 $i$ 位，**last** 表示第 $i$ 位之前的所有位取过的 $1$ 的个数。第 $i$ 位上的数字 $x$ 有三种情况：
1. 限定值 $x=0$：此时第 $i$ 位只能取 $0$，不影响 $1$ 的个数，后面所有位可以取 $k-last$ 个 $1$，可以考虑不考虑。
2. 限定值 $x=1$：此时可以取 $0$ 或 $1$。当此位取 $0$ 时，后面的所有位可以取 $k-last$ 个 $1$；当此位取 $1$ 时，后面所有位只能取 $k-last-1$ 个 $1$。
3. 限定值 $x>1$：当前位取 $0$ 或 $1$ 时，处理同 $2$；其他值直接不能取，在本题中没意义。

![](https://cdn.acwing.com/media/article/image/2021/01/22/42785_6920cc9f5c-QQ%E6%B5%8F%E8%A7%88%E5%99%A8%E6%88%AA%E5%9B%BE20210122131839.png)

```cpp
#include <bits/stdc++.h>
using namespace std;

int c[35][35]; //存储组合数
int x, y, K, B;
void init() {
	for(int i = 0; i < 35; i++)
		for(int j = 0; j <= i; j++) {
			if(!j) c[i][j] = 1;
			else c[i][j] = c[i - 1][j] + c[i - 1][j - 1];
		}
}

int dp(int n) {
	if(!n) return 0;
	vector<int> num;
	while(n) num.push_back(n % B), n /= B;
	int ans = 0, one = 0; //one表示已经填了多少个1
	for (int i = num.size() - 1; i >= 0; i--) {
		int o = num[i];
		//若o = 0, 那 o 只能等于 0,到下一位次继续进行划分;
		if(o) { //对第i为进行讨论
			ans += c[i][K - one]; //第 i 位填 0, 剩下 i - 1 位填 K - one 个 1
			//第i位填1的情况
			if(o > 1) {
				if(K - one - 1 >= 0) ans += c[i][K - one - 1]; //第 i 位填 1 , 剩下的 i - 1 位填剩余个 1
				break; //方案数已经枚举完毕, 无须再进行讨论
			} else {
				one++; //o = 1, 剩下的位次不能乱填, 所以再进行讨论划分
				if(one > K) break;
			}
		}
		if(!i && one == K) ans++; //进行到底了, 那么也算一种方案数, 算上
	}
	return ans;
}
```

- **状态**：$f_{i\ j}$ 为长度为 $i$，首位数字为 $j$。

```cpp
#include <bits/stdc++.h>
```

## [Windy数](https://www.acwing.com/problem/content/1085/)

此题与上两题很相似，但是需要考虑到前导零，因此策略有所改变。

**数位DP**：

假设我们当前枚举到第 $i$ 位（设共有 $n$ 位），且第 $i$ 位上的数字为 $x$，那么对于答案中第 $i$ 位数字 $j$ 来说，有两类：

1. $0\sim x-1$ (如果第 $i$ 位是最高位，这里是 $1\sim x-1$)
   - 括号中提到的 $1\sim x-1$ 后面会解释，我们用 $last$ 记录上一位的数字，然后枚举 $j$，如果 $|j-last| \ge 2$ 就累加答案，`res += f[i + 1][j]`；
   
2. $x$
   - 不需要枚举 $j$，$last = x$，再枚举之后的数位即可。

上述做完之后，由于上面的答案都是 $n$ 位的，对于数位个数低于 $n$ 的，再累加到答案中就行了（没有前导 0，肯定是 $n$ 位）。

**$f$ 数组的处理**：

表示一共有 $i$ 位，且最高位数字为 $j$ 的满足 $windy$ 数定义的数的个数。

状态转移：因为第 $i$ 位是 $j$ 已经确定，考虑第 $i-1$ 位，设第 $i-1$ 位数字为 $k$，那么根据 windy 数定义只要 $|k-j| \ge 2$ 就可以转移过来。

$$
f_{i\ j} = \sum_{k=0}^{9} (|k-j|\ge 2) f_{i-1\ k}
$$
**关于前导 0**：

上面提到了枚举的第 $i$ 位是最高位，那么不能填 $0$ ，这里解释一下，如果我们填 $0$ ，那么答案就会加上 `f[i+1][0]`。举这样一个例子，
对于数字 $13$ ，他是满足 windy 数定义的，那么加上前导 0 之后的 $013$ 就不会被 `f[3][0]` 加进去，原因就是 $|0-1|<2$，这样就导致答案漏掉。

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 15;
int a, b, f[N][10];
void init() { //预处理
    for (int i = 0; i <= 9; i++) f[1][i] = 1;
    for (int i = 2; i < N; i++)
        for (int j = 0; j <= 9; j++)
            for (int k = 0; k <= 9; k++)
                if(abs(j - k) >= 2) f[i][j] += f[i - 1][k];
}

int dp(int n) { //数位DP
    if (!n) return 0; //不包含前导0
    
    vector<int>num;
    while(n) num.push_back(n % 10), n /= 10;
    
    int res = 0, ls = -10;
    for (int i = num.size() - 1; i >= 0; i--) {
        int x = num[i];
        for (int j = (i == num.size() - 1); j < x; j++)
            if(abs(j - ls) >= 2) res += f[i + 1][j];
        if(abs(ls - x) < 2) break;
        ls = x;
        if(!i) res++;
    }
    
    for (int i = 1; i < num.size(); i++) //枚举不是n位的
        for (int j = 1; j <= 9; j++)
            res += f[i][j]; //答案累积
    return res;
}

signed main() {
    init();
    cin >> a >> b;
    cout << dp(b) - dp(a - 1) << endl;
    return 0;
}
```

## [数字游戏 II](https://www.acwing.com/problem/content/1086/)

在几道数位 DP 题目练习过后，这类题目重点在于找到左边那一类如何直接计算。

对于这一题来说，假设我们当前枚举到的第 $i$ 位，且第 $i$ 位上的数字是 $x$，那么对于答案中的第 $i$ 位数字 $j$ 来说，可以填两类数：

1. **$0\sim x-1$**
   - 我们用 `last` 表示到当前为止，前面数位上的数字之和。对此，当前第 $i$ 位数字为 $j$，前面数字之和为 `last`，那么后 $i$ 位（包括 $j$ 这一位）数字之和 `sum` 与 `last` 的关系就是 `(last + sum) % N == 0`，那么 `sum % N` 的结果等价于 `(-last) % N`。
   - 所以 `res += f[i+1][j][(-last % N)]`；（后面会提到 f 数组的处理）

2. **x**
   - 如果 $j$ 填 $x$，那么不用枚举了，`last += x`，再枚举下一位即可。

---

`f[i][j][k]` 表示一共有 $i$ 位，且最高位数字是 $j$，且所有位数字和模 $N$ 结果为 $k$ 的数的个数。

状态转移：
- 因为第 $i$ 位已经是 $j$，且所有数字之和模 $N$ 为 $k$，所以我们考虑第 $i-1$ 位，假设第 $i-1$ 位数字是 $x$，由于 $j$ 已经知道，
- 那么剩下的 $i-1$ 位数字之和模 $N$ 的结果就是 `(k-j)%N`，那么状态转移方程就是：
  $$
  f[i][j][k] = \sum_{k=0}^{N-1} \sum_{x=0}^{9} f[i-1][x][(mod(k-j,N))]
  $$

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 35, M = 110;
int a, b, p, f[N][10][M];
int mod(int x, int y) { //超级取模
    return (x % y + y) % y;
}

void init() { //超级预处理
    memset(f, 0, sizeof(f));
    for (int i = 0; i <= 9; i++) f[1][i][i % p]++; //超级初始化
    for (int i = 2; i < N; i++) //超级枚举i
        for (int j = 0; j <= 9; j++) //超级枚举j
            for (int k = 0; k < p; k++) //超级枚举k
                for (int x = 0; x <= 9; x++) //超级枚举x
                    f[i][j][k] += f[i - 1][x][mod(k - j, p)]; //超级转移方程
}

int dp(int n) { //超级DP
    if(!n) return 1; //超级特判
    
    vector<int>num;
    while(n) num.push_back(n % 10), n /= 10;
    
    int res = 0, ls = 0;
    for (int i = num.size() - 1; i >= 0; i--) {
        int x = num[i];
        for (int j = 0; j < x; j++)
            res += f[i + 1][j][mod(-ls, p)]; //超级计算
        ls += x; //超级迭代
        if (!i && ls % p == 0) res++; //超级特殊判断
    }
    return res;
}
 
void solve() {
    init();
    cout << dp(b) - dp(a - 1) << endl;
}

signed main() {
    while(cin >> a >> b >> p) solve();
    return 0;
}
```

## [不要62](https://www.acwing.com/problem/content/1087/)

~~还是相同的套路~~，假设数字 `num` 当前枚举到第 $i$ 位，且第 $i$ 位数字是 $x$，那么对于答案的第 $i$ 位来说，假设是 $j$，有两种填法：

1. **$0\sim x-1$**
   - 我们用 `last` 表示 `num` 中第 $i$ 位的高一位数字，那么根据题意，$j$ 不能是 $4$ 且当 `last` 是 $6$ 的时候 $j$ 不能为 $2$。
     这两种情况在枚举的时候特判一下即可。
   - 然后，累加到答案中，`res += f[i+1][j]`。

2. **$x$**
   - 那么这一位就不用处理了，更新一下 `last` 即可，`last = x`。

---

`f[i][j]` 表示一共有 $i$ 位，且最高位是 $j$ 的满足新的土牌照定义的数的个数。

- 第 $i$ 位是 $j$ 已知，那么考虑第 $i-1$ 位，假设为 $k$，根据牌照定义，
  - $j$ 为 $6$ 时 $k$ 不为 $2$ 且 $k$ 不能等于 $4$。
- 那么得到转移方程，见代码~

```cpp
#include <bits/stdc++.h>
#define int long long
using namespace std;

const int N = 35;
int a, b, f[N][10];
void init() {
    for (int i = 0; i <= 9; i++) if(i != 4) f[1][i] = 1;
    for (int i = 2; i < N; i++)
        for (int j = 0; j <= 9; j++) {
            if (j == 4) continue;
            for (int k = 0; k <= 9; k++) {
                if(k == 4 || (j == 6 && k == 2)) continue;
                f[i][j] += f[i - 1][k];
            }
        }
}

int dp(int n) {
    if(!n) return 1;
    
    vector<int>num;
    while (n) num.push_back(n % 10), n /= 10;
    
    int res = 0, ls = 0;
    for (int i = num.size() - 1; i >= 0; i--) {
        int x = num[i];
        for (int j = 0; j < x; j++) {
            if ((ls == 6 && j == 2) || (j == 4)) continue;
            res += f[i + 1][j];
        }
        if((ls == 6 && x == 2) || (x == 4)) break;
        ls = x;
        if(!i) res++;
    }
    return res;
}

void solve() {
    cout << dp(b) - dp(a - 1) << endl;
}

signed main() {
    init();
    while(cin >> a >> b && a && b) solve();
    return 0;
}
```

## [恨7不成妻](https://www.acwing.com/problem/content/1088/)

其实这道题是要一步一步都弄清楚，也不是很难。

**状态表示**：

`f[i][j][a][b]` 表示：所有 $i$ 位数，最高位是 $j$，这个数字本身模 $7$ 为 $a$，各位数之和模 $7$ 为 $b$。

$i$ 位数可以从 $i - 1$ 位数转移而来。

记 $i - 1$ 位数本身模 $7$ 为 $x$，记 $i$ 位数本身模 $7$ 为 $a$, 那么：
$ a = (j \times 10^{i-1} + x) \implies x = a - j \times 10^{i-1} \text{(均为模 7 下的运算)} $。

记 $i - 1$ 位数各个位的数之和模 $7$ 为 $x$，记 $i$ 位数各个位的数之和模 $7$ 为 $b$，那么：
$ b = j + x \implies x = b - j \text{(均为模 7 下的运算)} $。

可得 `f[i][j][a][b]` 应该从 `f[i - 1][k][a - j * 10^{i-1}][b - j]` 转移而来。

**属性**：

因为要求的是平方和，所以需要处理的值较复杂。

观察得所有 $i$ 位数应该形如 $\overline{jA}$ ($A$ 是一个 $i-1$ 位数)。

平方和即表示为:
$ (jA_1)^2 + (jA_2)^2 + \ldots + (jA_t)^2 $
$ = t(j \times 10^{i-1})^2 + 2(j \times 10^{i-1})(A_1 + A_2 + \ldots A_t) + (A_1^2 + A_2^2 + \ldots A_t^2) $

因此发现，需要存储的值有 $t$ (合法的数的个数，记为 $s_0$)、$A$ 的和 (记为 $s_1$)、$A$ 的平方和 (记为 $s_2$)。

**状态转移**：

$t$: 因为表示的是合法的数的个数，因此直接累加上一个状态的 $t$ 即可。

$A$ 的和:
$ jA_1 + jA_2 + \ldots jA_t = t(j \times 10^{i-1}) + (A_1 + A_2 + \ldots A_t) $

A 的平方和:
$ t(j \times 10^{i-1})^2 + 2(j \times 10^{i-1})(A_1 + A_2 + \ldots A_t) + (A_1^2 + A_2^2 + \ldots A_t^2) $
$ => s'_0(j \times 10^{i-1})^2 + 2(j \times 10^{i-1})s'_1 + s'_2 $。

这样就知道如何预处理 `f[i][j][s_1][s_2]` 就可以开始遍历窗口了。

```cpp
#include <bits/stdc++.h>
#define int long long
using namespace std;

const int N = 20, P = 1e9 + 7;
int T, L, R;
struct node {
    int s0, s1, s2;
} f[N][10][10][10];

int power7[N], power9[N];
int mod(int v, int u) { //超级mod
    return (v % u + u) % u;
}

void init() {
    for (int i = 0; i <= 9; i++)
        if (i != 7) {
            auto &v = f[1][i][i % 7][i % 7]; //简写
            v.s0++; //个数+1
            v.s1 += i; //和+i
            v.s2 += i * i; //平方和+i^2
        }
    for (int i = 2, p = 10; i < N; i++, p *= 10)
        for (int j = 0; j <= 9; j++) {
            if(j == 7) continue; //不可以有7
            for (int a = 0; a < 7; a++)
                for (int b = 0; b < 7; b++)
                    for (int k = 0; k <= 9; k++) {
                        if(k == 7) continue; //不可以有7
                        auto &v1 = f[i][j][a][b], v2 = f[i - 1][k][mod(a - j * mod(p, 7), 7)][mod(b - j, 7)];
                        //一下都在套状态转移方程
                        v1.s0 = mod(v1.s0 + v2.s0, P);
                        v1.s1 = mod(v1.s1 + j * (p % P) % P * v2.s0 % P + v2.s1, P);
                        v1.s2 = mod(v1.s2 + v2.s0 * j % P * (p % P) * j % P * (p % P) % P
                                    + 2 * (j * (p % P)) % P * v2.s1 % P 
                                    + v2.s2, P);
                    }
        }
    power7[0] = power9[0] = 1; //这里算10^i % 7的余数和10^i % 1e9的余数
    for (int i = 1; i < N; i++) {
        power7[i] = power7[i - 1] * 10 % 7;
        power9[i] = power9[i - 1] * 10 % P;
    }
}

//求出i位，且最高位是j，且本身模7不等于a，且各位数之和模7不等b的集合
//因为预处理的f[i][j][a][b]是本身模7等于a，且各位数之和模7等于b的集合，因此需要两趟for循环来实现
node get(int i, int j, int a, int b) {
    int s0 = 0, s1 = 0, s2 = 0;
    for (int x = 0; x < 7; x++)
        for (int y = 0; y < 7; y++) {
            if (x == a || y == b) continue;
            auto v = f[i][j][x][y];
            s0 = mod(s0 + v.s0, P);
            s1 = mod(s1 + v.s1, P);
            s2 = mod(s2 + v.s2, P);
        }
    return {s0 , s1 , s2};
}

int dp(int n) {
    if(!n) return 0;

    int backup_n = n % P; //备份一个n
    vector<int> num;
    while (n) num.push_back(n % 10), n /= 10;

    int res = 0, last_a = 0, last_b = 0; //last_a表示前缀本身的值，last_b表示前缀各位数之和
    for (int i = num.size() - 1; i >= 0; i--) {
        int x = num[i];
        for (int j = 0; j < x; j++) {
            if(j == 7) continue;
            int a = mod(-last_a * power7[i + 1], 7) , b = mod(-last_b, 7); //位下文做准备
            auto v = get(i + 1, j, a, b); //求得本身模7不等于a，并且各位数之和模7不等b的集合，此时就可以使用预处理出来的结构体
            
            //根据公式求s2，时刻提醒记得及时取模！！！！！！
            //这里的last_a其实也可以作为j，毕竟它也是首位数
            res = mod(res + 
                    (last_a % P) * (last_a % P) % P * (power9[i + 1] % P) % P * (power9[i + 1] % P) % P * v.s0 % P + 
                    2 * (last_a % P) * (power9[i + 1] % P) % P * v.s1 % P + 
                    v.s2,
                P);
        }
        
        if (x == 7) break; //不满足条件
        last_a = last_a * 10 + x; //前缀和
        last_b += x; //前缀各个数位的和
        
        //记得特判n本身
        if (!i && last_a % 7 && last_b % 7) res = mod(res + backup_n * backup_n, P); //全都选了也是一种答案
    }

    return res;
}

void solve() {
    cin >> L >> R;
    cout << mod(dp(R) - dp(L - 1), P) << endl;
}

signed main() {
    init();
    cin >> T;
    while(T--) solve();
    return 0;
}
```

# 单调队列优化DP

## [最大子序和](https://www.acwing.com/problem/content/description/137/)

看到求一段**连续区间**的和的问题，会想到用**前缀和**进行优化，然后就是**枚举**区间的问题。

暴力枚举**子区间**是一种方法，但没有优化空间；因此不妨枚举**子区间的右端点**。

- **状态**：$f_i$ 为以 $i$ 为右端点，长度不超过 $m$ 的区间的最大和。
- **状态转移方程**：这里，我们设 $s_i$ 为 $1\sim i$ 的和，也就是 $i$ 的前缀和。那么状态转移方程就容易推出了：

$$
f_i=\max\{s_i-s_j\}(1\le i-j\le m)
$$

- **初始化**：抱歉，本题没有初始化。
- **答案**：$\max\{f_i\}$。

注意到**状态转移方程**可以把其中的常量提取出来，即：
$$
f_i=s_i-\min\{s_j\}(1\le i-j\le m)
$$
然后就是从前往后维护一个长度为 $m$ 的区间的最小值了，这就变成了我们熟悉的单调队列维护区间最值了。

```cpp
#include <bits/stdc++.h>
#define maxn 300010
#define int long long
using namespace std;

int n, m, ans = -LONG_LONG_MAX, h = 0, t = 0, a[maxn], q[maxn], s[maxn], f[maxn];
signed main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) s[i] = s[i - 1] + a[i]; //前缀和
    for (int i = 1; i <= n; i++) {
        int x = s[i];
        while(h <= t && i - q[h] > m) h++; //区间长度超了，把队头扔掉
        f[i] = x - s[q[h]]; //状态转移方程
        while(h <= t && x <= s[q[t]]) t--; //队尾太大了，扔掉
        q[++t] = i; //放入队尾
    }
    for (int i = 1; i <= n; i++) ans = max(ans, f[i]); //答案
    cout << ans << endl;
    return 0;
}
```

## [修剪草坪](https://www.acwing.com/problem/content/description/1089/)

考虑用**动态规划**来求解本问题。

由于**连续选择**超过 $m$ 个元素时，这些元素的**贡献**为 $0$（相当于没选）。

而本题，所有的元素值都是**正整数**，故我们的方案中，连续选择的元素数量一定是**不超过** $m$ 的。

可以用**反证法**证明，如果方案中有超过 $m$ 个连续元素，则我们不选中间的一个，使他断开，必然不会使方案变差。

于是，我们就可以通过最后一次**没有选择**的元素，对状态进行划分。

- **状态**：$f_i$ 表示以 $i$ 为右端点的**前缀数组**的方案中**收益**最大的**价值**。
- **状态转移方程**：我们设 $s_i$ 为 $1\sim i$ 的和，也就是 $i$ 的前缀和。那么状态转移方程就容易推出了：

$$
f_i=\max\{f_{j-1}+s_i-s_{j}\}(0\le i-j\le m)
$$

- **初始化**：抱歉，本题没有初始化。
- **答案**：$f_n​$。

跟上一题一样，首先我们把**常量**提出来：$f_i=s_i+\max\{f_{j-1}-s_j\}(0\le i-j\le m)$。

由于 $j$ 是有范围的: $i - m \le j \le i$，于是问题就转化为**滑动窗口求极值**的问题了。

我们用一个记录 $f_{j-1} - s_j$ 的**单调递减**的**队列**在**队头**维护一个**最大值**即可。

```cpp
#include <bits/stdc++.h>
#define maxn 300010
#define int long long
using namespace std;

int n, m, ans = 0, h = 0, t = 0, a[maxn], q[maxn], s[maxn], f[maxn];
int g(int x) {
    return f[max(0ll, x - 1)] - s[x]; //状态转移方程
}

signed main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) s[i] = s[i - 1] + a[i];
    for (int i = 1; i <= n; i++) {
        while(h <= t && i - q[h] > m) h++; //离开窗口
        f[i] = max(s[i] + g(q[h]), f[i - 1]); //注意这里要取最大值
        while(h <= t && g(i) >= g(q[t])) t--; //太小了，退下
        q[++t] = i; //放入队列
    }
    cout << f[n] << endl;
    return 0;
}
```

## [旅行问题](https://www.acwing.com/problem/content/description/1090/)

看到**环形问题**，就会想到在先前**环形区间DP**中提到的一个解决方法：**拉环成链**。

复制一份数组接在**原数组**后面，这样原问题就**等价于**在**新数组**中存在长度为 $n+1$ 的**合法区间**。

然后考虑区间何时**合法**的问题，题中提及：

1. 会先在一个点**获得油量**。
2. 然后**消耗油量**到达下一个点。
3. 消耗后的油量**不能小于 $0$**。

这极像一个我们熟悉的**括号序列模型** — 任意**前缀区间**必然满足左括号数 $\ge​$ 右括号数。

故想到用**前缀和思想**来分析：该**区间**如果**合法**，则必然**任意前缀**获得油量 $\ge$ 消耗油量。

> 这里我们要**明确**每个数组下标的实际含义，才能调对本题的**边界**，不然会 `WA` 疯。

---

- 先讨论一下顺时针怎么写？

1. $ d_i $: 点 $ i $ → 点 $ i + 1 $ 消耗的油量。
2. $ p_i $: 到点 $ i $ 能获得的油量。

相对应的前缀和含义：

$ s_i $：从 $1$ 出发到 $ i $ 所获得的总油量**减去**从 $1$ 出发到 $ i + 1 $ 所消耗的总油量。

因为我们要到 $ i + 1 $ 点，需要满足**此前的剩余油量** $+$ **该点获得的油量** $\ge$ **到达下点消耗的油量**。

> 闫氏 DP 分析法，启动！
>
> **状态表示-集合 $ f_i $**：以 $ i $ 为起点和终点，顺时针走一圈的方案。
>
> **状态表示-属性 $ f_i $**：方案是否可行 `true`。
>
> **状态计算**：想要从 $i​$ 点走到 $j​$ 点，必须满足 $i\implies j\text{ 能获得的油量 }+\text{ 此前的剩余油量 }\ge i\implies j\text{ 需要耗费的油量}​$，换成数学公式就需要满足 $\sum_{b=1}^j p_b-\sum_{a+1}^{i-1} p_a\ge \sum_{b=1}^j d_b-\sum_{a+1}^{i-1} d_a​$，即 $\sum_{b=1}^j p_b-\sum_{a+1}^{i-1} p_a-(\sum_{b=1}^j d_b-\sum_{a+1}^{i-1} d_a)\ge 0​$。换成我们的前缀和就是 $s_{j}-s_{i-1}\ge 0​$。
>
> 那么就可以推出状态转移方程啦！
>
> $ f_i = \left\lfloor \min \left\{ s_j-s_{i-1} \right\} \geq 0 \right\rfloor \quad (i \leq j \leq i + n - 1) $
>
> 和往常一样，提出常量:
> $ f_i = \left\lfloor \min\{s_j \}-s_{i-1} \geq 0 \right\rfloor \quad (i \leq j \leq i + n - 1) $
>
> 这样就变成一个滑动窗口求最小值的问题了，直接套单调队列即可。

- 而逆时针只要反过来思考一下就可以了，即换成后缀和。

```cpp
#include <bits/stdc++.h>
#define maxn 2000010
#define int long long
using namespace std;

int n, p[maxn >> 1], d[maxn >> 1], s[maxn], q[maxn], h[maxn];
bool f[maxn];
signed main() {
    cin >> n;
    for (int i = 1; i <= n; i++) scanf("%d%d", &p[i], &d[i]);
    for (int i = 1; i <= n; i++) s[i] = s[i + n] = p[i] - d[i];
    for (int i = 1; i <= 2 * n; i++) s[i] += s[i - 1];
    int hh = 0, tt = -1;
    for (int i = 2 * n; i >= 1; i--) {
        if(hh <= tt && q[hh] >= i + n) hh++; //超出[i, i + n - 1]
        while(hh <= tt && s[q[tt]] >= s[i]) tt--;
        q[++tt] = i;
        if(i <= n && s[q[hh]] >= s[i - 1]) f[i] = 1; //顺时针
    }
    d[0] = d[n];
    for (int i = n; i >= 1; i--) h[i] = h[i + n] = p[i] - d[i - 1];
    for (int i = 2 * n; i >= 1; i--) h[i] += h[i + 1];
    hh = 0, tt = -1;
    for (int i = 1; i <= 2 * n; i++) {
        if(hh <= tt && q[hh] <= i - n) hh++; //超出[i - n + 1, i]
        while(hh <= tt && h[q[tt]] >= h[i]) tt--;
        q[++tt] = i;
        if(i > n && h[q[hh]] >= h[i + 1]) f[i - n] = 1; //顺时针
    }
    for (int i = 1; i <= n; i++)
        if(f[i]) printf("TAK\n");
        else printf("NIE\n");
    return 0;
}
```

## [烽火传递](https://www.acwing.com/problem/content/description/1091/)

求一维数组选择方案，不妨用动态规划将问题拆分成一个个子问题来求解，最后合并。

状态表示-集合 $ f_i $：以 $ i $ 为 右端点的前缀区间，选择第 $ i $ 个元素的方案。

状态表示-属性 $ f_i $：方案选的元素总贡献最小 $\min$。

状态计算：
$ f_i = \min\{f_j + w_i\} \quad (0 \leq i - j - 1 \leq m - 1) $

$ \Rightarrow f_i = w_i + \min\{f_j\} \quad (1 \leq i - j \leq m) $

有限范围的 $ j $ 和分离开常量后的最小值函数，直接套单调队列模板即可。

---

然后是关于如何统计答案。

y 总演示的是在最后，暴力枚举最终状态 $ f[i] $，其中 $ n - m + 1 \leq i \leq n $-(y总常数有点大)-。

这里可以巧妙利用我们的滑动窗口。

首先我们知道，单调队列维护的是 $ i - m \leq j \leq i - 1 $ 的区间的 $ f_j $ 状态最小值。

循环迭代完第 $ n $ 轮后，$ j $ 的范围为 $ n - m \leq j \leq n $。

$ f[n] $ 进入滑动窗口，但是我们的代码中，判断滑动窗口的大小要在 $ n + 1 $ 轮里进行。

所以结束的时候，把滑动窗口往后多划一位，把 $ j $ 的范围定在 $ n - m + 1 \leq j \leq n $。

则单调队列的队头元素，就是该范围内的值最小的最终状态，输出队头元素即可。

```cpp
#include <bits/stdc++.h>
#define maxn 200010
#define int long long
using namespace std;

int n, m, ans, h = 0, t = 0, a[maxn], q[maxn], f[maxn];
signed main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) {
        while(h <= t && i - q[h] > m) h++;
        f[i] = a[i] + f[q[h]];
        while(h <= t && f[i] <= f[q[t]]) t--;
        q[++t] = i;
    }
    if(n - q[h] >= m) h++;
    cout << f[q[h]] << endl;
    return 0;
}
```

## [绿色通道](https://www.acwing.com/problem/content/1092/)

直接做不是很好做，不妨把问题转化为我们熟悉的模型来求解。

显然，答案是存在**单调性**的：

1. 任意比答案小的间距的选择方案，其元素总和必然超过 $𝑚$。
2. 任意比答案大或相等的间距，必然存在一个方案，使得元素总和小于等于 $m$。

对于 $2$ 是显然的，我们可以在原合法方案上，删去一些数，从而实现**间距**变大的操作。

对于 $1$，我们可以用 反证法：若小于**答案**的**间距**存在符合条件的选元素方案，则我们的**答案**应该是该**间距**，这与原**答案**矛盾。

找出该**单调性**，我们就可以上**二分**了。

现问题就转化成了：在确定**最小间距**情况下，能否找出选择**元素总和**小于等于 $m$ 的方案。

该问题等价于： 在确定**最小间距**情况下，选择**元素总和**最小的方案**价值**是否小于等于 $m​$。

这样本题就变成：[烽火传递](https://www.acwing.com/problem/content/description/1091/)。

```cpp
#include <bits/stdc++.h>
#define maxn 50010
using namespace std;

int n, T, a[maxn], q[maxn], f[maxn];
bool check(int m) {
    int h = 0, t = 0;
    memset(q, 0, sizeof(q));
    for (int i = 1; i <= n; i++) {
        while(h <= t && i - q[h] > m) h++;
        f[i] = a[i] + f[q[h]];
        while(h <= t && f[i] <= f[q[t]]) t--;
        q[++t] = i;
    }
    if(n - q[h] >= m) h++;
    return f[q[h]] <= T;
}

signed main() {
    cin >> n >> T;
    for (int i = 1; i <= n; i++) cin >> a[i];
    int l = 0, r = n, ans = 0;
    while(l <= r) {
        int mid = (l + r) >> 1;
        if(check(mid + 1)) ans = mid, r = mid - 1;
        else l = mid + 1;
    }
    cout << ans << endl;
    return 0;
}
```

## [理想的正方形](https://www.acwing.com/problem/content/1093/)

首先考虑把每一行中长度为 $n$ 的区间的最大值用用单调队列处理出来，存入一个新的数组。其次在新的数组里把每一列中长度为 $n$ 的区间的最大值用单调队列处理出来，最小值也是如此。

```cpp
#include <bits/stdc++.h>
#define maxn 1010
using namespace std;

int A, B, n, ans = 1e9;
int a[maxn][maxn], row_max[maxn][maxn], col_max[maxn][maxn], row_min[maxn][maxn], col_min[maxn][maxn], q[maxn];
signed main() {
    cin >> A >> B >> n;
    for (int i = 1; i <= A; i++)
        for (int j = 1; j <= B; j++)
            cin >> a[i][j];
   	//处理每一行
    for (int i = 1; i <= A; i++) {
        int hh = 0, tt = 0;
        memset(q, 0, sizeof(q)); //注意每次都要清空（否则会死的比较惨）
        for (int j = 1; j <= B; j++) {
            while(hh <= tt && j - q[hh] >= n) hh++;
            while(hh <= tt && a[i][j] >= a[i][q[tt]]) tt--;
            q[++tt] = j;
            if(j >= n) row_max[i][j - n + 1] = a[i][q[hh]];
        }
    }
    //处理新数组的每一列
    for (int i = 1; i <= B - n + 1; i++) {
        int hh = 0, tt = 0;
        memset(q, 0, sizeof(q));
        for (int j = 1; j <= A; j++) {
            while(hh <= tt && j - q[hh] >= n) hh++;
            while(hh <= tt && row_max[j][i] >= row_max[q[tt]][i]) tt--;
            q[++tt] = j;
            if(j >= n) col_max[j - n + 1][i] = row_max[q[hh]][i];
        }
    }
    for (int i = 1; i <= A; i++) {
        int hh = 0, tt = 0;
        memset(q, 0, sizeof(q));
        for (int j = 1; j <= B; j++) {
            while(hh <= tt && j - q[hh] >= n) hh++;
            while(hh <= tt && a[i][j] <= a[i][q[tt]]) tt--;
            q[++tt] = j;
            if(j >= n) row_min[i][j - n + 1] = a[i][q[hh]];
        }
    }
    for (int i = 1; i <= B - n + 1; i++) {
        int hh = 0, tt = 0;
        memset(q, 0, sizeof(q));
        for (int j = 1; j <= A; j++) {
            while(hh <= tt && j - q[hh] >= n) hh++;
            while(hh <= tt && row_min[j][i] <= row_min[q[tt]][i]) tt--;
            q[++tt] = j;
            if(j >= n) col_min[j - n + 1][i] = row_min[q[hh]][i];
        }
    }
    for (int i = 1; i <= A - n + 1; i++)
        for (int j = 1; j <= B - n + 1; j++)
            ans = min(col_max[i][j] - col_min[i][j], ans); //求答案
    cout << ans << endl;
    return 0;
}
```
