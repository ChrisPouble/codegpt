# Flood Fill

## 池塘计数

一年前在三鑫就做过这道题。

所以说是不可能不会的好吧。

```cpp
#include <bits/stdc++.h>
#define maxn 1010
using namespace std;

int n, m, ans;
bool vis[maxn][maxn];
char c[maxn][maxn];
struct node {
	int x, y;
};

void bfs(int sx, int sy) {
	queue<node>q;
	q.push({sx, sy});
	while(!q.empty()) {
		int x = q.front().x;
		int y = q.front().y;
		q.pop();
		for (int i = x - 1; i <= x + 1; i++)
			for (int j = y - 1; j <= y + 1; j++)
				if(i >= 1 && i <= n && j >= 1 && j <= m && !vis[i][j] && c[i][j] == 'W') {
					vis[i][j] = 1;
					q.push({i, j});
				}
	}
}

signed main() {
	cin >> n >> m;
	for (int i = 1; i <= n; i++)
		for (int j = 1; j <= m; j++)
			cin >> c[i][j];
	for (int i = 1; i <= n; i++)
		for (int j = 1; j <= m; j++)
			if (!vis[i][j] && c[i][j] == 'W') bfs(i, j), ans++;
	cout << ans << endl; 
	return 0;
}
```

## 城堡问题

与上一题相同，只是多了一个二进制处理，不是很难。

```cpp
#include <bits/stdc++.h>
#define maxn 1010
using namespace std;

int n, m, ans1, ans2;
//南、东、北、西 
int dx[5] = {0, 1, 0, -1, 0};
int dy[5] = {0, 0, 1, 0, -1};
bool vis[maxn][maxn];
int a[maxn][maxn];
struct node {
	int x, y;
};

void bfs(int sx, int sy) {
	queue<node>q;
	q.push({sx, sy}), vis[sx][sy] = 1;
	int res = 0;
	while(!q.empty()) {
		int x = q.front().x;
		int y = q.front().y;
		res++;
		q.pop();
		for (int i = 1; i <= 4; i++) {
			int tx = x + dx[i], ty = y + dy[i];
			if(tx < 1 || tx > n || ty < 1 || ty > m || vis[tx][ty]) continue;
			int k = a[tx][ty], v[5];
			memset(v, 0, sizeof(v));
			for (int j = 3; j >= 0; j--)
				if (k >= (1 << j)) v[j + 1] = 1, k -= (1 << j);
			if(i == 1 && !v[2]) {
				vis[tx][ty] = 1;
				q.push({tx, ty});
			}
			if(i == 2 && !v[1]) {
				vis[tx][ty] = 1;
				q.push({tx, ty});
			}
			if(i == 3 && !v[4]) {
				vis[tx][ty] = 1;
				q.push({tx, ty});
			}
			if(i == 4 && !v[3]) {
				vis[tx][ty] = 1;
				q.push({tx, ty});
			}
		}
	}
	ans2 = max(ans2, res);
}

signed main() {
	cin >> n >> m;
	for (int i = 1; i <= n; i++)
		for (int j = 1; j <= m; j++)
			cin >> a[i][j];
	for (int i = 1; i <= n; i++)
		for (int j = 1; j <= m; j++)
			if (!vis[i][j]) bfs(i, j), ans1++;
	cout << ans1 << endl;
	cout << ans2 << endl;
	return 0;
}
```

## 山峰和山谷

同样的 `bfs`，但是需要记录遍历的联通块附近是否有高于或低于自己的区域。

```cpp
#include <bits/stdc++.h>
#define maxn 1010
using namespace std;

int n, peak, valley, a[maxn][maxn];
int dx[9] = {0, 0, 1, 0, -1, 1, 1, -1, -1};
int dy[9] = {0, 1, 0, -1, 0, 1, -1, 1, -1};
bool vis[maxn][maxn];
struct node {
	int x, y;
};

void bfs(int sx, int sy, int &has_highter, int &has_lower) {
	queue<node>q;
	q.push({sx, sy});
	vis[sx][sy] = 1;
	while(!q.empty()) {
		int x = q.front().x;
		int y = q.front().y;
		q.pop();
		for (int i = 1; i <= 8; i++) {
			int tx = x + dx[i], ty = y + dy[i];
			if(tx < 1 || tx > n || ty < 1 || ty > n) continue;
			if(a[tx][ty] != a[x][y]) {
				if (a[tx][ty] > a[x][y]) has_highter = 1; // 存在比自己高的
				else has_lower = 1; // 存在比自己低的
            } else if(!vis[tx][ty]) vis[tx][ty] = 1, q.push({tx, ty});
		}
	}
}

signed main() {
	cin >> n;
	for (int i = 1; i <= n; i++)
		for (int j = 1; j <= n; j++)
			cin >> a[i][j];
	for (int i = 1; i <= n; i++)
		for (int j = 1; j <= n; j++)
			if (!vis[i][j]) {
				int sx = i, sy = j, is_highter = 0, is_lower = 0;
				bfs(sx, sy, is_highter, is_lower);
				if(!is_highter) peak++; // 没有比自己高的
				if(!is_lower) valley++; // 没有比自己低的
			}
	cout << peak << " " << valley << endl;
	return 0;
}
```

# 最短路模型

## 迷宫问题

本题不能直接存储起点到每个点的路径，但是可以储存每个店前面的那个店，这样就可以做到路径存储的效果。

```cpp
#include <bits/stdc++.h>
#define x first
#define y second
using namespace std;

typedef pair<int ,int> PII;
const int N = 1010;
PII pre[N][N];//存储当前位置的前驱
bool st[N][N];//表示当前位置的状态，若已用则为true,反之为false.
int n, c[N][N];
int dx[4] = {-1, 1, 0, 0};
int dy[4] = {0, 0, -1, 1};
void bfs() {
    queue <PII> q;
    q.push({n, n});
    st[n][n] = 1;
    while(q.size()){
        auto t = q.front();
        q.pop();
        for (int i = 0; i < 4; i++) {
            int x = t.x + dx[i], y = t.y + dy[i];
            if(x >= 1 && x <= n && y >= 1 && y <= n && c[x][y] == 0 && !st[x][y]) {
                st[x][y] = 1;
                q.push({x, y});
                pre[x][y] = t; //存储一下当前合理位置的前驱,即该位置由哪个位置引入
            }
        }
    }

    //打印方案 
    int x = 1, y = 1;
    while (x != n || y != n) { //当x = y = n时,循环截止,因为下标 n 为第一个元素，它没有前驱
        cout << x - 1 << ' ' << y - 1 << endl;
        auto t = pre[x][y];
        x = t.x, y = t.y;
    }
    cout << n - 1 << ' ' << n - 1 << endl;//单独打印第一个元素的下标

}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++)
            cin >> c[i][j];
    bfs();
    return 0;
}
```

## 武士风度的牛

超水题目，随便做。

```cpp
#include <bits/stdc++.h>
#define maxn 200
using namespace std;

int n, m, sx, sy, ex, ey;
int dx[8] = {1, 1, -1, -1, 2, 2, -2, -2};
int dy[8] = {2, -2, 2, -2, 1, -1, 1, -1};
char a[maxn][maxn];
bool vis[maxn][maxn];
struct node {
    int x, y, pts;
};

void bfs() {
    queue<node>q;
    q.push({sx, sy, 0});
    vis[sx][sy] = 1;
    while (!q.empty()) {
        int x = q.front().x;
        int y = q.front().y;
        int pts = q.front().pts;
        q.pop();
        if (x == ex && y == ey) {
            cout << pts << endl;
            exit(0);
        }
        for (int i = 0; i < 8; i++) {
            int tx = x + dx[i], ty = y + dy[i];
            if (tx < 1 || tx > n || ty < 1 || ty > m || vis[tx][ty] || a[tx][ty] == '*') continue;
            q.push({tx, ty, pts + 1});
            vis[tx][ty] = 1;
        }
    }
}

signed main() {
    cin >> m >> n;
    for (int i = 1; i <= n; i++) {
        string s;
        cin >> s;
        for (int j = 1; j <= m; j++) {
            a[i][j] = s[j - 1];
            if (a[i][j] == 'K') sx = i, sy = j;
            if (a[i][j] == 'H') ex = i, ey = j;
        }
    }
    bfs();
	return 0;
}
```

## 抓住那头牛

同样简单，但是需要剪枝。

- 如果使用完第二种操作的 $X<0$，那么可以不考虑。因为这种情况下 $X+1$ 相当于走了没有意义的一步，而 $2\times X$ 则会使牛的位置更小，更加远离目标。
- 如果使用完第三种操作的 $x>2\times m$，那么同样可以不考虑。因为这种情况下想要回到 $m$ 肯定会比原来更难。
- 如果 $n>m$，那么只能一点一点往后走，即答案为 $n-m$。

```cpp
#include <bits/stdc++.h>
#define maxn 200
using namespace std;

int n, m;
map<int, int>vis;
struct node {
    int x, pts;
};

void bfs() {
    queue<node>q;
    q.push({n, 0});
    vis[n] = 1;
    while (!q.empty()) {
        int x = q.front().x;
        int pts = q.front().pts;
        q.pop();
        if (x == m) {
            cout << pts << endl;
            exit(0);
        }
        int tx1 = x + 1;
        int tx2 = x - 1;
        int tx3 = 2 * x;
        if (!vis[tx1]) q.push({tx1, pts + 1}), vis[tx1] = 1;
        if (tx2 >= 0 && !vis[tx2]) q.push({tx2, pts + 1}), vis[tx2] = 1;
        if (tx3 <= 2 * m && !vis[tx3]) q.push({tx3, pts + 1}), vis[tx3] = 1;
    }
}

signed main() {
	cin >> n >> m;
	if (n >= m) cout << n - m << endl;
	else bfs();
	return 0;
}
```

# 多源BFS

## 矩阵距离

本题说简单点就是让矩阵中的 $0$ 去找离自己最近的 $1$。

但是会有些麻烦，我们可以逆着思考：让 $1$ 去找 $0$。

```cpp
#include <bits/stdc++.h>
#define maxn 1010
using namespace std;

int n, m;
char c[maxn][maxn];
int dis[maxn][maxn];
int dx[4] = {-1, 1, 0, 0};
int dy[4] = {0, 0, -1, 1};
struct node {
	int x, y, pts;
};

void bfs() {
    memset(dis, -1, sizeof(dis)); 
    queue<node>q;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            if (c[i][j] == '1') {
                q.push({i, j, 0});
                dis[i][j] = 0;
            }
    while (!q.empty()) {
        int x = q.front().x;
        int y = q.front().y;
        int pts = q.front().pts;
		q.pop();
        for (int i = 0; i < 4; i++) {
            int tx = x + dx[i], ty = y + dy[i];
            if (tx < 1 || tx > n || ty < 1 || ty > m) continue;
            if (dis[tx][ty] == -1) {
                dis[tx][ty] = pts + 1;
                q.push({tx, ty, pts + 1});
            }
        }
    }
}

signed main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++)
        cin >> c[i] + 1;
    bfs();
    for (int i = 1; i <= n; i++, puts(""))
        for (int j = 1; j <= m; j++)
            cout << dis[i][j] << " ";
    return 0;
}
```

# 最小步数模型

## 魔板

没啥好说的。但是大模拟。

```cpp
#include <bits/stdc++.h>
#define maxn 1010
using namespace std;

int n, m;
stack<char> ans;
char g[3][5];
unordered_map<string, pair<char, string> >pre;
unordered_map<string, int> dis;
void st(string state) {
    for (int i = 0; i < 4; i++) g[1][i] = state[i]; // 将状态的前四个字符赋值给g的第一行
    for (int i = 7, j = 0; j < 4; i--, j++) g[2][j] = state[i]; // 将状态的后四个字符赋值给g的第二行
}

string get() {
	string res;
    for (int i = 0; i < 4; i++) res += g[1][i]; // 将g的第一行字符拼接成字符串
    for (int i = 3; i >= 0; i--) res += g[2][i]; // 将g的第二行字符逆序拼接成字符串
    return res;
}

string move0(string state) {
    st(state);
    for (int i = 0; i < 4; i++) swap(g[1][i], g[2][i]); // 交换g的第一行和第二行的字符
    return get();
}

string move1(string state) {
    st(state);
    int v0 = g[1][3], v1 = g[2][3]; // 保存g的第一行和第二行的最后一个字符
    for (int i = 3; i > 0; i--) {
        g[1][i] = g[1][i - 1]; // 将g的第一行字符向右移动一位
        g[2][i] = g[2][i - 1]; // 将g的第二行字符向右移动一位
    }
    g[1][0] = v0, g[2][0] = v1; // 将保存的字符放到g的第一行和第二行的第一个位置
    return get();
}

string move2(string state) {
    st(state);
    int v = g[1][1]; // 保存g的第一行的第二个字符
    g[1][1] = g[2][1]; // 将g的第二行的第二个字符赋值给g的第一行的第二个字符
    g[2][1] = g[2][2]; // 将g的第二行的第三个字符赋值给g的第二行的第二个字符
    g[2][2] = g[1][2]; // 将g的第一行的第三个字符赋值给g的第二行的第三个字符
    g[1][2] = v; // 将保存的字符放到g的第一行的第三个位置
    return get();
}

int bfs(string st, string en) {
	if (st == en) return 0;
    queue<string>q;
    q.push(st);
    dis[st] = 0;
    while (!q.empty()) {
        string t = q.front();
		q.pop();
		string m[3];
		m[0] = move0(t);
        m[1] = move1(t);
        m[2] = move2(t);
		for (int i = 0; i < 3; i++) {
            if (!dis.count(m[i])) {
            	dis[m[i]] = dis[t] + 1;
            	pre[m[i]] = {'A' + i, t};
            	q.push(m[i]);
            	if (m[i] == en) return dis[m[i]];
			}
        }
    }
    return -1;
}

signed main() {
	string st, en;
	for (int i = 0, x; i < 8; i++) {
		cin >> x;
		en += char(x + '0');
	}
	for (int i = 1; i <= 8; i++) st += char('0' + i);
	cout << bfs(st, en) << endl;
	while (en != st) {
		ans.push(pre[en].first);
		en = pre[en].second;
	}
	while (!ans.empty()) {
		cout << ans.top();
		ans.pop();
	}
    return 0;
}
```

# 双端队列广搜

## 电路维修

双端队列广搜可以使更加优秀的状态先被搜索到，很大程度上降低了代码的运行次数。

---

这道题需要用到两个偏移量：

- 第一个偏移量:

  > ```cpp
  > int moved[4][2] = {
  > 	{-1, -1},
  > 	{-1, 1},
  > 	{1, 1},
  > 	{1, -1},
  > };
  > ```
  >
  > ![](https://cdn.luogu.com.cn/upload/image_hosting/i7i9cpf8.png)
  >
  > 题目要求其实是求从 $(0,0)$ 走到 $(n,m)$ 的最短次数而不是从左上第一个方格走到右下最后一个方格.
  >
  > 现在解释第一个偏移量：如图，第一个偏移量记录的其实是图中标红的点，由于题目要求只能斜着走不能横竖着走，所以每次移动都以某个红点为中心向四周 $1,2,3,4$ 位置移动。

- 第二个偏移量:

  > ```cpp
  > int movei[4][2] = {
  > 	{-1, -1},
  > 	{-1, 0},
  > 	{0, 0},	
  > 	{0, -1},
  > };
  > ```
  >
  > ![](https://cdn.luogu.com.cn/upload/image_hosting/udjpfep4.png)
  >
  > 设图中中心点的坐标为 $(x,y)$。
  >
  > 第二个偏移量关注的才是每一个方格。
  >
  > 如图，判断从 $1$ 号点到中心点经过了哪个方格 $\to$ 即：根据图中中心点的坐标更新从 $1$ 号点到中心点经过的方格的坐标 $(x-1,y-1)$。
  >
  > 判断从 $2$ 号点到中心点经过了哪个方格 $\to$ 即：根据图中中心点的坐标更新从 $2$ 号点到中心点经过的方格的坐标 $(x-1,y)$。
  >
  > 判断从 $3$ 号点到中心点经过了哪个方格 $\to$ 即：根据图中中心点的坐标更新从 $3$ 号点到中心点经过的方格的坐标 $(x,y)$。
  >
  > 判断从 $4$ 号点到中心点经过了哪个方格 $\to$ 即：根据图中中心点的坐标更新从 $4$ 号点到中心点经过的方格的坐标 $(x,y-1)$。
  >
  > 这个偏移量就可以帮助我们判断是否应该去旋转。

除了偏移量，还需要一个数组：

```cpp
char cp[] = "\\/\\/";
```

这个数组可以与第二偏移量相结合，帮助判断是否应该旋转。

---

回到最本质的，如果一个方格需要旋转，则把对应的节点放在队伍后面。否则，放在最前面。

```cpp
#include <bits/stdc++.h>
using namespace std;

typedef pair<int, int > PII;

int n, m, T;
const int N = 510;
char g[N][N];
int d[N][N];
int moved[4][2] = {
	{-1, -1},
	{-1, 1},
	{1, 1},
	{1, -1},
};

int movei[4][2] = {
	{-1, -1},
	{-1, 0},
	{0, 0},	
	{0, -1},
};

inline int bfs() {
	memset(d, 0x3f, sizeof d);

	deque<PII> dq;
	dq.push_front({0, 0});
	d[0][0] = 0;

	char cp[] = "\\/\\/";

	while (dq.size()) {
		auto t = dq.front();
		dq.pop_front();

		int x = t.first, y = t.second;
		for (int i = 0; i < 4; i ++) {
			int a = x + moved[i][0], b = y + moved[i][1];
			if (x >= 0 && x <= n && y >= 0 && y <= m) {
				int j = x + movei[i][0], k = y + movei[i][1];
				int w;
				if (g[j][k] != cp[i]) w = 1;
				else w = 0;

				if (d[a][b] > d[x][y] + w) {
					d[a][b] = d[x][y] + w;
					if (w) dq.push_back({a, b});
					else dq.push_front({a, b});
				}
			}
		}
	}

	if (d[n][m] == 0x3f3f3f3f) return -1;
	else return d[n][m];
}

inline void solve() {
	cin >> n >> m;
	for(int i = 0; i < n; i++) cin >> g[i];

	int res = bfs();

	if(res == -1) puts("NO SOLUTION");
	else cout << res << endl;
}

int main() {
	cin >> T;
	while (T--) solve();
	return 0;
}
```

# 双向广搜

## 字串变换

这个算法与双端队列广搜不同。本算法需要两个队列，一个从起点广搜，另一个从终点广搜，如果搜到了同一个，则代表找到了想要的答案。

---

设每次决策数量是 $k$，如果采用普通 `BFS`，那么最坏情况下的搜索空间是 $k^{10}$。

如果采用双向 `BFS`，由于从起点和终点最多会分别扩展 $5$ 步。那么搜索空间可降至 $2k^5$。剪枝效果极好。

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 6;

int n;
string A, B;
string a[N], b[N];
int extend(queue<string>& q, unordered_map<string, int>&da, unordered_map<string, int>& db,
           string a[N], string b[N]) {
	int d = da[q.front()];
	while (q.size() && da[q.front()] == d) {
		auto t = q.front();
		q.pop();

		for (int i = 0; i < n; i++)
			for (int j = 0; j < t.size(); j++)
				if (t.substr(j, a[i].size()) == a[i]) {
					string r = t.substr(0, j) + b[i] + t.substr(j + a[i].size());
					if (db.count(r)) return da[t] + db[r] + 1;
					if (da.count(r)) continue;
					da[r] = da[t] + 1;
					q.push(r);
				}
	}

	return 11;
}

int bfs() {
	if (A == B) return 0;
	queue<string> qa, qb;
	unordered_map<string, int> da, db;

	qa.push(A), qb.push(B);
	da[A] = db[B] = 0;

	int step = 0;
	while (qa.size() && qb.size()) {
		int t;
		if (qa.size() < qb.size()) t = extend(qa, da, db, a, b);
		else t = extend(qb, db, da, b, a);

		if (t <= 10) return t;
		if (++step == 10) return -1;
	}

	return -1;
}

int main() {
	cin >> A >> B;
	while (cin >> a[n] >> b[n]) n++;

	int t = bfs();
	if (t == -1) puts("NO ANSWER!");
	else cout << t << endl;

	return 0;
```

# A*

## 第K短路

$\textup{A*}$ 算法定义了一个估价函数 $f_x = g_x + h_x$，其中：
- $g_x$ 为从初始状态到达当前状态的实际代价；
- $h_x$ 为从当前状态到达目标状态的最佳路径的估计代价。

每次取出 $f_x$ 最优的状态 $x$，扩展其所有子状态，并用优先队列来维护这些状态。

- 求解 $k$ 短路问题

> 在求解 $k$ 短路问题时，令 $h_x$ 为从当前结点到达终点 $t$ 的最短路径长度。可以通过在反向图上对点 $t$ 跑单源最短路预处理出每个结点的这个值。
> 
> 状态表示为 $(x, g_x)$，其中 $x$ 为当前到达的结点，$g_x$ 为已经走过的距离。

算法流程：

1. 将初始状态 $(s, 0)$ 加入优先队列。
2. 每次取出 $f_x = g_x + h_x$ 最小的状态，枚举该状态到达的结点 $x$ 的所有出边。
3. 将对应的子状态加入优先队列。
4. 当访问到一个结点第 $k$ 次时，对应的状态的 $g_x$ 即为从 $s$ 到该结点的第 $k$ 短路。

优化：已取出的状态到达一个结点的次数大于 $k$ 次时，可不扩展其子状态，因为前 $k$ 次已形成 $k$ 条合法路径。

该算法最坏时间复杂度为 $O(nk \log n)$，但可在相同复杂度内求出从起始点 $s$ 到每个结点的前 $k$ 短路。


```cpp
#include <bits/stdc++.h>
#define x first
#define y second
using namespace std;

typedef pair<int, int> PII;
typedef pair<int, PII> PIII;

const int N = 1010, M = 200010;

int n, m, S, T, K;
int dis[N], cnt[N];
bool st[N];
vector<PII> linker[N], linkai[N];
void add(int a, int b, int c) {
    linker[a].push_back({b, c});
    linkai[b].push_back({a, c});
}

void dijkstra() { // 从终点开始跑 dijkstra
    priority_queue<PII, vector<PII>, greater<PII> > heap;
    heap.push({0, T});
    memset(dis, 0x3f, sizeof(dis));
    dis[T] = 0;
    
    while (heap.size()) {
        auto t = heap.top();
        heap.pop();
        
        int u = t.y;
        if (st[u]) continue;
        st[u] = 1;
        
        for (auto i:linkai[u]) {
            int v = i.x, w = i.y;
            if (dis[v] > dis[u] + w) {
                dis[v] = dis[u] + w;
                heap.push({dis[v], v});
            }
        }
    }
}

int Astar() {
    priority_queue<PIII, vector<PIII>, greater<PIII> > heap;
    heap.push({dis[S], {0, S}});
    
    while (heap.size()) {
        auto t = heap.top();
        heap.pop();
        
        int u = t.y.y, dist = t.y.x;
        cnt[u]++;
        
        if (cnt[T] == K) return dist;
        
        for (auto i:linker[u]) {
            int v = i.x, w = i.y;
            if (cnt[v] < K) heap.push({dist + w + dis[v], {dist + w, v}});
            // 注意不能在这里判断，因为不能保证是否是 K 短路
        }
    }
    
    return -1;
}

int main() {
    cin >> n >> m;
    
    for (int i = 1, a, b, c; i <= m; i++) {
        cin >> a >> b >> c;
        add(a, b, c);
    }
    
    cin >> S >> T >> K;
    if (S == T) K++;
    
    dijkstra();
    cout << Astar() << endl;
    
    return 0;
}
```

## 八数码

由于 $\textup{A*}$ 算法在无解情况下的运行次数与普通 BFS 一样，因此我们先要判断是否有解。

经过简单观察，可以发现终点状态的逆序对数量为 $0$。也就是说，我们需要通过交换 'x' 与其相邻的数字，以达到最终状态的逆序对数量为 $0$。

而交换操作可看做两种：

- 横向交换，如：

  > ```
  > 1 2 3
  > 4 5 6
  > x 7 8
  > ```
  >
  > 在交换 'x' 和 '7' 之后，逆序对的数量并无变化。
  >
  > ```
  > 1 2 3
  > 4 5 6
  > 7 x 8
  > ```

- 竖向交换，如：

  > ```
  > 1 2 3
  > 4 5 6
  > 7 x 8
  > ```
  >
  > 在交换 'x' 和 '5' 之后，逆序对的数量增多了两个。
  >
  > ```
  > 1 2 3
  > 4 x 6
  > 7 5 8
  > ```
  >
  > 当然，也可能造成逆序对减少两个的情况，在这并不列出。

综上，我们可以知道每次交换只能增加或减少偶数个逆序对的数量，也可得出结论：如果起点状态逆序对为奇数，则无解。

---

那么应该如何跑 $\textup{A*}$ 算法呢？

其实就很简单了，启发式函数就是求曼哈顿距离。之后再用个 map 记录一下路线就可以了，别的就是普通 `BFS` 了。

```cpp
#include <bits/stdc++.h>
#define x first
#define y second
using namespace std;

// 定义pair类型，用于存储状态和对应的启发式估计值
typedef pair<int, string> PIS;

string start, x;
unordered_map<string, int> d; // 存储每个状态到起始状态的距离
priority_queue<PIS, vector<PIS>, greater<PIS> > heap; // 小根堆，用于存储待扩展的状态
unordered_map<string, pair<string, char> > last; // 存储状态转移信息

// 启发式函数，计算当前状态到目标状态的曼哈顿距离
int f(string m) {
    int res = 0;
    for (int i = 0; i < 9; i++) {
        if (m[i] != 'x') {
            int t = m[i] - '1'; // 将字符转换为对应的数字（0-8）
            res += abs(i / 3 - t / 3) + abs(i % 3 - t % 3); // 计算曼哈顿距离
        }
    }
    return res;
}

// A*算法实现，返回从初始状态到目标状态的最短路径对应的操作序列
string bfs(string start) {
    string end = "12345678x"; // 目标状态

    heap.push({f(start), start}); // 将起始状态加入堆中
    d[start] = 0; // 起始状态到自身的距离为0

    // 定义操作数组和坐标变化数组
    char oper[] = "udlr";
    int dx[4] = {-1, 1, 0, 0}, dy[4] = {0, 0, -1, 1};

    while (!heap.empty()) {
        int st = heap.top().x; // 取出堆顶元素
        string state = heap.top().y;
        heap.pop();
        
        if (state == end) break; // 如果达到目标状态，则退出循环

        int x, y; // 空格（'x'）的位置
        for (int i = 0; i < 9; i++) {
            if (state[i] == 'x') {
                x = i / 3, y = i % 3;
                break;
            }
        }

        string init = state; // 保存当前状态，用于恢复
        for (int i = 0; i < 4; i++) { // 尝试四个方向的移动
            int a = x + dx[i], b = y + dy[i];
            if (a < 0 || a >= 3 || b < 0 || b >= 3) continue; // 如果移动越界，则跳过
            swap(state[a * 3 + b], state[x * 3 + y]); // 执行移动操作
            if (!d.count(state) || d[state] > d[init] + 1) { // 如果新状态未被访问过，或者通过当前路径到达新状态的距离更短
                d[state] = d[init] + 1; // 更新距离
                heap.push({f(state) + d[state], state}); // 将新状态加入堆中，优先级为其启发式估计值与实际距离之和
                last[state] = {init, oper[i]}; // 记录状态转移信息
            }
            state = init; // 恢复状态，以便尝试下一个方向的移动
        }
    }

    // 构建操作序列
    string ans;
    while (end != start) {
        ans += last[end].y; // 添加操作
        end = last[end].x; // 回溯到前一个状态
    }
    reverse(ans.begin(), ans.end()); // 反转操作序列，得到最终答案
    return ans;
}

int main() {
    char c;
    while (cin >> c) { // 输入初始状态，空格分隔数字或'x'
        start += c;
        if (c != 'x') x += c;
    }

    // 检查初始状态是否可解（逆序对的数量应为偶数）
    int res = 0;
    for (int i = 0; i < 8; i++) {
        for (int j = i + 1; j < 8; j++) {
            if (x[i] > x[j]) res++;
        }
    }

    if (res % 2) { // 如果逆序对数量为奇数，则输出"unsolvable"
        printf("unsolvable\n");
    } else { // 否则，输出从初始状态到目标状态的最短路径对应的操作序列
        cout << bfs(start) << endl;
    }

    return 0;
}
```

## 分成互质组

每次遍历的数，枚举将这个数放在哪一个组，之后继续枚举即可。

```cpp
#include <bits/stdc++.h>
#define maxn 15
using namespace std;

bool vis[maxn];
int n, ans = maxn, a[maxn];
vector<int> f[maxn];
int gcd(int x, int y) {
    if (!y) return x;
    return gcd(y, x % y);
}

void dfs(int sum, int cnt) { // 已经存了多少个元素、已经存了多少个组
    if (sum == n) {
        ans = min(ans, cnt);
        return ;
    }
    bool is_ = 0;
    for (int i = 1; i <= n; i++) {
        int u = a[i], is = 0;
        if (vis[i]) continue;
        for (auto v:f[cnt])
            if (gcd(u, v) != 1)
                is = 1;
        if (!is) {
            is_ = 1;
            f[cnt].push_back(u), vis[i] = 1;
            dfs(sum + 1, cnt);
            f[cnt].pop_back(), vis[i] = 0;
        }
    }
    if (!is_) dfs(sum, cnt + 1);
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    dfs(0, 1);
    cout << ans << endl;
    return 0;
}
```

# DFS之剪枝与优化

剪枝的几种方法：

1. 优化搜索顺序。大多数情况下，我们应该优先搜索分支较少的节点，因为这样就可以更快地遇到剪枝。
2. 排除等效冗余：例如排列组合时的相同排列
3. 可行性剪枝：例如加上一只小猫之后超出重量，那就不考虑这只小猫
4. 最优化剪枝：如果当前搜索到的结果已经超过当前最优答案了，则退出

## 小猫爬山

本题可以考虑剪枝方法：

1. 优化搜索顺序 $\color\green ✔$

- 就是换个搜索顺序，但效率不同，也就是优先搜枝条少的搜索子树
  这里我们可以优先搜索质量重的猫，所以我们可以把所有的猫按重量从大到小排序
2. 排除等效冗余 $\color\red✖$

- 因为整棵搜索树都会搜到，所以不能排除等效冗余

3. 可行性剪枝 $\color\green ✔$

- 如果当前位置放了一个猫就会超重，那么就不要放

4. 最优化剪枝 $\color\green ✔$

- 如果当前答案不优于最优解，那么就不要搜

```cpp
#include <bits/stdc++.h>
#define maxn 20
using namespace std;

bool vis[maxn];
int n, w, ans = maxn, a[maxn];
int f[maxn];
bool cmp(int x, int y) { return x > y; }
void dfs(int sum, int cnt) { // 已经存了多少个元素、已经存了多少个组
    if (ans < cnt) return ;
    if (sum == n + 1) {
        ans = cnt;
        return ;
    }
    for (int i = 1; i <= cnt; i++)
        if (f[i] + a[sum] <= w) {
            f[i] += a[sum];
            dfs(sum + 1, cnt);
            f[i] -= a[sum];
        }
    f[cnt + 1] = a[sum];
    dfs(sum + 1, cnt + 1);
    f[cnt + 1] = 0;
}

int main() {
    cin >> n >> w;
    for (int i = 1; i <= n; i++) cin >> a[i];
    sort(a + 1, a + n + 1, cmp);
    dfs(1, 1);
    cout << ans << endl;
    return 0;
}
```

## 数独

本题可以考虑剪枝方法：

1. 优化搜索顺序 $\color\green ✔$

- 很明显，我们肯定是从当前能填合法数字最少的位置开始填数字

2. 排除等效冗余 $\color\red✖$

- 因为整棵搜索树都会搜到，所以不能排除等效冗余

3. 可行性剪枝 $\color\green ✔$

- 这一步可以放在枚举能放的位置上进行优化
- 由于每一行每一列每一个宫格都不能重复，所以我们可以用位运算优化枚举数的过程

4. 最优化剪枝 $\color\red✖$

- 本题要求找到一个唯一的方案，因此不能使用

还有一个优化方法：

- 二进制优化：由于搜索中将会用到很多次 `check()`，这就要求 `check()` 的高效性了。可以使用二进制快速进行判断，减少常数。

```cpp
#include <bits/stdc++.h>
#define lowbit(x) ((x) & (-x))
using namespace std;

const int N = 9, M = 1 << N;

//ones表示0-2^9里每个数有多少个1，map快速地找出这行哪一列可以填，比如map[(10)2] = 1就知道第二列可以填1
int ones[M], mp[M];
//分别表示行，列，大方格子有哪些数字没有填
int rows[N], cols[N], cell[3][3];
char s[100];

//is_set = true则在x, y填上t, 否则则把x,y处的数字删掉, t 是0-8
void draw(int x, int y, int t, int is_set) {
    if (is_set)  s[x * N + y] = '1' + t; // t 是0-8
    else s[x * N + y] = '.';

    int v = 1 << t;
    if (!is_set) v = -v;

    //如果某位没被放，则它的二进制位应该是1， 所以应该减去v
    //如果放了，它的二进制应该是0，则经过上面的取反，负负得正，-v实际上就是把二进制0变为1
    rows[x] -= v;
    cols[y] -= v;
    cell[x / 3][y / 3] -= v;
}

//x行y列可以填哪个数字, 最后得到2^i + 2^j..+..，这些i, j就是可以填的数字，最后通过map[2^i]来得到这个数字
int get(int x, int y) {
    return rows[x] & cols[y] & cell[x / 3][y / 3];
}

int init() {
    //cnt表示还剩多少个数字没有填
    int cnt = 0;
    //初始状态state的9位二进制全是1
    int state = (1 << N) - 1;

    //如果row[0] = 111111111 代表第一行可以填9个数, 这里行号是0-8
    for (int i = 0; i < 9; i++) rows[i] = state;
    for (int i = 0; i < 9; i++) cols[i] = state;
    for (int i = 0; i < 3; i++)
        for (int j = 0; j < 3; j++)
            cell[i][j] = state;

    for (int i = 0, k = 0; i < N; i++)
        for (int j = 0; j < N; j++, k++)
            if (s[k] != '.') draw(i, j, s[k] - '1', 1);
            else cnt++;

    return cnt;
}

bool dfs(int cnt) {
    //填完所有数字，则返回
    if (!cnt) return 1;  
    //最多可以填多少个数字
    int minv = 10; 
    int x, y;
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            if (s[i * N + j] == '.') {
                //可以填的数字状态，如010001，是1则表示可以填
                int state = get(i, j);     
                 //选一个1的个数最少的，这样的分支数量最少
                if (ones[state] < minv) {
                    minv = ones[state];
                    x = i, y = j;
                } 
            }
        }
    }

    //依次做 lowbit 操作，选择每个分支
    for (int i = get(x, y); i ; i -= lowbit(i)) {

        // 这个t就是要填充的数字
        int t = mp[lowbit(i)];

        // 填这个数字
        draw(x, y, t, 1);    

        // 这次填充成功，则返回
        if (dfs(cnt - 1)) return 1;   

        // 失败则回溯
        draw(x, y, t, 0);
    }

    return 0;
}

int main() {
    //打表，快速地知道可以哪一个数字
    for (int i = 0; i < N; i++) mp[1 << i] = i;

    //ones记录每个状态有多少个1，用于选择分支少的开始搜索, 其中M = 1 << N
    for (int i = 0; i < M; i++)
        for (int j = i; j; j -= lowbit(j))
            ones[i] += 1;

    while (cin >> s, s[0] != 'e') {
        int k = init();
        dfs(k);
        puts(s);
    }

    return 0;
}
```

## 木棒

本题在 $1\sim 50$ 中枚举最小答案长度，且因为答案的倍数一定是所有木棍总和，因此我们只枚举木棍总和的约数。

1. 优化搜索顺序 $\color\green ✔$

- 很明显，我们肯定是从大到小枚举木棍。

2. 排除等效冗余 $\color\green ✔$

   > 1. 按照组合的顺序枚举木棍
   > 2. 如果当前这根木棍失败了，则跳过所有相同的木棍
   > 3. 如果第一次尝试拼入木棍就失败的话，那么这个分治必然也是失败的，因为在拼入这些木棍前，面对的原始木棍都是还没有拼接的，他们都是等效的。
   > 4. 如果最后一次尝试拼入木棍就失败的话，那么分治必然也是失败的。因为如果存在把后面的木棍搬过来可以拼成一根木棒的方案，那么为什么不把刚刚失败的木棍拼在后面呢？

3. 可行性剪枝 $\color\green ✔$

- 如果当前枚举到的木棍加上正在拼凑的木棒超过枚举的最小长度，则跳出

4. 最优化剪枝 $\color\red✖$

- 因为本题要求只要找到一个答案就可以了，因此不需要使用

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 70;

int n;
int length, sum; // length 表示枚举的木棒的长度，sum 表示当前已经拼好的木棒的总长度
int w[N]; // 每根木棍的长度
bool st[N]; // 状态数组

// u：当前正在拼哪根木棒，s：当前木棒中已有木棍的总长度，start：从哪个位置开始枚举木棍
bool dfs(int u, int s, int start) {
    if (u * length == sum) return true; // 如果当前木棒的数量 * 长度 = sum，则找到了答案，直接返回 true
    if (s == length) return dfs(u + 1, 0, 0); // 如果当前木棒的长度等于了 length，则新开一根木棒

    // 否则枚举从 start 开始枚举将哪根木棍拼到当前木棒中
    for (int i = start; i < n; i ++ ) { // 剪枝 3-1：确定枚举顺序，以组合的形式枚举方案
        if (st[i]) continue; // 如果当前木棍用过了，则跳过
        if (s + w[i] <= length) { // 如果当前木棒拼上该木棍没超过 length
            st[i] = true;
            if (dfs(u, s + w[i], i + 1)) return true; // 继续搜索当前木棒的下一根木棍拼哪根
            st[i] = false;

            // 执行到这 dfs(u, s + w[i], i + 1) 为 false: 说明当前木棍搜索失败了

            // 剪枝 3-3：如果是木棒的第一根木棍就搜索失败了，则一定搜不到方案
            if (!s) return false; // s = 0 表示刚才木棒拼的是第一根木棍

            // 剪枝 3-4：如果是木棒的最后一根木棍搜索失败了，也一定搜不到方案
            if (s + w[i] == length) return false;

            // 剪枝 3-2：如果当前木棍没有搜到方案，则跳过所有长度相等的木棍
            int j = i;
            while (j < n && w[i] == w[j]) j ++ ;
            i = j - 1;
        }
    }

    return false;
}

bool cmp(int x, int y) { return x > y; }

int main() {
    while (cin >> n, n) {
        // 多组测试数据清空
        memset(st, 0, sizeof st);
        sum = 0;

        for (int i = 0; i < n; i ++ ) {
            cin >> w[i];
            sum += w[i];
        }

        // 剪枝 2：优化搜索顺序，木棍长度从大到小排序
        sort(w, w + n, cmp);

        length = 1; // 每根木棒的长度从 1 开始枚举，从小到大枚举，题目要求分组最多，那么第一个满足条件的就是答案
        while (true) { // 循环一定会退出，最坏情况下所有木棍拼一根木棒
            if (sum % length == 0 && dfs(0, 0, 0)) { // 剪枝 1：只有 length 是 sum 的约数才有可能凑出多个等长的木棒
                cout << length << endl;
                break;
            }
            length ++ ; // 当前 length 无法得到答案，则长度 + 1
        }
    }

    return 0;
}
```
