# DFS

## [排列数字](https://www.acwing.com/problem/content/844/)

定义一个数组 $ans$，负责存储答案；定义一个数组 $vis$，$vis_i=1$ 表示 $i$ 已经被选过了；搜索边界是 $ans$ 的数量为 $n+1$ 就输出；搜索过程是枚举 $1\sim n$，没选过就选，并用 $vis$ 和 $ans$ 记录，回溯时再清空 $vis$ 和 $ans$ 就行了。

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, vis[10], ans[10]; 
void dfs(int x) { //负责当ans下标
	if(x == n + 1) { //满了
		for (int i = 1; i <= n; i++) cout << ans[i] << " "; //输出
		cout << endl;
		return ;
	}
	for (int i = 1; i <= n; i++)
		if(!vis[i]) vis[i] = 1, ans[x] = i, dfs(x + 1), vis[i] = 0; //记录、回溯
}

signed main() {
	cin >> n;
	dfs(1);
	return 0;
}
```

## [n-皇后问题](https://www.acwing.com/problem/content/845/)

定义三个数组：$vis1_i$ 表示第 $i$ 列上有无皇后，$vis2_i$ 表示行号 $+$ 列号 $=i$ 的对角线上有无皇后，$vis3_i$ 表示列号 $-$ 行号 $+n$（$n$ 是偏移量）的对角线上有无皇后，之后再用 $ans$ 数组记录答案就可以了。直到 $ans$ 数组数量为 $n+1$ 就输出。

```cpp 
#include <bits/stdc++.h>
using namespace std;

int n, ans[10]; 
bool vis1[30], vis2[30], vis3[30]; //i、x + i、i - x + n 
void dfs(int x) {
	if(x == n + 1) {
		for (int i = 1; i <= n; i++) {
			for (int j = 1; j < ans[i]; j++) cout << ".";
			cout << "Q";
			for (int k = ans[i] + 1; k <= n; k++) cout << ".";
			cout << endl;
		}
		cout << endl;
		return;
	}
	for (int i = 1; i <= n; i++) {
		if(!vis1[i] && !vis2[i + x] && !vis3[i - x + n]) {
			vis1[i] = 1, vis2[i + x] = 1, vis3[i - x + n] = 1, ans[x] = i;
			dfs(x + 1);
			vis1[i] = 0, vis2[i + x] = 0, vis3[i - x + n] = 0;
		}
	}
}

signed main() {
	cin >> n;
	dfs(1);
	return 0;
}
```

# BFS

## [走迷宫](https://www.acwing.com/problem/content/846/)

不停地向四个方向试探，如果越界、已访问过或者脚底是 $1$，则不考虑；否则，放入队列，记录访问。

```cpp
#include <bits/stdc++.h>
using namespace std;

struct node {
	int x, y, pre;
};

int n, m, a[125][125];
int dx[4] = {0, -1, 0, 1};
int dy[4] = {-1, 0, 1, 0};
bool vis[125][125];
queue<node>q;
int bfs(int x, int y) {
    q.push({x, y, 0});
    vis[x][y] = 1;
    while(!q.empty()) {
        int x = q.front().x;
        int y = q.front().y;
        int pre = q.front().pre;
        q.pop();
        if(x == n && y == m) return pre;
        for (int i = 0; i < 4; i++) {
            int tx = x + dx[i], ty = y + dy[i];
            if(tx < 1 || tx > n || ty < 1 || ty > m || vis[tx][ty] || a[tx][ty]) continue;
            q.push({tx, ty, pre + 1});
            vis[tx][ty] = 1;
        }
    }
}

signed main() {
	scanf("%d%d", &n, &m);
	for (int i = 1; i <= n; i++)
		for (int j = 1; j <= m; j++)
			scanf("%d", &a[i][j]);
	cout << bfs(1, 1) << endl;
    return 0;
}
```

# 树与图的搜索

## [树的重心](https://www.acwing.com/problem/content/848/)

首先使用`DFS`算出每棵子树的大小。接着再来一次`DFS`，设这个点为 $x$，则 $x$ 上面那一块的大小就是 $n-siz_x$，剩下的分裂出来的子树大小为 $siz_{son}$，取这些块的最大值。再取这些最大值的最小值就是重心了。

```cpp
#include <bits/stdc++.h>
#define maxn 100010
using namespace std;

int n, ans = 1e9, fa[maxn], siz[maxn];
vector<int> linker[maxn], answer;
void dfs(int x, int fa) {
    siz[x] = 1;
    for (auto i : linker[x])
        if (i != fa) dfs(i, x), siz[x] += siz[i];
}

void dep1(int x, int fa) {
    int res = n - siz[x];
    for (auto i : linker[x])
        if (i != fa) res = max(siz[i], res);
    ans = min(ans, res);
    for (auto i : linker[x])
        if (i != fa) dep1(i, x);
}

int main() {
    cin >> n;
    for (int i = 1, x, y; i < n; i++) cin >> x >> y, linker[x].push_back(y), linker[y].push_back(x);
    dfs(1, 0);
    dep1(1, 0);
    cout << ans << endl;
    return 0;
}
```

## [图中点的层次](https://www.acwing.com/problem/content/849/)

使用`BFS`，如果搜索到了 $n$ 就输出步数，一直没有找到就输出 $-1$。

```cpp
#include <bits/stdc++.h>
#define maxn 100010
using namespace std;

struct node {
    int x, pts;
};

int n, m, vis[maxn];
vector<int>linker[maxn];
int main() {
    cin >> n >> m;
    for (int i = 1, x, y; i <= m; i++) cin >> x >> y, linker[x].push_back(y);

        int pts = q.front().pts;

        if(x == n) {
            cout << pts << endl;
            return 0;
        }
        for (auto v:linker[x]) {
            if(vis[v]) continue;
            q.push({v, pts + 1});
            vis[v] = 1;
        }
    }
    cout << -1 << endl;
    return 0;
}
```

# 拓扑排序

## [有向图的拓扑序列](https://www.acwing.com/problem/content/850/)

先把所有入度为 $0$ 的点放入队列，之后开始松弛每一个点，直到一个点的入度为 $0$，就放入答案队列里，如果答案队列的长度没有达到 $n$，说明肯定有点没有进来，输出 $-1$；否则，输出答案队列。

```cpp
#include <bits/stdc++.h>
#define maxn 100010
using namespace std;

int n, m, inv[maxn];
vector<int>linker[maxn];
void bfs() {
    queue<int>q, ans;
    for (int i = 1; i <= n; i++)
        if(!inv[i]) q.push(i); //入读为0
    while(!q.empty()) {
        int x = q.front();
        q.pop();
        ans.push(x); //放入答案队列
        for (auto v:linker[x]) {
            inv[v]--;
            if(inv[v] == 0) q.push(v); //入读为0
        }
    }
    if(ans.size() != n) { //答案队列长度不是n
        cout << -1 << endl;
        return ;
    }
    while(!ans.empty()) {
        cout << ans.front() << " ";
        ans.pop();
    }
}

int main() {
    scanf("%d%d", &n, &m);
    for (int i = 1, x, y, w; i <= m; i++) scanf("%d%d", &x, &y), linker[x].push_back(y), inv[y]++;
	bfs();
	return 0;
}
```

# Dijkstra

## [Dijkstra求最短路 I](https://www.acwing.com/problem/content/851/)

`Dijkstra` 是一种贪心算法，这里介绍的是 $O(n^2)$ 的朴素 `Dijkstra`。

1. 用一个 $dis$ 数组保存源点到其余各个节点的距离，$dis_i$ 表示源点到节点 $i$ 的距离。初始时，$dis$ 数组的各个元素为无穷大。

   用一个状态数组 $vis$ 记录是否找到了源点到该节点的最短距离，$state_i$ 如果为真，则表示找到了源点到节点 $i$ 的最短距离，$state_i$ 如果为假，则表示源点到节点 i 的最短距离还没有找到。初始时，$state$ 各个元素为假。

![](http://dyoi.net:2024/file/873/%E9%A2%98%E8%A7%A3%E9%9C%80%E8%A6%81%E5%9B%BE%E7%89%8711.png)

2. 源点到源点的距离为 $0$。即 $dist_1 = 0$。

![](http://dyoi.net:2024/file/873/%E9%A2%98%E8%A7%A3%E9%9C%80%E8%A6%81%E5%9B%BE%E7%89%8712.png)

3. 遍历 $dist$ 数组，找到一个节点，这个节点是：没有确定最短路径的节点中距离源点最近的点。假设该节点编号为 $i$。此时就找到了源点到该节点的最短距离，$state_i$ 置为 $1$。

![](http://dyoi.net:2024/file/873/%E9%A2%98%E8%A7%A3%E9%9C%80%E8%A6%81%E5%9B%BE%E7%89%8714.png)

4. 遍历 $i$ 所有可以到达的节点 $j$，如果 $dist_j$ 大于 $dist_i$ 加上 $i->j$ 的距离，即 $dist_j>dist_i+w_{i\ j}$（$w_{i\ j}$ 为 $i -> j$ 的距离），则更新 $dist_j=dist_i+w_{i\ j}$。

![](http://dyoi.net:2024/file/873/%E9%A2%98%E8%A7%A3%E9%9C%80%E8%A6%81%E5%9B%BE%E7%89%8715.png)

5. 重复 $3,4$ 步骤，直到所有节点的状态都被置为 $1$。

![](http://dyoi.net:2024/file/873/%E9%A2%98%E8%A7%A3%E9%9C%80%E8%A6%81%E5%9B%BE%E7%89%870.png)

6. 此时 $dist$ 数组中，就保存了源点到其余各个节点的最短距离。

![](http://dyoi.net:2024/file/873/%E9%A2%98%E8%A7%A3%E9%9C%80%E8%A6%81%E5%9B%BE%E7%89%8716.png)

```cpp
#include <bits/stdc++.h>
#define maxn 100010
using namespace std;

struct edge {
    int to, w;
};

int n, m, dis[maxn];
bool vis[maxn];
vector<edge>linker[maxn]; //邻接表 
void add(int x, int y, int w) {
    linker[x].push_back({y, w}); //邻接表建边 
}

void dijkstra(int st) {
    memset(dis, 0x3f, sizeof(dis));
    dis[st] = 0;
    for (int i = 1; i <= n; i++) {
        int x = n + 1;
        for (int j = 1; j <= n; j++)
            if(!vis[j] && dis[j] < dis[x]) x = j; //找出可以用来更新，且离起点最短的点 
        vis[x] = 1; //以后不能更新了
        for (auto k:linker[x]) {
            int v = k.to, w = k.w;
            dis[v] = min(dis[v], dis[x] + w); //更新 
        }
    }
}

signed main() {
    cin >> n >> m;
    for (int i = 1, x, y, w; i <= m; i++) {
        cin >> x >> y >> w, add(x, y, w);
    }
    dijkstra(1);
    cout << (dis[n] == 0x3f3f3f3f ? -1 : dis[n]) << endl;
    return 0;
}
```

## [Dijkstra求最短路 II ](https://www.acwing.com/problem/content/852/2)

这里介绍的是时间复杂度为 $O(m\log n)$ 的堆优化版 `Dijkstra`。

其实就是使用堆来加速了查询离起点最小的可以更新的点，其它**几乎**一样。

```cpp
#include <bits/stdc++.h>
#define maxn 150010
using namespace std;

struct edge { //存边的信息 
    int to, w; //去向和边权
};

struct node { //存点的信息 
    int dis, x; //dis值和节点
};

int n, m, dis[maxn];
bool vis[maxn];
priority_queue<node>que; //堆 
vector<edge>linker[maxn];
void add(int x, int y, int w) {
    linker[x].push_back({y, w});
}

bool operator<(node x, node y) { //堆优化，启动！ 
    return x.dis > y.dis;
}

void dijkstra(int st) {
    memset(dis, 0x3f, sizeof(dis));
    dis[st] = 0, que.push({0, st});
    while(!que.empty()) {
        int x = que.top().x;
        que.pop();
        if(vis[x]) continue; //如果已经更新过了，就没有更新的必要 
        vis[x] = 1; //标记
        for (auto k:linker[x]) {
            int v = k.to, w = k.w;
            if(dis[v] > dis[x] + w) {
                dis[v] = dis[x] + w;
                que.push({dis[v], v}); //更新 
            }
        }
    }
}

signed main() {
    cin >> n >> m;
    for (int i = 1, x, y, w; i <= m; i++) {
        cin >> x >> y >> w, add(x, y, w);
    }
    dijkstra(1);
    cout << (dis[n] == 0x3f3f3f3f ? -1 : dis[n]) << endl;
    return 0;
}
```

## 证明：Dijkstra 为什么不能处理带有负边权的图

**分析**：如图所示：

![](https://cdn.acwing.com/media/article/image/2019/11/16/7416_7cf3c78208-200fb33caef2d3ce435369aaa777f83.png)

若通过 `Dijkstra` 算法可以求出从 $1$ 号点到达 $4$ 号点所需的步数为 $3$（每次选择离源点最短距离的点更新其他点），但实际上从 $1$ 号点到达 $4$ 号点所需步数为 $1$（$1 –> 2 –> 3$），因此不能使用 `Dijkstra` 解决含负权图的问题。

**正确的分析**

`Dijkstra` 算法的 $3$ 个步骤

1. 找到当前未标识的且离源点最近的点 $t$。
2. 对 $t$ 号点点进行标识
3. 用t号点更新其他点的距离

反例：

![](https://cdn.acwing.com/media/article/image/2020/11/27/7416_181359f430-8886031fd08869a6757d4c704c6f355.png)

结果：

`Dijkstra` 算法在图中走出来的最短路径是 $1 -> 2 -> 4 -> 5$，算出 $1$ 号点到 $5$ 号点的最短距离是 $2 + 2 + 1 = 5$，然而还存在一条路径是 $1 -> 3 -> 4 -> 5$，该路径的长度是 $5 + (-2) + 1 = 4$，因此 `Dijkstra` 算法失效。

# Bellmon-Ford

## [有边数限制的最短路](https://www.acwing.com/problem/content/855/)

先来讲讲 `Bellmon-Ford` 算法：

循环 $n$ 次，每次遍历 $m$ 条边，看看是否可以更新，时间复杂度为 $O(nm)$。

```cpp
void Bellman_Ford(int st) {
    memset(dis, 0x3f, sizeof(dis));
    dis[st] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) backup[j] = dis[j]; //为了防止串联，例如dis[2]被dis[1]更新了，dis[3]又被dis[2]更新了，这样就被更新两次了
        for (int j = 1; j <= m; j++) {
            int a = edge[j].a, b = edge[j].b, w = edge[j].w;
            dis[b] = min(dis[b], backup[a] + w);
        }
    }
}
```

那么如何解决这道题呢？

我们知道，`Bellmon-Ford` 是 $n$ 次来更新的（因为就 $n$ 个点），而题目要求只能更新 $k$ 次，所以我们只需要把外层循环的 $n$ 改成 $k$ 就可以了。

```cpp
#include <bits/stdc++.h>
#define maxn 10010
using namespace std;

struct Edge { //存储一条边
    int a, b, w;
} edge[maxn];

int n, m, k, dis[maxn], backup[maxn];
void Bellman_Ford(int st) {
    memset(dis, 0x3f, sizeof(dis));
    dis[st] = 0;
    for (int i = 1; i <= k; i++) {
        for (int j = 1; j <= n; j++) backup[j] = dis[j]; //防止串联
        for (int j = 1; j <= m; j++) {
            int a = edge[j].a, b = edge[j].b, w = edge[j].w;
            dis[b] = min(dis[b], backup[a] + w); //更新
        }
    }
}

signed main() {
    cin >> n >> m >> k;
    for (int i = 1, x, y, w; i <= m; i++) cin >> x >> y >> w, edge[i] = {x, y, w};
    Bellman_Ford(1);
    if(dis[n] > 0x3f3f3f3f / 2) cout << "impossible" << endl; //因为有可能是负边权，导致<0x3f3f3f3f，所以这里是>0x3f3f3f3f / 2
    else cout << dis[n] << endl;
    return 0;
}
```

# Spfa

## [spfa求最短路](https://www.acwing.com/problem/content/853/)

考虑到 `Bellmon_Ford` 算法会遍历所有的边，但是有很多的边遍历了其实没有什么意义，我们只用遍历那些到源点距离变小的点所连接的边即可，只有当一个点的前驱结点更新了，该节点才会得到更新；因此考虑到这一点，我们将创建一个队列每一次加入距离被更新的结点。

而且，为了优化，`Spfa` 还想了个办法：只要一个点在队列里面，就不要让这个点进去了。因为这只不过是更新一下节点的值而已，队列里面的点会帮着更新的。

```cpp
#include <bits/stdc++.h>
#define maxn 150010
using namespace std;

struct edge {
    int to, w; //去向和边权
};

int n, m, dis[maxn];
bool vis[maxn]; //是否在队列里 
queue<int>que;
vector<edge>linker[maxn];
void add(int x, int y, int w) {
    linker[x].push_back({y, w});
}

void dijkstra(int st) {
    memset(dis, 0x3f, sizeof(dis));
    dis[st] = 0, vis[st] = 1, que.push(st);
    while(!que.empty()) {
        int x = que.front();
        que.pop();
        vis[x] = 0;
        for (auto k:linker[x]) {
            int v = k.to, w = k.w;
            if(dis[v] > dis[x] + w) {
                dis[v] = dis[x] + w;
                if(!vis[v]) vis[v] = 1, que.push(v);
            }
        }
    }
}

signed main() {
    cin >> n >> m;
    for (int i = 1, x, y, w; i <= m; i++) {
        cin >> x >> y >> w, add(x, y, w);
    }
    dijkstra(1);
    if(dis[n] == dis[0]) cout << "impossible" << endl; //因为Spfa不会遍历所有边，只会遍历相连的，因此只需要"=="就可以了即可 
    else cout << dis[n] << endl;
    return 0;
}
```

## [spfa判断负环](https://www.acwing.com/problem/content/854/)

求负环的常用方法，基于 `Spfa`，一般都用方法 $2$（该题也是用方法 $2$）：

- 方法 $1$：统计每个点入队的次数，如果某个点入队 $n$ 次，则说明存在负环。
- 方法 $2$：统计当前每个点的最短路中所包含的边数，如果某点的最短路所包含的边数大于等于 $n$，则也说明存在负环。

当然，为了方便理解，我们定义一个**虚拟节点**连接所有点，边权为 $0$。

1. 把**虚拟节点**放入队列里（等同于把所有点放入队列里），且 $vis_{虚拟节点}=1$（表示已经进入过队列里了），注意，这里不需要初始化 `dist` 数组（因为我们是观察 $dist$ 的趋向，全是 $0$ 也没关系）
2. 定义 $cnt_i$ 为从 $i$ 点到虚拟节点的最短路径需要经过的边的数量 $-1$（不能算上那个虚拟节点的边，所以要 $-1$），进行松弛，其中 `dis[v] = dis[x] + w, cnt[v] = cnt[x] + 1`。如果 $cnt_v \ge n$，说明存在负环，则输出。
3. 其中，为了保证 `Spfa` 的速度，因此使用 `vis` 数组（目的和用法一样）。

```cpp
#include <bits/stdc++.h>
#define maxn 150010
using namespace std;

struct edge {
    int to, w; //去向和边权
};

int n, m, dis[maxn], cnt[maxn]; //cnt表示边数
bool vis[maxn];
queue<int>que;
vector<edge>linker[maxn];
void add(int x, int y, int w) {
    linker[x].push_back({y, w});
}

void SPFA() {
    for (int i = 1; i <= n; i++) {
        vis[i] = 1;
        que.push(i);
    }
    while(!que.empty()) {
        int x = que.front();
        que.pop();
        vis[x] = 0;
        for (auto k:linker[x]) {
            int v = k.to, w = k.w;
            if(dis[v] > dis[x] + w) {
                dis[v] = dis[x] + w;
                cnt[v] = cnt[x] + 1;
                if(cnt[x] >= n) { //经过了n条边，n+1个点
                    cout << "Yes" << endl;
                    exit(0);
                }
                if(!vis[v]) que.push(v);
            }
        }
    }
}

signed main() {
    cin >> n >> m;
    for (int i = 1, x, y, w; i <= m; i++)
        cin >> x >> y >> w, add(x, y, w);
    SPFA();
    cout << "No" << endl;
    return 0;
}
```

# Floyd

## [Floyd求最短路](https://www.acwing.com/problem/content/description/856/)

其实 `Floyd` 很好理解，枚举 $k,i,j$，看看从 $i$ 到 $j$ 的距离能否被从 $i$ 到 $k$ 的距离 $+$ 从 $k$ 到 $j$ 的距离更新，即 `dis[i][j] = min(dis[i][j], dis[i][k] + dis[k][j])`。

```cpp
#include <bits/stdc++.h>
#define maxn 210
#define INF 1000000000
using namespace std;

int n, m, k, x, y, dis[maxn][maxn]; //这个邻接矩阵真麻烦，要判断重边
void Floyed() {
    for (int k = 1; k <= n; k++)
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= n; j++)
                dis[i][j] = min(dis[i][j], dis[i][k] + dis[k][j]);
}

signed main() {
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i++) 
        for (int j = 1; j <= n; j++)
            if(i == j) dis[i][j] = 0;
            else dis[i][j] = INF;
    for (int i = 1, w; i <= m; i++) cin >> x >> y >> w, dis[x][y] = min(dis[x][y], w); //判断重边
    Floyed();
    while(k--) {
        cin >> x >> y;
        if(dis[x][y] > INF / 2) cout << "impossible" << endl;
        else cout << dis[x][y] << endl;
    }
    return 0;
}
```

# Prim

## [Prim算法求最小生成树](https://www.acwing.com/problem/content/860/)

- 朴素 `Prim` 算法。

`Prim` 算法采用的是一种贪心的策略。

每次将离连通部分的最近的点和点对应的边加入的连通部分，连通部分逐渐扩大，最后将整个图连通起来，并且边长之和最小。

我们将图中各个节点用数字 $1 \sim n$ 编号。

![](https://cdn.acwing.com/media/article/image/2021/03/04/55289_508d58977c-05.png)

1. 用一个 $vis$ 数组表示节点是否已经连通。$vis_i=1$，表示已经连通，$vis_i=0$，表示还没有连通。初始时，$vis$ 各个元素为假。即所有点还没有连通。
   用一个 $dis$ 数组保存各个点到连通部分的最短距离，$dis_i$ 表示 $i$ 节点到连通部分的最短距离。初始时，$dis$ 数组的各个元素为无穷大。

![](https://cdn.luogu.com.cn/upload/image_hosting/0c78p90h.png)

2. 从 $1$ 号节点开始扩充连通的部分，所以 $1$ 号节点与连通部分的最短距离为 $0$，即 $dis_1$ 置为 $0$。

![](https://cdn.luogu.com.cn/upload/image_hosting/5uh4ullc.png)

3. 遍历 $dis$ 数组，找到一个还没有连通起来，但是距离连通部分最近的点，假设该节点的编号是 $i$。$i$ 节点就是下一个应该加入连通部分的节点，$vis_i$ 置为 1。
   用青色点表示还没有连通起来的点，红色点表示连通起来的点。
   这里青色点中距离最小的是 $dis_1$，因此 $vis_1$ 置为 $1$。

   ![](https://cdn.luogu.com.cn/upload/image_hosting/k0f5zbz6.png)

4. 遍历所有与 $i$ 相连但没有加入到连通部分的点 $j$，如果 $j$ 距离连通部分的距离大于 $i->j$ 之间的距离，即 $dis_j > w_{i\ j}$（$w_{i\ j}$ 为 $i->j$ 节点之间的距离），则更新 $dis_j$ 为 $w_{i\ j}$。
   与节点 $1$ 相连的有 $2， 3， 4$ 号节点。$1->2$ 的距离为 $100$，小于 $dis_2$，$dis_2$ 更新为 $100$。$1->4$ 的距离为 $140$，小于 $dis_4$，$dis_4$ 更新为 $140$。$1->3$ 的距离为 $150$，小于 $dis_3$，$dis_3$ 更新为 $150$。

   ![](https://cdn.luogu.com.cn/upload/image_hosting/d5e4poiw.png)

重复 $3,4$ 步骤，可以得到最后的答案。

![](https://cdn.luogu.com.cn/upload/image_hosting/gl012dka.png)

代码如下：

```cpp
#include <bits/stdc++.h>
#define maxn 100010
using namespace std;

struct edge {
    int to, w;
};

int n, m, ans, dis[maxn];
bool vis[maxn]; //判断是否在点集合里
vector<edge>linker[maxn];
void add(int x, int y, int w) {
    linker[x].push_back({y, w});
    linker[y].push_back({x, w});
}

void prim(int st) {
    memset(dis, 0x3f, sizeof(dis));
    dis[st] = 0; //dis[1] = 0
    for (int i = 1; i < n; i++) {
        int x = n + 1;
        for (int j = 1; j <= n; j++)
            if(!vis[j] && dis[j] < dis[x]) x = j; //找出最小的没有进入集合的点
        if(dis[x] == dis[n + 1]) { //找不到
            cout << "impossible" << endl;
            exit(0);
        }
        vis[x] = 1;
        for (auto k:linker[x]) {
            int v = k.to, w = k.w;
            if(!vis[v]) dis[v] = min(dis[v], w); //已经在点集合里的点不能更新
        }
    }

}

signed main() {
    cin >> n >> m;
    for (int i = 1, x, y, w; i <= m; i++) {
        cin >> x >> y >> w, add(x, y, w);
    }
    prim(1);
    for (int i = 1; i <= n; i++) ans += dis[i]; //答案
    cout << ans << endl;
    return 0;
}
```

- 堆优化版 `Prim` 算法。

其实与 `Dijkstra` 的堆优化差不多，照着写就行了。

```cpp
#include <bits/stdc++.h>
#define maxn 100010
using namespace std;

struct edge {
    int to, w;
};

struct node { //代表一个点的dis值和id 
	int dis, x;
}; 

int n, m, ans, dis[maxn];
bool vis[maxn]; //判断是否在点集合里
priority_queue<node>q;
vector<edge>linker[maxn];
bool operator<(node x, node y) { //堆优化，启动！ 
	return x.dis > y.dis;
} 

void add(int x, int y, int w) {
    linker[x].push_back({y, w});
    linker[y].push_back({x, w});
}

void prim(int st) {
    memset(dis, 0x3f, sizeof(dis));
    dis[st] = 0;
    q.push({0, st});
    while(!q.empty()) {
        int x = q.top().x;
        q.pop();
//        if(vis[x]) continue;
        vis[x] = 1; //标记已经进入点集合了 
        for (auto k:linker[x]) {
            int v = k.to, w = k.w;
            if(!vis[v] && w < dis[v]) dis[v] = w, q.push({w, v}); //已经在点集合里的点不能更新
        }
    }
}

signed main() {
    cin >> n >> m;
    for (int i = 1, x, y, w; i <= m; i++) {
        cin >> x >> y >> w, add(x, y, w);
    }
    prim(1);
    for (int i = 1; i <= n; i++)
		if(dis[i] == 0x3f3f3f3f) { cout << "impossible" << endl; return 0; }
		else ans += dis[i];
	cout << ans << endl;
    return 0;
}
```

# Kruskal

## [Kruskal算法求最小生成树](https://www.acwing.com/problem/content/861/)

- 将所有边按照权值的大小进行升序排序，然后从小到大一一判断。

- 如果这个边与之前选择的所有边不会组成回路，就选择这条边分；反之，舍去。

- 直到具有 $n$ 个顶点的连通网筛选出来 $n-1$ 条边为止。

- 筛选出来的边和所有的顶点构成此连通网的最小生成树。

- 判断是否会产生回路的方法为：使用并查集。

- 在初始状态下给各个个顶点在不同的集合中。、

- 遍历过程的每条边，判断这两个顶点的是否在一个集合中。

- 如果边上的这两个顶点在一个集合中，说明两个顶点已经连通，这条边不要。如果不在一个集合中，则要这条边。

---

举个例子，下图一个连通网，`Kruskal` 算法对应的最小生成树，需要经历以下几个步骤：

![](https://cdn.acwing.com/media/article/image/2022/03/24/55289_97bd668aab-1.gif)

1. 将连通网中的所有边按照权值大小做升序排序。

![](https://cdn.acwing.com/media/article/image/2022/03/24/55289_9f4ead1eab-2.gif)

2. 从 $B-D$ 边开始挑选，由于尚未选择任何边组成最小生成树，且 $B-D$ 自身不会构成环路，所以 $B-D$ 边可以组成最小生成树。

![](https://cdn.acwing.com/media/article/image/2022/03/24/55289_beee9e38ab-4.gif)

3. $D-T$ 边不会和已选 $B-D$ 边构成环路，可以组成最小生成树：

![](https://cdn.acwing.com/media/article/image/2022/03/24/55289_beee9e38ab-4.gif)

4. $A-C$ 边不会和已选 $B-D,D-T$ 边构成环路，可以组成最小生成树：

![](https://cdn.acwing.com/media/article/image/2022/03/24/55289_c637cb18ab-5.gif)

5. $C-D$ 边不会和已选 $A-C,B-D,D-T$ 边构成环路，可以组成最小生成树：

![](https://cdn.acwing.com/media/article/image/2022/03/24/55289_cafcd7bbab-6.gif)

6. $C-B$ 边会和已选 $C-D,B-D$ 边构成环路，因此不能组成最小生成树：

![](https://cdn.acwing.com/media/article/image/2022/03/24/55289_d5719212ab-7.gif)

7. $B-T,A-B$ 两条边都会和已选 $A-C,C-D,B-D,D-T$ 构成环路，都不能组成最小生成树。而 $S-A$ 不会和已选边构成环路，可以组成最小生成树。

![](https://cdn.acwing.com/media/article/image/2022/03/24/55289_db160ef0ab-8.gif)

8. 最终最小生成树如下：

![](https://cdn.acwing.com/media/article/image/2022/03/24/55289_e6f44f5eab-9.gif)

```cpp
#include <bits/stdc++.h>
#define maxn 200010 
using namespace std;

int n, m, ans, fa[maxn]; //并查集，启动！
struct Edge {
	int a, b, w;
} edge[maxn];

bool cmp(Edge a, Edge b) { //排序，启动！
	return a.w < b.w;
}

int find(int x) { //查询 
    if (x == fa[x]) return x;
    else {
        fa[x] = find(fa[x]);
        return fa[x];
    }
}

void merge(int x, int y) { //合并 
    int xx = find(x), yy = find(y);
    if (xx != yy) fa[xx] = yy;
}

void Kruskal() {
	int res = 0;
	for (int i = 1; i <= m; i++) {
		int u = edge[i].a, v = edge[i].b, w = edge[i].w;
		if(find(u) != find(v)) res++, ans += w, merge(u, v); //不在同一个集合，合并！
	}
	if(res != n - 1) { //没有连起来
		cout << "impossible" << endl;
		exit(0);
	}
}

signed main() {
	cin >> n >> m;
	for (int i = 1; i <= n; i++) fa[i] = i; //初始化
	for (int i = 1; i <= m; i++) {
		int x, y, w;
		cin >> x >> y >> w, edge[i] = {x, y, w};
	}
	sort(edge + 1, edge + m + 1, cmp); //排序
	Kruskal();
	cout << ans << endl;
	return 0;
}
```

# 二分图

## [染色法判定二分图](https://www.acwing.com/problem/content/862/)

二分图的定义：

- 有两顶点集且图中每条边的的两个顶点分别位于两个顶点集中，每个顶点集中没有边直接相连接！

- 用自然语言来讲，二分图就是图中点通过移动能分成左右两部分，左侧的点只和右侧的点相连，右侧的点只和左侧的点相连。
- 下图就是个二分图：

![](https://cdn.acwing.com/media/article/image/2022/03/28/55289_ac17cbdcae-%E4%BA%8C%E5%88%86%E5%9B%BE.png#=100*100)

- 而这张图却不是：

![](https://cdn.acwing.com/media/article/image/2022/03/28/55289_d09603aaae-1.png)

---

**如果判断一个图是不是二分图？**

- 开始对任意一未染色的顶点染色。
- 判断其相邻的顶点中，若未染色则将其染上和相邻顶点不同的颜色。
- 若已经染色且颜色和相邻顶点的颜色相同则说明不是二分图，若颜色不同则继续判断。

这里使用 `DFS` 搞定，当然也可以用 `BFS`。

```cpp
#include <bits/stdc++.h>
#define maxn 100010
using namespace std;

int n, m, color[maxn];
vector<int>linker[maxn];
bool dfs(int u, int c) {
	color[u] = c; //染色 
	for (auto v:linker[u]) {
		if(!color[v]) { if(dfs(v, 3 - c)) return 1; } //如果相邻的点无颜色，则染上不同的颜色 
		else if(color[v] == c) return 1; //如果颜色相同，说明不是二分图 
	}
	return 0;
}

void add(int u, int v) { //加入一条边 
	linker[u].push_back(v);
	linker[v].push_back(u);
}

signed main() {
	cin >> n >> m;
	for (int i = 1, u, v; i <= m; i++) {
		cin >> u >> v;
		add(u, v);
	}
	for (int i = 1; i <= n; i++)
		if(!color[i] && dfs(i, 1)) { //如果从这里搜过去搜到的不是二分图 
			cout << "No" << endl;
			return 0;
		}
	cout << "Yes" << endl;
	return 0;
}
```

## [二分图的最大匹配](https://www.acwing.com/problem/content/863/)

匈牙利算法是基于 Hall 定理中充分性证明的思想，它是部图匹配最常见的算法，该算法的核心就是寻找增广路径，它是一种用增广路径求二分图最大匹配的算法。

什么，看得头大，那看看下面的例子。

---

假设你是班主任 $X$ ，你现在要分配同桌，男女分配。图中的连线表示一个女生想要一个男生坐同桌，而男生只会从想要跟自己坐同桌的女生坐同桌。如下图：

![](https://cdn.luogu.com.cn/upload/image_hosting/75f4nyu3.png)

**匈牙利算法的工作模式会教你这样做**：

1. 先试着给男 $1$ 找同桌，发现第一个和他相连的女 $1$，且没有人抢走这个同桌。连上一条蓝线。

![](https://cdn.luogu.com.cn/upload/image_hosting/ax1z6te0.png)

2. 接着给男 $2$ 找同桌，发现第一个和他相连的女 $2$，连上一条蓝线。

![](https://cdn.luogu.com.cn/upload/image_hosting/hhtsrllv.png)

3. 接下来是男 $3$，很遗憾女 $1$ 已经有同桌了，怎么办呢？

我们试着给之前女 $1$ 匹配的男生（也就是男 $1$）另外分配一个同桌。

（黄色表示这条边被临时拆掉）。

![](https://cdn.luogu.com.cn/upload/image_hosting/01dbx04y.png)

与男 $1$ 相连的第二个女生是女 $2$，但是女 $2$ 也有同桌了，怎么办呢？我们再试着给女 $2$ 进行分配。重新找个妹子（注意这个步骤和上面是一样的，这是一个递归的过程）

![](https://cdn.luogu.com.cn/upload/image_hosting/7cz83ixb.png)

此时发现男 $2$ 还能找到女 $3$，那么之前的问题迎刃而解了，回溯回去。

![](https://cdn.luogu.com.cn/upload/image_hosting/r8ts94j9.png)

最后的结果就是：

![](https://cdn.luogu.com.cn/upload/image_hosting/14ev1qqu.png)

4. 接下来是男 $4$。
   很遗憾，按照第三步的节奏我们没法给男 $4$ 腾出来一个女生做同桌（别人不能给，否则自己就没有同桌了）。

```cpp
#include <bits/stdc++.h>
#define maxn 510
using namespace std;

bool vis[maxn]; //vis标记是否在递归过程中出现过，相当于临时预定 
int match[maxn];
int ans, n1, n2, m;
vector<int>linker[maxn];
void add(int a, int b) {
    linker[a].push_back(b); //还是我们最熟悉的邻接表加边
}

bool dfs(int x) {
    for (auto v:linker[x]) { //遍历每个想跟自己坐同桌的女生 
        if (!vis[v]) { //如果还没有被预定 
            vis[v] = 1; //定下这个同桌（不管能不能拿下，反正只要标记找过了就行） 
            if (!match[v] || dfs(match[v])) { //如果还没有同桌，或者说可以让对应的男生让出来 
                match[v] = x; //定下这个同桌 
                return 1; //表示自己能找到同桌 
            }
        }
    }
    return 0; //OK，可以跟垃圾桶做同桌了。 
}

int main(){
    scanf("%d%d%d", &n1, &n2, &m);
    for (int i = 1, a, b; i <= m; i++) scanf("%d%d", &a, &b), add(a, b); //加入一条关系边 
    for (int i = 1; i <= n1; i++) {
        memset(vis, 0, sizeof(vis));
        /*
		每次清空vis的原因很简单，感受一下这个样例：
		2 2 3
		1 1
		1 2
		2 1
		其中，男1与女1在第一轮匹配中很成功地成为了同桌。而男2想与女1做同桌。
		如果不清空，那么本来可以成功的男2（让男1找女2，自己找女1）就只能没有同桌了
		还有就是因为每次匹配的过程不一样 
		*/
        if(dfs(i)) ans++; //能够匹配上，ans+1 
    }
    printf("%d", ans);
    return 0;
}
```
