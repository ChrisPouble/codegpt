# 背包问题

## [01背包问题](https://www.acwing.com/problem/content/2/)

1. **状态：**$f_{i\ j}$ 表示前 $i$ 种物品背包容量为 $j$ 时能获得的最大价值

2. **状态转移方程：**

```cpp
if(j >= a[i].t) f[i][j] = max(f[i - 1][j], f[i - 1][j - a[i].t] + a[i].w); //找到最大值 
else f[i][j] = f[i - 1][j];
```

3. **初始化：**$f_{0\ 0} = 0$
4. **答案：**$f_{i\ V}$

上代码（二维数组解法）：

```cpp
#include <bits/stdc++.h>
using namespace std;

int T, m, ans, f[1010][1010];
struct node {
	int t, w;
}a[1010];

int main() {
	cin >> T >> m;
	for (int i = 1; i <= m; i++)
		cin >> a[i].t >> a[i].w;
	for (int i = 1; i <= m; i++)
		f[i][0] = 0; //初始化 
	for (int i = 1; i <= m; i++)
		for (int j = 1; j <= T; j++)
			if(j >= a[i].t) f[i][j] = max(f[i - 1][j], f[i - 1][j - a[i].t] + a[i].w); //找到最大值 
			else f[i][j] = f[i - 1][j];
	for (int i = 1; i <= m; i++)
		ans = max(ans, f[i][T]); //找答案 
	cout << ans << endl;
    return 0;
}
```

很明显，空间占用的太多了，那怎么办呢？

我们发现其实我们在更新 $f_{i\ j}$ 的时候只会用到两个值：$i和i-1$，所以我们的第一维只需要两个维度就可以了。我们称这种思路为滚动数组。

```cpp
#include <bits/stdc++.h>
using namespace std;

int T, m, ans, f[2][1010]; //f[i % 2][j]表示前i种草药时间为j时能采摘到达最大价值 
struct node {
	int t, w;
}a[1010];

int main() {
	cin >> T >> m;
	for (int i = 1; i <= m; i++)
		cin >> a[i].t >> a[i].w;
	for (int i = 1; i <= m; i++)
		for (int j = 1; j <= T; j++)
			if(j >= a[i].t) f[i % 2][j] = max(f[(i - 1) % 2][j], f[(i - 1) % 2][j - a[i].t] + a[i].w);
			else f[i % 2][j] = f[(i - 1) % 2][j];
	cout << f[m % 2][T] << endl;
    return 0;
}
```

那么我们能不能再进一步优化空间呢？答案肯定是可以的

上面讲解滚动数组的时候我们已经知道每次只会用到两个值了，所以我们可以直接把第一维省去，直接由 $i-1$ 复制到 $i$，这样就可以了

不过需要注意的一点：如果我们枚举 $j$ 的时候从前往后遍历，是会错误的，我们可以尝试一下：

很明显，我们这样子的结果错了，这是为啥呢？

其实很简单，由于每次更新 $f_j$ 时，会用到 $f_k \{1  \le k < j \}$，而这时的 $f_k$ 已经被更新过了，也就是说，我们会把一种物品选过好几遍！

那怎么办呢？其实很容易发现，从前到后不行，那就从后到前！

上代码：

```cpp
#include <bits/stdc++.h>
using namespace std;

int T, m, ans, f[1010];
struct node {
	int t, w;
}a[1010];

int main() {
	cin >> T >> m;
	for (int i = 1; i <= m; i++)
		cin >> a[i].t >> a[i].w;
	for (int i = 1; i <= m; i++)
		for (int j = T; j >= 1; j--)
			if(j >= a[i].t) f[j] = max(f[j], f[j - a[i].t] + a[i].w); //找出最大值 
	cout << f[T] << endl;
    return 0;
}
```

## [完全背包问题](https://www.acwing.com/problem/content/3/)

1. **状态：**$f_{i\ j}$ 表示前 $i$ 种物品背包容量为 $j$ 时能获得的最大价值

2. **状态转移方程：**

```cpp
f[i][j] = max(f[i][j], f[i - 1][j - v[i] * s] + s * w[i]); //其中的s是枚举一个物品装多少次的，0<=s<=j/v[i]
```

3. **初始化：**$f_{0\ 0} = 0$
4. **答案：**$\max\{f_{i\ V}\}$

那么可以写下以下代码：

```cpp
#include <bits/stdc++.h>
#define maxn 1010
using namespace std;

int n, k, v[maxn], w[maxn], f[maxn][maxn];
signed main() {
	cin >> n >> k;
	for (int i = 1; i <= n; i++) cin >> v[i] >> w[i];
	for (int i = 1; i <= n; i++)
		for (int j = 1; j <= k; j++)
			for (int s = 0; s * v[i] <= j; s++) 
				f[i][j] = max(f[i][j], f[i - 1][j - v[i] * s] + s * w[i]);
	cout << f[n][k] << endl;
	return 0;
}
```

## [多重背包问题 I](https://www.acwing.com/problem/content/4/)

与暴力完全背包同一个思路，时间复杂度 $O(nVs)$。

```cpp
#include <bits/stdc++.h>
#define maxn 1010
using namespace std;

int n, V, v[maxn], w[maxn], s[maxn], f[maxn][maxn];
signed main() {
	cin >> n >> V;
	for (int i = 1; i <= n; i++) cin >> v[i] >> w[i] >> s[i];
	for (int i = 1; i <= n; i++)
		for (int j = 1; j <= V; j++)
			for (int k = 0; k * v[i] <= j && k <= s[i]; k++) //必须保证k<=s[i]
				f[i][j] = max(f[i][j], f[i - 1][j - v[i] * k] + k * w[i]);
	cout << f[n][V] << endl;
	return 0;
}
```

## [多重背包问题 II](https://www.acwing.com/problem/content/5/)

时间复杂度承受不下的，暴力方法会超时。

我们每遇到一个物品，可以把它的 $s_i$ 分成二进制，例如 $2^0,2^1,2^2...2^k$，如果后面会有多余的，那就在后面的分段中放上 $c$，以达到分块的目的，这可以理解成一种倍增思想。

![](https://cdn.luogu.com.cn/upload/image_hosting/d7wl1xg0.png)

之后分成几个背包，每个背包的 $v,w$ 分别是 $v\times s,w\times s$，进行 $01$ 背包就可以了。我们称这种优化为二进制优化。

```cpp
#include <bits/stdc++.h>
#define maxn 2010
using namespace std;

int n, V, cnt, v[maxn * 12], w[maxn * 12], f[maxn]; //由于空间是nlog(s)，所以要乘以12
signed main() {
	cin >> n >> V;
	for (int i = 1; i <= n; i++) {
	    int a, b, c, k = 1;
	    cin >> a >> b >> c;
	    while(k <= c) { //如果还能接着分成2^k形式
	        cnt++;
	        v[cnt] = a * k; //v数组
	        w[cnt] = b * k; //w数组
	        c -= k, k *= 2; //原来的s减去k，k翻倍
	    }
	    if(c > 0) { //有剩余
	        cnt++;
	        v[cnt] = a * c; //仿照上面
	        w[cnt] = b * c;
	    }
	}
	for (int i = 1; i <= cnt; i++)
		for (int j = V; j >= v[i]; j--) //别问我下面哪个if是干嘛的，就是无聊打的
			if(j >= v[i]) f[j] = max(f[j], f[j - v[i]] + w[i]);
	cout << f[V] << endl;
	return 0;
}
```

## [数字三角形](https://www.acwing.com/problem/content/900/)

1. **状态：**$f_{i\ j}$ 表示 $(i,j)$ 到底部的最大的。

2. **状态转移方程：**

```cpp
f[i][j] = max(f[i + 1][j], f[i + 1][j + 1]) + a[i][j];
```

3. **初始化：**$f_{r\ i} = a_{r\ i}$
4. **答案：**$f_{1\ 1}$

```cpp
#include <bits/stdc++.h>
#define maxn 1010
using namespace std;

int r, a[maxn][maxn], f[maxn][maxn];
int main() {
    cin >> r;
    for (int i = 1; i <= r; i++)
        for (int j = 1; j <= i; j++)
            cin >> a[i][j];
    for (int i = 1; i <= r; i++)
        f[r][i] = a[r][i];
    for (int i = r - 1; i >= 1; i--)
        for (int j = 1; j <= i; j++)
            f[i][j] = max(f[i + 1][j], f[i + 1][j + 1]) + a[i][j];
    cout << f[1][1];
    return 0;
}
```

## [最长上升子序列](https://www.acwing.com/problem/content/897/)

1. **状态：**$f_i$ 表示以 $i$ 为结尾的最长上升子序列的长度。

2. **状态转移方程：**

```cpp
f[i] = max(f[i], f[j] + 1);
```

3. **初始化：**$f_i=1$
4. **答案：**$\max \{f_i\}$

时间复杂度 $Θ(n^2)$。

```cpp
#include <bits/stdc++.h>
#define maxn 1010
using namespace std;

int n, ans, a[maxn], f[maxn];
signed main() {
	cin >> n;
	for (int i = 1; i <= n; i++) cin >> a[i];
	for (int i = 1; i <= n; i++) {
		f[i] = 1;
		for (int j = 1; j < i; j++)
			if(a[i] > a[j]) f[i] = max(f[i], f[j] + 1);
	}
	for (int i = 1; i <= n; i++) ans = max(ans, f[i]);
	cout << ans << endl;
	return 0;
}
```

## [最长上升子序列 II](https://www.acwing.com/problem/content/898/)

这道题 $n\le 10^5$，那么我们得考虑一种 $Θ(n\log n)$ 级别的算法了。

观察我们的转移方程：`if(a[i] > a[j]) f[i] = max(f[i], f[j] + 1);`，其中的要求就是 $a_i>a_j,f_i<f_{j+1}$ 时才会更新。

考虑建立一个数组 $q$：$q_i$ 表示最长上升子序列长度为 $i$ 的子序列末尾最小的元素（长度一定，末尾元素小的肯定比大的好，这是一种贪心的思路），由于最长上升子序列中，子序列长的末尾元素一定比子序列短的末尾元素大，表示 $q$ 一定是单调递增的，这样每次利用二分找到在 $q$ 中比 $a_i$ 小的最大的一个数，之后再更新 $q$ 就行了。最后的答案就是 $q$ 的长度。

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 10;
int n, cnt, a[N], q[N];
int find(int x) { //二分查找q中小于x的下标
    int l = 1, r = cnt, res = 0;
    while (l <= r) {
        int mid = (l + r) / 2;
        if(q[mid] < x) res = mid, l = mid + 1;
        else r = mid - 1;
    }
    return res;
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) {
        int x = find(a[i]);
        q[x + 1] = a[i]; //由于是最大的小于a[i]的数，因此要+1
        if(x + 1 > cnt) cnt++; //开辟新空间
    }
    cout << cnt << endl; //q的长度
    return 0;
}
```

## [最长公共子序列 ](https://www.acwing.com/problem/content/899/)

1. **状态：**$f_{i\ j}$ 表示 $a$ 的前 $i$ 个字符组成的字符串与 $b$ 的前 $j$ 个字符组成的字符串的最长公共子序列长度。

2. **状态转移方程：**

```cpp
if(a[i] == b[j]) f[i][j] = f[i - 1][j - 1] + 1;
else f[i][j] = max(f[i - 1][j], f[i][j - 1]);
//如果a[i]==b[j]，则f[i - 1][j - 1]合并后再加上i与j的合并就是f[i - 1][j - 1] + 1
//如果a[i]!=b[j]，则从f[i][j - 1]与f[i - 1][j]选取大点的那个ctrl+c过去了
```

3. **初始化：**$f_{0\ i} = f_{i\ 0} =0$
4. **答案：**$f_{n\ m}$

```cpp
#include <bits/stdc++.h>
#define maxn 1010
using namespace std;

int n, m, f[maxn][maxn];
char a[maxn], b[maxn];
signed main() {
	cin >> n >> m >> (a + 1) >> (b + 1);
	for (int i = 1; i <= n; i++)
		for (int j = 1; j <= m; j++)
			if(a[i] == b[j]) f[i][j] = f[i - 1][j - 1] + 1;
			else f[i][j] = max(f[i - 1][j], f[i][j - 1]);
	cout << f[n][m] << endl;
	return 0;
}
```

## [最短编辑距离](https://www.acwing.com/problem/content/904/)

1. **状态：**$f_{i\ j}$ 表示 $a$ 的前 $i$ 个字符组成的字符串变成 $b$ 的前 $j$ 个字符组成的字符串需要的操作次数。

2. **状态转移方程：**

```cpp
f[i][j] = min(f[i - 1][j] + 1, f[i][j - 1] + 1); //增和删
if (a[i] == b[j]) f[i][j] = min(f[i][j], f[i - 1][j - 1]); //不改
else f[i][j] = min(f[i][j], f[i - 1][j - 1] + 1); //改
/*
1)删除操作：把a[i]删掉之后a[1~i]和b[1~j]匹配
            所以之前要先做到a[1~(i-1)]和b[1~j]匹配
            f[i-1][j] + 1
2)插入操作：插入之后a[i]与b[j]完全匹配，所以插入的就是b[j] 
            那填之前a[1~i]和b[1~(j-1)]匹配
            f[i][j-1] + 1 
3)替换操作：把a[i]改成b[j]之后想要a[1~i]与b[1~j]匹配 
            那么修改这一位之前，a[1~(i-1)]应该与b[1~(j-1)]匹配
            f[i-1][j-1] + 1
            但是如果本来a[i]与b[j]这一位上就相等，那么不用改，即
            f[i-1][j-1] + 0

好的那么f[i][j]就由以上三个可能状态转移过来，取个min
*/
```

3. **初始化：**$f_{0\ i} = f_{i\ 0} =i,f_{i\ j} = inf$
4. **答案：**$f_{n\ m}$

```cpp
#include <bits/stdc++.h>
#define inf 1000000000
#define maxn 1010
using namespace std;

int lena, lenb, f[maxn][maxn]; //a的1~i转成b的1~j
char a[maxn], b[maxn];
int main() {
    scanf("%d%s", &lena, a + 1);
    scanf("%d%s", &lenb, b + 1);
    for (int i = 1; i <= lena; i++) 
        for (int j = 1; j <= lenb; j++) 
            f[i][j] = inf;
    for (int i = 1; i <= lena; i++) f[i][0] = i;
    for (int i = 1; i <= lenb; i++) f[0][i] = i;
    for (int i = 1; i <= lena; i++) {
        for (int j = 1; j <= lenb; j++) {
            f[i][j] = min(f[i - 1][j] + 1, f[i][j - 1] + 1); //增和删
            if (a[i] == b[j]) f[i][j] = min(f[i][j], f[i - 1][j - 1]); //不改
            else f[i][j] = min(f[i][j], f[i - 1][j - 1] + 1); //改
        }
    }
    printf("%d\n", f[lena][lenb]);
    return 0;
}
```

## [编辑距离](https://www.acwing.com/problem/content/description/904/)

其实就是把上一题重写了一遍，时间复杂度 $Θ(nms^2),s\text{是字符串的长度}$。

```cpp
#include <bits/stdc++.h>
#define inf 1000000000
#define maxn 1010
using namespace std;

int n, m, s, f[maxn][maxn]; //a的1~i转成b的1~j
char a[maxn][maxn], b[maxn];
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) scanf("%s", a[i] + 1);
    while(m--) {
        int res = 0;
        scanf("%s%d", b + 1, &s);
        for (int k = 1; k <= n; k++) {
            int lena = strlen(a[k] + 1), lenb = strlen(b + 1);
            for (int i = 1; i <= lena; i++) 
                for (int j = 1; j <= lenb; j++) 
                    f[i][j] = inf;
            for (int i = 1; i <= lena; i++) f[i][0] = i;
            for (int i = 1; i <= lenb; i++) f[0][i] = i;
            for (int i = 1; i <= lena; i++) {
                for (int j = 1; j <= lenb; j++) {
                    f[i][j] = min(f[i - 1][j] + 1, f[i][j - 1] + 1); //增和删
                    if (a[k][i] == b[j]) f[i][j] = min(f[i][j], f[i - 1][j - 1]); //不改
                    else f[i][j] = min(f[i][j], f[i - 1][j - 1] + 1); //改
                }
            }
            res += f[lena][lenb] <= s;
        }
        cout << res << endl;
    }
    return 0;
}
```

# 数位统计DP

## [计数问题](https://www.acwing.com/problem/content/340/)

**分类讨论：**

比如说要找 $[1,\overline{abcdefg}]$ 中的数中 $1$ 出现的个数。
就得先求 $1$ 在每一个位置上出现的次数。

比如我要找第 $4$ 位上出现的 $1$ 的数有几个
就是要找满足 $1\le \overline{xxx1yyy}\le \overline{abcdefg}$。

1. $$\overline{xxx}∈[000,\overline{abc}-1],\overline{yyy}∈[000,999],ans = ans+\overline{abc}*1000$$，如果前三位没填满，则后三位就可以随便填。

2. $\overline{xxx}=\overline{abc},yyy∈...$

   1. $d<1, \overline{yyy}\text{ 不存在},ans=ans+0$。
   2. $d=1,\overline{yyy}∈[000,\overline{efg}] , ans=ans+\overline{efg}+1$。
   3. $d>1,\overline{yyy}∈[000,999],ans=ans+1000$。

   如果前三位填满了，后三位怎么填取决于当前这一位。

这样就是求出来了 $[1,n]$ 的，如果想求 $[l,r]$ 的用一下前缀和就搞定了。

**但还是需要注意一些细节：**

1. $x(x\ne 0)$ 在第 $1$ 位上出现的次数（不用考虑前半段），那就忽略第一种情况。
2. $x=0$。$0$ 不能在第一位，且如果在第 $i$ 位枚举 $0$，那必须保证前 $i-1$ 位不是 $\underbrace{000...}_{i-1}$。

```cpp
#include <bits/stdc++.h>
#define maxn 18
#define int long long
using namespace std;

int a, b, cnt[maxn];
int get(int r, int l) { //获取从r到l这一串的数字
    int sum = 0;
    for (int i = r; i >= l; i--) sum = sum * 10 + cnt[i];
    return sum;
}

int ksm(int x, int s) { //快速冥
    if(s == 0) return 1;
    int t = ksm(x, s / 2);
    if(s & 1) return t * t * x;
    else return t * t;
}

int work(int n, int x) {
    if(n == 0) return 0;
    int len = 0, ans = 0;
    while(n) cnt[++len] = n % 10, n /= 10; //把每一位数都记录下来，低位的下标较低
    if(x) { //x是不是0
        for (int i = len; i >= 1; i--) { //从高位枚举低位
            if(i < len) ans += get(len, i + 1) * ksm(10, i - 1); //如果不是首位，则考虑第一种情况
            if(cnt[i] == x) ans += get(i - 1, 1) + 1; //接下俩考虑第二种情况
            else if(cnt[i] > x) ans += ksm(10, i - 1);
            else ans += 0;
        }
    } else {
        for (int i = len - 1; i >= 1; i--) { //从高位枚举低位
            ans += (get(len, i + 1) - 1) * ksm(10, i - 1); //由于不枚举首位，因此不需要特判
            if(cnt[i] == x) ans += get(i - 1, 1) + 1;
            else if(cnt[i] > x) ans += ksm(10, i - 1);
            else ans += 0;
        }
    }
    return ans;
}

void solve() {
    if(a > b) swap(a, b);
    for (int i = 0; i <= 9; i++) cout << work(b, i) - work(a - 1, i) << " "; //前缀和
    cout << endl;
}

signed main() {
    while(cin >> a >> b && (a || b)) solve();
    return 0;
}
```

# 状态压缩DP

## [蒙德里安的梦想](https://www.acwing.com/problem/content/293/)

摆放方块的时候，先放横着的，再放竖着的。总方案数等于只放横着的小方块的合法方案数。

如何判断，当前方案数是否合法？ 所有剩余位置能否填充满竖着的小方块。可以按列来看，每一列内部所有连续的空着的小方块需要是偶数个。

这是一道动态规划的题目，并且是一道 状态压缩的 `DP`：用一个 $n$ 位的二进制数，每一位表示一个物品，$0/1$ 表示不同的状态。因此可以用 $0→2^N−1$（$n$ 是二进制对应的十进制数）中的所有数来枚举全部的状态。

1. **状态：**$f_{i\ j}$ 表示已经将前 $i-1$ 列摆好，且从第 $i−1$ 列，**伸出**到第 $i$ 列的状态是 $j$ 的所有方案。
2. **状态转移方程：**

考虑枚举每一列进行 `DP`，之后枚举第 $i-1$ 列伸到第 $i$ 列的状态 $j$，再枚举第 $i-2$ 列伸到第 $i-1$ 列的状态 $k$。

但我们得保证它们中不能叠在一起，因此，`j & k`必须的保证为 $0$，因为如果不为 $0$ 说明肯定有一位上都是 $1$，在同一个位置，这样就不合法了。且要保证它们的间隔之间必须隔着的是偶数（否则竖着的小方块放不进去），可以通过 `j | k` 得到它们交在一起的状态，之后使用一个数组 `st[]` 表示一个状态是否合法就行了。

```cpp
for (int i = 1; i <= m; i++) //枚举每一列
    for (int j = 0; j < 1 << n; j++)
        for (int k = 0; k < 1 << n; k++)
            if ((j & k) == 0 && st[j | k]) f[i][j] += f[i - 1][k]; //计算总和
```

3. **初始化：**$f_{i\ j}=\inf,f_{0\ 0}=1$，这里需要回忆状态表示的定义
   按定义这里是：前第 $-1$ 列都摆好，且从 $-1$ 列到第 $0$ 列伸出来的状态为 $0$ 的方案数。
   首先，这里没有 $-1$ 列，最少也是 $0$ 列。
   其次，没有伸出来，即没有横着摆的。即这里第 $0$ 列只有竖着摆这 $1$ 种状态。
4. **答案：**$f_{m\ 0}$，最后一列伸出来为 $0$。

```cpp
#include <bits/stdc++.h>
#define maxn 12
#define int long long
using namespace std;

int n, m, f[maxn][1 << maxn];
bool st[1 << maxn];
void solve() {
    memset(st, 0, sizeof(st));
    memset(f, 0, sizeof(f));
    for (int i = 0; i < 1 << n; i++) { //枚举合法方案
        int cnt = 0;
        st[i] = 1;
        for (int j = 0; j < n; j++) {
            if(i >> j & 1) {
                if(cnt & 1) st[i] = 0;
                cnt = 0;
            }
            else cnt++;
        }
        if(cnt & 1) st[i] = 0;
    }
    f[0][0] = 1; //这里需要回忆状态表示的定义
    //按定义这里是：前第-1列都摆好，且从-1列到第0列伸出来的状态为0的方案数。
    //首先，这里没有-1列，最少也是0列。
    //其次，没有伸出来，即没有横着摆的。即这里第0列只有竖着摆这1种状态。
    for (int i = 1; i <= m; i++)
        for (int j = 0; j < 1 << n; j++)
            for (int k = 0; k < 1 << n; k++)
                if ((j & k) == 0 && st[j | k]) f[i][j] += f[i - 1][k];
    cout << f[m][0] << endl;
}

signed main() {
    while(cin >> n >> m, n || m) solve();
    return 0;
}
```

## [最短Hamilton路径](https://www.acwing.com/problem/content/93/)

1. **状态：**$f_{i\ j}$ 所有从 $0$ 走到 $j$，走过的所有点的情况是 $i$ 的所有路径

2. **状态转移方程：**

```cpp
for (int i = 0; i < 1 << n; i++)
    if (i & 1) //必须包含起点0
        for (int j = 0; j < n; j++)
            if(i >> j & 1) //确保j在
                for (int k = 0; k < n; k++)
                    if (i >> k & 1) //确保j的邻接点k在
                        f[i][j] = min(f[i][j], f[i - (1 << j)][k] + w[k][j]); //去掉j这种状态，且算上0->k的最短路径与(j, k)距离
//为什么要先枚举状态？因为点j的计算可能会依赖比它大或者比它小的点k，它所依赖的点的状态可能还没有被计算，所以外层循环不能按照点的顺序来枚举，从小到大，从大到小都不行。但是状态i的计算，只会依赖比它更小的状态j，因为需要把当前点从状态i中减去，所以外层循环要枚举状态，并且要从小到大枚举。
```

3. **初始化：**$f_{i\ j} = \inf,f_{1\ 0}=0$（什么都不选也是一种方案）
4. **答案：**$f_{2^n-1\ \ n-1}$

```cpp
#include <bits/stdc++.h>
#define maxn 21
using namespace std;

int n, w[maxn][maxn], f[1 << maxn][maxn];
signed main() {
    memset(f, 0x3f, sizeof(f));
    cin >> n;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            cin >> w[i][j];
    f[1][0] = 0;
    for (int i = 0; i < 1 << n; i++)
        if (i & 1) //必须包含起点0
            for (int j = 0; j < n; j++)
                if(i >> j & 1) //确保j在
                    for (int k = 0; k < n; k++)
                        if (i >> k & 1) //确保j的邻接点k在
                            f[i][j] = min(f[i][j], f[i - (1 << j)][k] + w[k][j]);
    cout << f[(1 << n) - 1][n - 1] << endl;
    return 0;
}
```

# 树形DP

## [没有上司的舞会](https://www.acwing.com/problem/content/287/)

1. **状态：**$f_{i\ j}$，$j$ 是 $0/1$，表示 $i$ 这个人是否来的最大获益值。

2. **状态转移方程：**

```cpp
f[x][0] += max(f[nxt][0], f[nxt][1]); //如果不来，那么下属来不来就没有关系了
f[x][1] += f[nxt][0]; //如果来，那么下属只能不来了
```

3. **初始化：**$f_{i\ j}=0$
4. **答案：**$\max\{f_{1\ 0},f_{1\ 1}\}$

```cpp
#include <iostream>
using namespace std;

int n, ans, a[6010], f[6010][2];
int e, To[6010], Begin[6010], Next[6010], Deg[6010], Fa[6010];
void dfs(int x) {
    for (int i = Begin[x]; i; i = Next[i]) {
        int nxt = To[i];
        dfs(nxt);
        f[x][0] += max(f[nxt][0], f[nxt][1]);
        f[x][1] += f[nxt][0];
    }
    f[x][1] += a[x];
}

void add(int u, int v) {
    e++;
    To[e] = v;
    Next[e] = Begin[u];
    Begin[u] = e;
    Fa[v] = u;
    Deg[u]++;
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i < n; i++) {
        int l, k;
        cin >> l >> k;
        add(k, l);
    }
    for (int i = 1; i <= n; i++) {
        if (!Fa[i]) {
            dfs(i);
            ans = max(f[i][0], f[i][1]);
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
```

# 记忆化搜索

## [滑雪](https://www.acwing.com/problem/content/903/)

记忆化搜索加快速度。

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, m, ans, a[310][310], f[310][310];
int dx[4] = { 0, 1, 0, -1 };
int dy[4] = { 1, 0, -1, 0 };
int dfs(int x, int y) {
    if (f[x][y]) return f[x][y]; //记忆化搜索
    int sum = 0;
    for (int i = 0; i < 4; i++) {
        int tx = dx[i] + x, ty = dy[i] + y;
        if (tx < 1 || tx > n || ty < 1 || ty > m || a[x][y] <= a[tx][ty]) continue;
        else sum = max(sum, dfs(tx, ty));
    }
    f[x][y] = sum + 1;
    return f[x][y];
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            cin >> a[i][j];
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            ans = max(ans, dfs(i, j));
    cout << ans << endl;
    return 0;
}
```
