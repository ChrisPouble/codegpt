# T3

题目要求把字符串里面所有的“WA”都换成“AC”，看似较为简单，但是有个例子。

```cpp
WWA
```

一眼看过去，直接报答案：

```cpp
WAC
```

然而实际答案却是：

```cpp
AAC
```

因此，我们考虑处理出每一个形如 $\texttt{WWW...WA}$ 的字符串，之后把它们替换为 $\texttt{AAA...AC}$ 即可。

```cpp
#include <bits/stdc++.h>
#define maxn 300010
using namespace std;

int cnt;
string ttttt; 
char s[maxn];
struct node {
	int l, r;
} a[maxn];

signed main() {
	cin >> ttttt;
	int len = ttttt.size();
	for (int i = len; i >= 1; i--) s[i] = ttttt[i - 1]; //弄出字符数组
	for (int i = len; i >= 1; i--) {
		if(s[i] != 'A' || i == 1) continue;
		int top = i;
		while(s[top - 1] == 'W') top--;
		if(top < i) a[++cnt] = {top, i};
		if(top > 1) top--;
		while(top >= 1 && s[top] != 'A') top--;
		i = top + 1;
	}
	for (int i = 1; i <= cnt; i++) {
		int l = a[i].l, r = a[i].r;
		for (int k = l + 1; k <= r; k++) s[k] = 'C';
		s[l] = 'A';
	}
	for (int i = 1; i <= len; i++) cout << s[i];
	return 0;
}
```

# T5

由于 $n\le 100$，可容纳 $O(n^4)$ 的时间复杂度，因此考虑使用广搜完成。

开局枚举 $i,j$，若它们之间有连边，则 $i\leftrightarrow j$ 的最短回文路径为 $1$。

之后，把 $i,j$ 都放入队列里，两个变量一起往外扩展。例如，$i$ 扩展到了 $u$，$j$ 扩展到了 $v$，则 $u\leftrightarrow v=(i\leftrightarrow j) + 2$。

```cpp
#include <bits/stdc++.h>
#define maxn 105
using namespace std;

int n;
char c[maxn][maxn];
int dis[maxn][maxn];
struct node {
	int u, v;
};

queue<node> q;
signed main() {
    cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> (c[i] + 1);
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            dis[i][j] = -1;
        }
    }
    for (int i = 1; i <= n; i++) {
        dis[i][i] = 0;
        q.push({i, i});
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (i != j && c[i][j] != '-') {
                dis[i][j] = 1;
                q.push({i, j});
            }
        }
    }
    while (!q.empty()) {
        int u = q.front().u;
		int v = q.front().v;
		q.pop();
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= n; j++)
                if (dis[i][j] == -1 && c[i][u] == c[v][j] && c[i][u] != '-') {
                    dis[i][j] = dis[u][v] + 2;
                    q.push({i, j});
                }
    }
    for (int i = 1; i <= n; i++, puts(""))
        for (int j = 1; j <= n; j++)
            cout << dis[i][j] << ' ';
    return 0;
}
```

# T6

树形 DP，考虑我们转移时需要什么信息：

- 当前子树根节点（当前结点）的度数；
- 子树里是否有度为 4 的点。

所以可令 $ f_{i,j,0/1} $ 表示以 $ i $ 为根，$ i $ 的度数为 $ j $，选择的树内有无度数为 4 的点时，这棵树最大的大小。

转移设计：

初始 $ f_{u,0,0} \leftarrow 1 $ （一个点当一棵树）；

$ f_{u,i,0} \leftarrow \max(f_{u,i-1,0} + 1)(\text{card}(son \text{ of } u) > 0) $（$ u $ 有儿子，这个儿子接到 $ u $ 的下面）

$ f_{u,i,1} \leftarrow \max(f_{u,i-1,1} + 1)(\text{card}(son \text{ of } u) > 0) $（同上）

$ f_{u,i,1} \leftarrow \max(f_{u,i-1,0}, f_{u,i-1,1}) + \max(f_{v,3,0}, f_{v,3,1})(v \in son \text{ of } u) $（$ v $ 作为一个儿子，原先度为 $3$，接上 $ u $ 之后变成 $4$）

$ \text{answer} = \max(f_{u,1,1}, f_{u,4,0}, f_{u,4,1}) $ (需要保证有度数为 $4$ 的点)

搜一遍就好了，时间复杂度 $ O(n) $。转移时要判断值不为零，免得出现虚空儿子充当度数的情况。

```cpp
#include <bits/stdc++.h>
#define maxn 200010 
using namespace std;

int n, u, v, ans, f[maxn][5][2];
vector<int> linker[maxn];
inline int max(int a,int b,int c,int d) {
	return max(max(a, b), max(c, d));
}

void dfs(int u, int fa) {
	f[u][0][0] = 1;
	for (auto v:linker[u]) {
		if(v == fa) continue;
		dfs(v, u);
		for (int i = 4; i >= 1; i--) { //倒序转移，防止需要的值被覆盖
			if (f[u][i - 1][0]) f[u][i][0] = max(f[u][i][0], f[u][i - 1][0] + 1);
			if (f[u][i - 1][1]) f[u][i][1] = max(f[u][i][1], f[u][i - 1][1] + 1);
			if ((f[u][i - 1][0] || f[u][i - 1][1]) && (f[v][3][0] || f[v][3][1]))
				f[u][i][1] = max(f[u][i][1], max(f[u][i - 1][0], f[u][i - 1][1]) + max(f[v][3][0], f[v][3][1]));
		}
	}
	ans = max(ans, f[u][1][1], f[u][4][0], f[u][4][1]);
	return;
}

signed main() {
	cin >> n;
	for (int i = 1; i < n; i++) {
		cin >> u >> v;
		linker[u].push_back(v);
		linker[v].push_back(u);
	}
	dfs(1, 0);
	cout << (ans ? ans : -1) << endl;
	return 0;
}
```
