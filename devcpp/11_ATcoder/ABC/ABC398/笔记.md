# T3

把每个数值都记录在桶里，桶存储的就是每种数值的数量。之后从大到小排序，枚举最大值、次大值...，直到发现与自己相等的数值超高过一，就输出答案。

```cpp
#include <bits/stdc++.h>
#define maxn 300010
using namespace std;

int n;
map<int, int>t;
struct node {
	int val, id;
} a[maxn];

bool cmp(node x, node y) {
	return x.val < y.val;
}

signed main() {
	cin >> n;
	for (int i = 1; i <= n; i++) {
		cin >> a[i].val;
		a[i].id = i, t[a[i].val]++;
	}
	sort(a + 1, a + n + 1, cmp);
	for (int i = n; i >= 1; i--)
		if(t[a[i].val] == 1) {
			cout << a[i].id << endl;
			return 0;
		}
	cout << -1 << endl;
	return 0;
}
```

# T4

烟雾太多不好做，考虑移动高桥和篝火。

记录下篝火的路径，若高桥走到了篝火到过的地方就有烟雾，否则没有。

注意高桥和篝火的移动应该跟烟雾反着来。

```cpp
#include<bits/stdc++.h>
#define int long long
#define mp make_pair
using namespace std;

int n, r, c, x, y;
string s;
map<pair<int, int>, int> vis;
signed main() {
	cin >> n >> r >> c >> s;
	s = ' ' + s, x = 0, y = 0, vis[mp(x, y)] = 1;
	for (int i = 1; i <= n; i++) {
		if(s[i] == 'N') {
			r++, x++;
			vis[mp(x, y)] = 1;
		} else if(s[i] == 'S') {
			r--, x--;
			vis[mp(x, y)] = 1;
		} else if(s[i] == 'W') {
			c++, y++;
			vis[mp(x, y)] = 1;
		} else {
			c--, y--;
			vis[mp(x, y)] = 1;
		}
		cout << vis[mp(r, c)];
	}
	return 0;
}
```

# T6

这道题虽然是道绿题，但是只要求 `S+'#'+S'` 的 KMP 就行了，其中的 `S'` 是 `S` 的反转。

```cpp
#include <bits/stdc++.h>
#define N 1000010
#define M 1000010
using namespace std;

int n, m, ne[N];
string s, r;
signed main() {
    cin >> s;
    for (int i = s.size() - 1; i >= 0; i--) r += s[i];
    string T = " " + r + "#" + s;
    for (int i = 2, j = 0; i < T.size(); i++) {
        while(j && T[i] != T[j + 1]) j = ne[j];
        if(T[i] == T[j + 1]) j++;
        ne[i] = j;
    }
    cout << s;
    for (int i = ne[T.size() - 1]; i < r.size(); i++) cout << r[i];
	return 0;
}
```
