# T2

按题目模拟即可，没什么可讲的。

```cpp
#include <bits/stdc++.h>
#define maxn 110
using namespace std;

int n, r = 1, later = maxn, a[maxn], ans[maxn];
int query_max() {
	int res = 0;
	for (int i = 1; i <= n; i++)
		if(later > a[i]) res = max(res, a[i]);
	return res;
}

signed main() {
	cin >> n;
	for (int i = 1; i <= n; i++) cin >> a[i];
	while(r <= n) {
		int ma = query_max(), k = 0;
		for (int j = 1; j <= n; j++)
			if(a[j] == ma) k++, ans[j] = r;
		later = ma, r += k;
	}
	for (int i = 1; i <= n; i++) cout << ans[i] << endl;
	return 0;
}
```

# T3

神奇的并查集前来助阵！

```cpp
#include <bits/stdc++.h>
#define maxn 200010
using namespace std;

int n, m, ans, fa[maxn];
struct Edge {
	int x, y;
} edge[maxn];

int find(int x) {
	if(x == fa[x]) return x;
	else return fa[x] = find(fa[x]);
}

void merge(int x, int y) {
	int xx = find(x), yy = find(y);
	fa[xx] = yy;
}

signed main() {
	cin >> n >> m;
	for (int i = 1; i <= n; i++) fa[i] = i;
	for (int i = 1, x, y; i <= m; i++)
		cin >> edge[i].x >> edge[i].y;
	for (int i = 1; i <= m; i++) {
		int x = edge[i].x, y = edge[i].y;
		if(find(x) != find(y)) merge(x, y);
		else ans++;
	}
	cout << ans << endl;
	return 0;
}
```

# T4

考虑两个相邻的整数对：$(a_{i-1}, a_i)$。原因是假如我们考虑不相邻的两对，根据题目的交换规则也没用。

然后我们要把另一个 $a_{i-1}$ 换到 $i$ 的位置。那么我们需要预处理出上一个 $a_{i-1}$ 的位置。有人或许会问：假如这是第一个 $a_{i-1}$ 怎么办？第一个 $a_{i-1}$ 我们把上一个的位置设为 0，直接判掉。

我们设上一个 $a_{i-1}$ 位置为 $x$。那么假如 $x$ 的相邻两个位置有 $a_i$，把 $a_i$ 换到位置 $x$ 之后就可以满足条件，反之不能。

注意：假如 $x = i - 2$，那么要排除，因为 $a_{i-1}$ 的一对已经相邻。

```cpp
#include <bits/stdc++.h>
#define maxn 200010
using namespace std;

int T, n, a[2 * maxn], p[2 * maxn], vis[2 * maxn];
void solve() {
	int ans = 0;
	for (int i = 1; i <= 2 * n; i++) p[i] = vis[i] = 0;
	cin >> n;
	for (int i = 1; i <= 2 * n; i++) cin >> a[i];
	for (int i = 1; i <= 2 * n; i++)
		if (vis[a[i]] == 0) vis[a[i]] = i;
		else p[i] = vis[a[i]];
	for (int i = 2; i <= 2 * n; i++)
		if(p[i - 1]) {
			int x = p[i - 1];
			if(x == i - 2) continue; //p[i - 1]与i-1相邻 
			if(a[x + 1] == a[i] || a[x - 1] == a[i]) ans++;
		}
	cout << ans << endl;
}

signed main() {
	cin >> T;
	while(T--) solve();
	return 0;
}
```

# T5

首先让 $S_i$ 连向 $T_i$，构成一个图。

如果没有环，可以发现答案就是图的边数。

如果一个点的出度 $\ge 2$，则没有答案，输出 $-1$。

接下来判断有环情况：

- > 若是一个纯环，如下图，则答案为边数 $+1$。
  >
  > ![](https://cdn.luogu.com.cn/upload/image_hosting/678o6qa1.png)
  >
  > 按照这个图我们可以写一个样例：
  >
  > ```cpp
  > abc
  > bca
  > ```
  >
  > 可以按照如此流程：`abc -> abd -> acd -> bcd -> bca`。
  >
  > 得出，如果是这种纯环，则答案为边数 $+1$。
  >
  > 但是，还要注意一点，如果 $26$ 个字母全在环上，那么就没有额外字母进行一个代替，只能输出 $-1$。

- > 如果一个杂环，如下图，则答案为边数。
  >
  > ![](https://cdn.luogu.com.cn/upload/image_hosting/ik43ifwn.png)
  >
  > 很明显，这就是纯环加上了额外字母之后的图，所以答案为边数。

```cpp
#include <bits/stdc++.h>
#define maxn 130
using namespace std;

int n, ans;
string s, t; 
char to[maxn];
int in[maxn];
bool f[maxn];
queue<char> q;
signed main() {
	cin >> n >> s >> t;
	if(s == t) {
		cout << 0 << endl;
		return 0;
	}
	s = " " + s, t = " " + t;
	for (int i = 1; i <= n; i++) {
		if(to[s[i]]) {
			if(to[s[i]] != t[i]) { //多分支 
				cout << -1 << endl;
				return 0;
			}
		} else {
			to[s[i]] = t[i];
			in[t[i]]++; //入度++ 
		}
	}
	for (char c = 'a'; c <= 'z'; c++)
		if(in[(int)c] == 0) q.push(c); //拓扑排序
	if(q.empty()) { //26个字母全在环上
		cout << -1 << endl;
		return 0;
	}
	while(!q.empty()) {
		char x = q.front();
		q.pop();
		if(to[(int)x]) ans++;
		in[to[(int)x]]--;
		if(in[(int)to[(int)x]] == 0) q.push((int)to[(int)x]);
		else f[(int)to[(int)x]] = 1; //说明入度 > 1（杂环的汇入点） 
	}
	for (char c = 'a'; c <= 'z'; c++) {
		if(in[(int)c] == 0) continue; //不在环上 
		bool flag = 0;
		char pos = to[(int)c];
		if(c == pos) continue; //自环（不需要管） 
		while(in[(int)pos]) {
			ans++;
			in[pos]--;
			if(f[pos]) flag = 1; //太好了，又杂环，我们有救了
			pos = to[(int)pos];
		}
		if(flag == 0) ans++; //没有杂环
	}
	cout << ans << endl;
	return 0;
}
```
